   166→ApplicationRuntime::ApplicationRuntime() {
   167→    start_time_ = std::chrono::steady_clock::now();
   168→    initialize_subsystems();
   169→    
   170→    StateTransition initial;
   171→    initial.from_state = RuntimeState::CREATED;
   172→    initial.to_state = RuntimeState::CREATED;
   173→    initial.timestamp = get_timestamp();
   174→    initial.reason = "ApplicationRuntime initialized";
   175→    initial.success = true;
   176→    state_transitions_.push_back(initial);
   177→    
   178→    if (config_.verbose) {
   179→        std::cout << "[MiniAndroid] ApplicationRuntime created" << std::endl;
   180→    }
   181→}
   182→
   183→ApplicationRuntime::~ApplicationRuntime() {
   184→    cleanup();
   185→    
   186→    auto end = std::chrono::steady_clock::now();
   187→    total_duration_ms_ = std::chrono::duration<double, std::milli>(end - start_time_).count();
   188→    
   189→    if (config_.verbose) {
   190→        std::cout << "[MiniAndroid] ApplicationRuntime destroyed (ran for " 
   191→                  << total_duration_ms_ << "ms)" << std::endl;
   192→    }
   193→}

   217→std::string ApplicationRuntime::get_timestamp() const {
   218→    auto now = std::chrono::system_clock::now();
   219→    auto time_t = std::chrono::system_clock::to_time_t(now);
   220→    std::stringstream ss;
   221→    ss << std::put_time(std::gmtime(&time_t), "%Y-%m-%dT%H:%M:%SZ");
   222→    return ss.str();
   223→}

   229→bool ApplicationRuntime::transition_to(RuntimeState new_state, const std::string& reason) {
   230→    auto start = std::chrono::steady_clock::now();
   231→    RuntimeState from = state_;
   232→    
   233→    state_ = new_state;
   234→    
   235→    auto end = std::chrono::steady_clock::now();
   236→    double duration = std::chrono::duration<double, std::milli>(end - start).count();
   237→    
   238→    record_transition(from, new_state, reason, true, duration);
   239→    
   240→    if (config_.verbose) {
   241→        std::cout << "[State] " << state_to_string(from) << " → " 
   242→                  << state_to_string(new_state) << ": " << reason << std::endl;
   243→    }
   244→    
   245→    return true;
   246→}
   247→
   248→void ApplicationRuntime::record_transition(RuntimeState from, RuntimeState to, 
   249→                                          const std::string& reason,
   250→                                          bool success, double duration_ms) {
   251→    StateTransition trans;
   252→    trans.from_state = from;
   253→    trans.to_state = to;
   254→    trans.timestamp = get_timestamp();
   255→    trans.reason = reason;
   256→    trans.success = success;
   257→    trans.duration_ms = duration_ms;
   258→    state_transitions_.push_back(trans);
   259→}

   270→bool ApplicationRuntime::execute_apk(const std::string& apk_path, const RuntimeConfig& config) {
   271→    config_ = config;
   272→    apk_path_ = apk_path;
   273→    
   274→    if (config_.verbose) {
   275→        std::cout << "\n========== MiniAndroid Runtime v0.1 ==========" << std::endl;
   276→        std::cout << "Executing: " << apk_path << std::endl;
   277→        std::cout << "==============================================\n" << std::endl;
   278→    }
   279→    
   280→    try {
   281→        // Phase A: Load APK and resolve basics
   282→        if (!load_apk(apk_path)) return false;
   283→        if (!resolve_manifest()) return false;
   284→        if (!load_dex()) return false;
   285→        
   286→        // Phase B: Resolve and create application
   287→        if (!resolve_launcher_activity()) return false;
   288→        if (!create_application()) return false;
   289→        
   290→        // Phase C: Execute DEX lifecycle methods
   291→        if (!execute_lifecycle()) return false;
   292→        
   293→        // Phase E: Resources
   294→        if (!initialize_resources()) return false;
   295→        if (!load_content_view()) return false;
   296→        
   297→        // Phase F: Layout
   298→        if (!perform_layout()) return false;
   299→        
   300→        // Phase H: Rendering
   301→        if (!render_frame()) return false;
   302→        
   303→        // Complete!
   304→        transition_to(RuntimeState::COMPLETED, "Pipeline completed successfully");
   305→        
   306→        if (config_.save_screenshot) {
   307→            save_screenshot(config_.screenshot_path);
   308→        }
   309→        
   310→        if (config_.generate_evidence) {
   311→            write_all_evidence(config_.output_dir);
   312→        }
   313→        
   314→        return true;
   315→        
   316→    } catch (const std::exception& e) {
   317→        FailureRecord f;
   318→        f.type = FailureType::INTERNAL_ERROR;
   319→        f.stage = "execute_apk";
   320→        f.details = e.what();
   321→        f.timestamp = get_timestamp();
   322→        f.apk_path = apk_path_;
   323→        f.is_blocking = true;
   324→        record_failure(f);
   325→        transition_to(RuntimeState::ERROR, std::string("Exception: ") + e.what());
   326→        return false;
   327→    }
   328→}

   333→
   334→bool ApplicationRuntime::load_apk(const std::string& apk_path) {
   335→    auto start = std::chrono::steady_clock::now();
   336→    
   337→    try {
   338→        if (config_.verbose) {
   339→            std::cout << "[Phase A] Loading APK: " << apk_path << std::endl;
   340→        }
   341→        
   342→        // Parse APK
   343→        ApkInfo info = apk_parser_->parse(apk_path);
   344→        
   345→        if (!info.is_valid) {
   346→            FailureRecord f;
   347→            f.type = FailureType::FILE_NOT_FOUND;
   348→            f.stage = "load_apk";
   349→            f.details = apk_parser_->get_last_error();
   350→            f.timestamp = get_timestamp();
   351→            f.apk_path = apk_path;
   352→            f.is_blocking = true;
   353→            record_failure(f);
   354→            transition_to(RuntimeState::LOAD_FAILED, "APK parsing failed: " + apk_parser_->get_last_error());
   355→            return false;
   356→        }
   357→        
   358→        apk_info_ = std::make_unique<ApkInfo>(info);
   359→        
   360→        // Read full APK data into memory for later extraction
   361→        std::ifstream file(apk_path, std::ios::binary | std::ios::ate);
   362→        if (!file.is_open()) {
   363→            FailureRecord f;
   364→            f.type = FailureType::FILE_NOT_FOUND;
   365→            f.stage = "load_apk";
   366→            f.details = "Cannot open file for reading";
   367→            f.timestamp = get_timestamp();
   368→            f.apk_path = apk_path;
   369→            f.is_blocking = true;
   370→            record_failure(f);
   371→            transition_to(RuntimeState::LOAD_FAILED, "Cannot read APK file");
   372→            return false;
   373→        }
   374→        
   375→        size_t size = file.tellg();
   376→        file.seekg(0, std::ios::beg);
   377→        apk_data_.resize(size);
   378→        file.read(reinterpret_cast<char*>(apk_data_.data()), size);
   379→        file.close();
   380→        
   381→        // Extract manifest data
   382→        manifest_data_ = apk_parser_->extract_entry(apk_path, "AndroidManifest.xml");
   383→        
   384→        auto end = std::chrono::steady_clock::now();
   385→        double duration = std::chrono::duration<double, std::milli>(end - start).count();
   386→        
   387→        transition_to(RuntimeState::APK_LOADED, "APK loaded successfully: " + apk_path);
   388→        
   389→        if (config_.verbose) {
   390→            std::cout << "  Package: " << info.package_name << std::endl;
   391→            std::cout << "  Main Activity: " << info.main_activity_full << std::endl;
   392→            std::cout << "  Size: " << size << " bytes" << std::endl;
   393→            std::cout << "  Duration: " << duration << "ms" << std::endl;
   394→        }
   395→        
   396→        return true;
   397→        
   398→    } catch (const std::exception& e) {
   399→        FailureRecord f;
   400→        f.type = FailureType::PARSE_ERROR;
   401→        f.stage = "load_apk";
   402→        f.details = e.what();
   403→        f.timestamp = get_timestamp();
   404→        f.apk_path = apk_path;
   405→        f.is_blocking = true;
   406→        record_failure(f);
   407→        transition_to(RuntimeState::LOAD_FAILED, std::string("Exception: ") + e.what());
   408→        return false;
   409→    }
   410→}

   412→bool ApplicationRuntime::resolve_manifest() {
   413→    auto start = std::chrono::steady_clock::now();
   414→    
   415→    try {
   416→        if (config_.verbose) {
   417→            std::cout << "[Phase A] Resolving manifest..." << std::endl;
   418→        }
   419→        
   420→        if (manifest_data_.empty()) {
   421→            FailureRecord f;
   422→            f.type = FailureType::PARSE_ERROR;
   423→            f.stage = "resolve_manifest";
   424→            f.details = "No manifest data available";
   425→            f.timestamp = get_timestamp();
   426→            f.apk_path = apk_path_;
   427→            f.is_blocking = true;
   428→            record_failure(f);
   429→            transition_to(RuntimeState::RESOLUTION_FAILED, "No manifest data");
   430→            return false;
   431→        }
   432→        
   433→        ManifestReader manifest_reader;
   434→        manifest_reader.set_verbose(config_.verbose);
   435→        ManifestInfo manifest_info = manifest_reader.parse(manifest_data_);
   436→        
   437→        if (!manifest_info.parse_success) {
   438→            FailureRecord f;
   439→            f.type = FailureType::PARSE_ERROR;
   440→            f.stage = "resolve_manifest";
   441→            f.details = manifest_info.error_message;
   442→            f.timestamp = get_timestamp();
   443→            f.apk_path = apk_path_;
   444→            f.is_blocking = true;
   445→            record_failure(f);
   446→            transition_to(RuntimeState::RESOLUTION_FAILED, "Manifest parsing failed");
   447→            return false;
   448→        }
   449→        
   450→        manifest_info_ = std::make_unique<ManifestInfo>(manifest_info);
   451→        
   452→        auto end = std::chrono::steady_clock::now();
   453→        double duration = std::chrono::duration<double, std::milli>(end - start).count();
   454→        
   455→        transition_to(RuntimeState::MANIFEST_RESOLVED, "Manifest resolved: " + manifest_info.package_name);
   456→        
   457→        if (config_.verbose) {
   458→            std::cout << "  Package: " << manifest_info.package_name << std::endl;
   459→            std::cout << "  Activities: " << manifest_info.activities.size() << std::endl;
   460→            std::cout << "  Duration: " << duration << "ms" << std::endl;
   461→        }
   462→        
   463→        return true;
   464→        
   465→    } catch (const std::exception& e) {
   466→        FailureRecord f;
   467→        f.type = FailureType::PARSE_ERROR;
   468→        f.stage = "resolve_manifest";
   469→        f.details = e.what();
   470→        f.timestamp = get_timestamp();
   471→        f.apk_path = apk_path_;
   472→        f.is_blocking = true;
   473→        record_failure(f);
   474→        transition_to(RuntimeState::RESOLUTION_FAILED, std::string("Exception: ") + e.what());
   475→        return false;
   476→    }
   477→}

   479→bool ApplicationRuntime::load_dex() {
   480→    auto start = std::chrono::steady_clock::now();
   481→    
   482→    try {
   483→        if (config_.verbose) {
   484→            std::cout << "[Phase A] Loading DEX..." << std::endl;
   485→        }
   486→        
   487→        // Extract classes.dex from APK
   488→        dex_data_ = apk_parser_->extract_entry_from_memory(apk_data_, "classes.dex");
   489→        
   490→        if (dex_data_.empty()) {
   491→            FailureRecord f;
   492→            f.type = FailureType::FILE_NOT_FOUND;
   493→            f.stage = "load_dex";
   494→            f.details = "classes.dex not found in APK";
   495→            f.timestamp = get_timestamp();
   496→            f.apk_path = apk_path_;
   497→            f.is_blocking = true;
   498→            record_failure(f);
   499→            transition_to(RuntimeState::LOAD_FAILED, "DEX not found in APK");
   500→            return false;
   501→        }
   502→        
   503→        // Parse DEX
   504→        DexReport report = dex_parser_->parse_data(dex_data_, apk_path_ + "/classes.dex");
   505→        
   506→        if (!report.is_valid) {
   507→            FailureRecord f;
   508→            f.type = FailureType::PARSE_ERROR;
   509→            f.stage = "load_dex";
   510→            f.details = dex_parser_->get_last_error();
   511→            f.timestamp = get_timestamp();
   512→            f.apk_path = apk_path_;
   513→            f.is_blocking = true;
   514→            record_failure(f);
   515→            transition_to(RuntimeState::LOAD_FAILED, "DEX parsing failed");
   516→            return false;
   517→        }
   518→        
   519→        dex_report_ = std::make_unique<DexReport>(report);
   520→        
   521→        auto end = std::chrono::steady_clock::now();
   522→        double duration = std::chrono::duration<double, std::milli>(end - start).count();
   523→        
   524→        transition_to(RuntimeState::DEX_LOADED, "DEX loaded: " + std::to_string(report.classes_count) + " classes");
   525→        
   526→        if (config_.verbose) {
   527→            std::cout << "  Version: " << report.dex_version << std::endl;
   528→            std::cout << "  Classes: " << report.classes_count << std::endl;
   529→            std::cout << "  Methods: " << report.methods_count << std::endl;
   530→            std::cout << "  Duration: " << duration << "ms" << std::endl;
   531→        }
   532→        
   533→        return true;
   534→        
   535→    } catch (const std::exception& e) {
   536→        FailureRecord f;
   537→        f.type = FailureType::PARSE_ERROR;
   538→        f.stage = "load_dex";

   553→bool ApplicationRuntime::resolve_launcher_activity() {
   554→    auto start = std::chrono::steady_clock::now();
   555→    
   556→    try {
   557→        if (config_.verbose) {
   558→            std::cout << "[Phase B] Resolving launcher activity..." << std::endl;
   559→        }
   560→        
   561→        if (!dex_report_) {
   562→            FailureRecord f;
   563→            f.type = FailureType::INVALID_RUNTIME_STATE;
   564→            f.stage = "resolve_launcher_activity";
   565→            f.details = "DEX not loaded";
   566→            f.timestamp = get_timestamp();
   567→            f.apk_path = apk_path_;
   568→            f.is_blocking = true;
   569→            record_failure(f);
   570→            transition_to(RuntimeState::RESOLUTION_FAILED, "DEX not loaded");
   571→            return false;
   572→        }
   573→        
   574→        // Use ClassResolver to find entry point
   575→        ExecutionTrace trace = class_resolver_->resolve(*dex_report_);
   576→        
   577→        execution_trace_ = std::make_unique<ExecutionTrace>(trace);
   578→        
   579→        if (!trace.success || !trace.entry_point.resolved) {
   580→            FailureRecord f;
   581→            f.type = FailureType::METHOD_RESOLUTION_FAILURE;
   582→            f.stage = "resolve_launcher_activity";
   583→            f.details = trace.status_message;
   584→            f.timestamp = get_timestamp();
   585→            f.apk_path = apk_path_;
   586→            f.is_blocking = true;
   587→            record_failure(f);
   588→            transition_to(RuntimeState::RESOLUTION_FAILED, "Cannot resolve launcher activity");
   589→            return false;
   590→        }
   591→        
   592→        entry_point_ = std::make_unique<EntryPoint>(trace.entry_point);
   593→        activity_class_name_ = trace.entry_point.readable_class;
   594→        
   595→        auto end = std::chrono::steady_clock::now();
   596→        double duration = std::chrono::duration<double, std::milli>(end - start).count();
   597→        
   598→        transition_to(RuntimeState::ACTIVITY_RESOLVED, 
   599→                     "Launcher resolved: " + activity_class_name_);
   600→        
   601→        if (config_.verbose) {
   602→            std::cout << "  Activity: " << activity_class_name_ << std::endl;
   603→            std::cout << "  Method: " << trace.entry_point.method_name << std::endl;
   604→            std::cout << "  Duration: " << duration << "ms" << std::endl;
   605→        }
   606→        
   607→        return true;
   608→        
   609→    } catch (const std::exception& e) {
   610→        FailureRecord f;
   611→        f.type = FailureType::CLASS_RESOLUTION_FAILURE;
   612→        f.stage = "resolve_launcher_activity";
   613→        f.details = e.what();
   614→        f.timestamp = get_timestamp();
   615→        f.apk_path = apk_path_;
   616→        f.is_blocking = true;
   617→        record_failure(f);
   618→        transition_to(RuntimeState::RESOLUTION_FAILED, std::string("Exception: ") + e.what());
   619→        return false;
   620→    }
   621→}

   747→// ============================================================================
   748→// Phase C: DEX Execution
   749→// ============================================================================
   750→
   751→bool ApplicationRuntime::execute_on_create() {
   752→    auto start = std::chrono::steady_clock::now();
   753→    
   754→    try {
   755→        if (config_.verbose) {
   756→            std::cout << "[Phase C] executing onCreate via DEX interpreter..." << std::endl;
   757→        }
   758→        
   759→        if (!entry_point_ || !entry_point_->has_bytecode) {
   760→            // Record that we're using stubbed execution
   761→            FailureRecord f;
   762→            f.type = FailureType::UNIMPLEMENTED_API;
   763→            f.stage = "execute_on_create";
   764→            f.details = "Using stubbed onCreate (no bytecode or interpreter limited)";
   765→            f.timestamp = get_timestamp();
   766→            f.apk_path = apk_path_;
   767→            f.dex_method = entry_point_ ? entry_point_->method_name : "unknown";
   768→            f.is_blocking = false;
   769→            record_failure(f);
   770→            
   771→            // Simulate onCreate behavior
   772→            lifecycle_events_.push_back("onCreate(stubbed)");
   773→            
   774→            ApiCallRecord call;
   775→            call.class_name = activity_class_name_;
   776→            call.method_name = "onCreate";
   777→            call.descriptor = "(Landroid/os/Bundle;)V";
   778→            call.category = ApiCategory::ACTIVITY;
   779→            call.call_timestamp = get_timestamp();
   780→            call.sequence = ++api_call_sequence_;
   781→            call.success = true;
   782→            call.result_summary = "onCreate executed (stubbed)";
   783→            call.implementation_status = ImplementationStatus::STUBBED;
   784→            record_api_call(call);
   785→            
   786→            return true;
   787→        }
   788→        
   789→        // Try to use batch interpreter if available
   790→        DexInterpreterBatch batch_interpreter;
   791→        BatchInterpreterConfig config;  // Use BatchInterpreterConfig not InterpreterConfig
   792→        config.verbose = config_.verbose;
   793→        config.max_instructions = config_.max_instructions;
   794→        
   795→        instruction_trace_ = std::make_unique<BatchExecutionTrace>(
   796→            batch_interpreter.execute_entry_point(*entry_point_, *dex_report_, config));
   797→        
   798→        bool success = instruction_trace_->completed_successfully || 
   799→                      instruction_trace_->instructions.size() > 0;
   800→        
   801→        if (!success) {
   802→            FailureRecord f;
   803→            f.type = FailureType::UNIMPLEMENTED_OPCODE;
   804→            f.stage = "execute_on_create";
   805→            f.details = instruction_trace_->halt_reason;
   806→            f.timestamp = get_timestamp();
   807→            f.apk_path = apk_path_;
   808→            f.dex_method = entry_point_->method_name;
   809→            f.pc = instruction_trace_->final_pc;
   810→            f.is_blocking = true;
   811→            record_failure(f);
   812→        }
   813→        
   814→        // Record method dispatch
   815→        MethodDispatchRecord dispatch;
   816→        dispatch.caller_class = "ActivityThread";
   817→        dispatch.target_class = activity_class_name_;
   818→        dispatch.method_name = "onCreate";
   819→        dispatch.descriptor = "(Landroid/os/Bundle;)V";
   820→        dispatch.dispatch_type = DispatchType::VIRTUAL;
   821→        dispatch.success = success;
   822→        dispatch.status_detail = success ? "Executed successfully" : instruction_trace_->halt_reason;
   823→        dispatch.sequence = ++dispatch_sequence_;
   824→        record_dispatch(dispatch);
   825→        
   826→        auto end = std::chrono::steady_clock::now();
   827→        double duration = std::chrono::duration<double, std::milli>(end - start).count();
   828→        
   829→        if (config_.verbose) {
   830→            std::cout << "  Instructions executed: " << instruction_trace_->executed_instructions << std::endl;
   831→            std::cout << "  Completed: " << (instruction_trace_->completed_successfully ? "Yes" : "No") << std::endl;
   832→            if (!instruction_trace_->halt_reason.empty()) {
   833→                std::cout << "  Halt reason: " << instruction_trace_->halt_reason << std::endl;
   834→            }
   835→            std::cout << "  Duration: " << duration << "ms" << std::endl;
   836→        }
   837→        
   838→        return success;
   839→        
   840→    } catch (const std::exception& e) {
   841→        FailureRecord f;
   842→        f.type = FailureType::INTERNAL_ERROR;
   843→        f.stage = "execute_on_create";
   844→        f.details = e.what();
   845→        f.timestamp = get_timestamp();
   846→        f.apk_path = apk_path_;
   847→        f.dex_method = entry_point_ ? entry_point_->method_name : "unknown";
   848→        f.is_blocking = true;
   849→        record_failure(f);
   850→        transition_to(RuntimeState::ERROR, std::string("Exception: ") + e.what());
   851→        return false;
   852→    }
   853→}

  1015→bool ApplicationRuntime::perform_layout() {
  1016→    auto start = std::chrono::steady_clock::now();

  1018→    try {
  1019→        if (config_.verbose) {
  1020→            std::cout << "[Phase F] Performing measure/layout..." << std::endl;

  1022→        
  1023→        // The render pipeline handles layout internally when we call render
  1024→        // For now, we just verify the object heap has views to lay out
  1025→        
  1026→        int view_count = 0;
  1027→        std::vector<uint32_t> ids = heap_->get_all_ids();
  1028→        for (uint32_t id : ids) {
  1029→            RuntimeObject* obj = heap_->get(id);
  1030→            if (obj && (obj->get_runtime_class().find("View") != std::string::npos ||
  1031→                       obj->get_runtime_class().find("Layout") != std::string::npos)) {
  1032→                view_count++;
  1033→            }
  1034→        }
  1035→        
  1036→        if (config_.verbose) {
  1037→            std::cout << "  Views in heap: " << view_count << std::endl;
  1038→        }
  1039→        
  1040→        transition_to(RuntimeState::LAYOUT_READY, "Layout ready for rendering");
  1041→        
  1042→        // Record layout API calls
  1043→        ApiCallRecord call;
  1044→        call.class_name = "View";
  1045→        call.method_name = "measure";
  1046→        call.descriptor = "(II)V";
  1047→        call.category = ApiCategory::LAYOUT;
  1048→        call.call_timestamp = get_timestamp();
  1049→        call.sequence = ++api_call_sequence_;
  1050→        call.success = true;
  1051→        call.result_summary = "Measure pass complete for " + std::to_string(view_count) + " views";
  1052→        call.implementation_status = ImplementationStatus::PARTIAL;
  1053→        record_api_call(call);
  1054→        
  1055→        ApiCallRecord layout_call;
  1056→        layout_call.class_name = "View";
  1057→        layout_call.method_name = "layout";

  1062→        layout_call.success = true;
  1063→        layout_call.result_summary = "Layout pass complete";
  1064→        layout_call.implementation_status = ImplementationStatus::PARTIAL;
  1065→        record_api_call(layout_call);
  1066→        
  1067→        auto end = std::chrono::steady_clock::now();
  1068→        double duration = std::chrono::duration<double, std::milli>(end - start).count();
  1069→        
  1070→        if (config_.verbose) {
  1071→            std::cout << "  Duration: " << duration << "ms" << std::endl;
  1072→        }
  1073→        
  1074→        return true;
  1075→        
  1076→    } catch (const std::exception& e) {
  1077→        FailureRecord f;
  1078→        f.type = FailureType::RENDER_FAILURE;
  1079→        f.stage = "perform_layout";
  1080→        f.details = e.what();

  1082→        f.apk_path = apk_path_;
  1083→        f.is_blocking = false;

  1086→    }
  1087→}

  1093→bool ApplicationRuntime::render_frame() {
  1094→    auto start = std::chrono::steady_clock::now();
  1095→    
  1096→    try {
  1097→        if (config_.verbose) {
  1098→            std::cout << "[Phase H] Rendering frame..." << std::endl;
  1099→        }
  1100→        
  1101→        // Get text content from resources or use default
  1102→        std::string text_content = "Hello MiniAndroid";  // Default fallback
  1103→        
  1104→        // Try to get text from string resources
  1105→        auto& strings = resource_manager_->get_string_resources();
  1106→        auto hello_text = strings.get_by_name("hello_text");
  1107→        if (hello_text.has_value()) {
  1108→            text_content = hello_text->value;
  1109→        } else {
  1110→            // Try other common names
  1111→            auto app_name = strings.get_by_name("app_name");
  1112→            if (app_name.has_value()) {
  1113→                text_content = app_name->value;
  1114→            }
  1115→        }
  1116→        
  1117→        // Render using pipeline
  1118→        bool success = render_pipeline_->render(*heap_, text_content);
  1119→        
  1120→        if (!success) {
  1121→            FailureRecord f;
  1122→            f.type = FailureType::RENDER_FAILURE;
  1123→            f.stage = "render_frame";
  1124→            f.details = "Render pipeline failed";
  1125→            f.timestamp = get_timestamp();
  1126→            f.apk_path = apk_path_;
  1127→            f.is_blocking = true;
  1128→            record_failure(f);
  1129→            transition_to(RuntimeState::RENDER_FAILED, "Rendering failed");
  1130→            return false;
  1131→        }
  1132→        
  1133→        auto end = std::chrono::steady_clock::now();
  1134→        double duration = std::chrono::duration<double, std::milli>(end - start).count();
  1135→        
  1136→        transition_to(RuntimeState::FRAME_RENDERED, "Frame rendered successfully");
  1137→        
  1138→        // Record rendering API calls
  1139→        ApiCallRecord call;
  1140→        call.class_name = "View";
  1141→        call.method_name = "draw";
  1142→        call.descriptor = "(Landroid/graphics/Canvas;)V";
  1143→        call.category = ApiCategory::RENDERER;
  1144→        call.call_timestamp = get_timestamp();
  1145→        call.sequence = ++api_call_sequence_;
  1146→        call.success = true;
  1147→        call.result_summary = "Frame rendered (" + std::to_string(render_pipeline_->get_statistics().total_commands_issued) + " commands)";
  1148→        call.implementation_status = ImplementationStatus::IMPLEMENTED;
  1149→        record_api_call(call);
  1150→        
  1151→        if (config_.verbose) {
  1152→            const RenderStatistics& stats = render_pipeline_->get_statistics();
  1153→            std::cout << "  Framebuffer: " << render_pipeline_->get_framebuffer().get_width() 
  1154→                      << "x" << render_pipeline_->get_framebuffer().get_height() << std::endl;
  1155→            std::cout << "  Commands issued: " << stats.total_commands_issued << std::endl;
  1156→            std::cout << "  Pixels written: " << stats.pixels_written << std::endl;
  1157→            std::cout << "  Duration: " << duration << "ms" << std::endl;
  1158→        }
  1159→        
  1160→        return true;
  1161→        
  1162→    } catch (const std::exception& e) {
  1163→        FailureRecord f;
  1164→        f.type = FailureType::RENDER_FAILURE;
  1165→        f.stage = "render_frame";
  1166→        f.details = e.what();
  1167→        f.timestamp = get_timestamp();
  1168→        f.apk_path = apk_path_;
  1169→        f.is_blocking = true;
  1170→        record_failure(f);
  1171→        transition_to(RuntimeState::RENDER_FAILED, std::string("Exception: ") + e.what());
  1172→        return false;
  1173→    }
  1174→}

  1225→void ApplicationRuntime::record_api_call(const ApiCallRecord& call) {
  1226→    api_calls_.push_back(call);
  1227→    
  1228→    // Update database
  1229→    std::string key = call.class_name + "#" + call.method_name;
  1230→    auto it = api_database_.find(key);
  1231→    if (it != api_database_.end()) {
  1232→        it->second.call_count++;
  1233→    } else {
  1234→        ApiDatabaseEntry entry;
  1235→        entry.class_name = call.class_name;
  1236→        entry.method_name = call.method_name;
  1237→        entry.descriptor = call.descriptor;
  1238→        entry.category = call.category;
  1239→        entry.first_seen_exp = 7;  // EXP-007
  1240→        entry.call_count = 1;
  1241→        entry.implementation_status = call.implementation_status;
  1242→        api_database_[key] = entry;
  1243→    }
  1244→}

  1413→    test["apk_path"] = apk_path_;
  1414→    
  1415→    json stages = json::object();
  1416→    
  1417→    stages["apk_loads"] = {{"pass", apk_info_ != nullptr}, {"detail", apk_info_ ? "Parsed OK" : "Failed"}};
  1418→    stages["manifest_loads"] = {{"pass", manifest_info_ != nullptr}, {"detail", manifest_info_ ? "Parsed OK" : "Failed"}};
  1419→    stages["dex_loads"] = {{"pass", dex_report_ != nullptr}, {"detail", dex_report_ ? "Parsed OK" : "Failed"}};
  1420→    stages["launcher_resolves"] = {{"pass", entry_point_ != nullptr && entry_point_->resolved}, 
  1421→                                   {"detail", entry_point_ ? entry_point_->readable_class : "Not resolved"}};
  1422→    stages["activity_resolves"] = {{"pass", activity_object_id_ != 0}, 
  1423→                                    {"detail", "ID=" + std::to_string(activity_object_id_)}};
  1424→    stages["oncreate_executes"] = {{"pass", state_ >= RuntimeState::ACTIVITY_CREATED}, 
  1425→                                    {"detail", state_to_string(state_)}};
  1426→    stages["objects_created"] = {{"pass", heap_ && heap_->size() > 0}, 
  1427→                                 {"detail", std::to_string(heap_ ? heap_->size() : 0) + " objects"}};
  1428→    stages["resources_resolve"] = {{"pass", resource_manager_ && resource_manager_->get_string_resources().get_string_count() > 0}, 
  1429→                                    {"detail", std::to_string(resource_manager_ ? resource_manager_->get_string_resources().get_string_count() : 0) + " strings"}};
  1430→    stages["layout_inflates"] = {{"pass", resource_manager_ && resource_manager_->get_last_inflate_result().success}, 
  1431→                                  {"detail", resource_manager_ ? resource_manager_->get_last_inflate_result().root_view_class : "N/A"}};
  1432→    stages["renderer_executes"] = {{"pass", state_ >= RuntimeState::FRAME_RENDERED}, {"detail", state_to_string(state_)}};
  1433→    stages["framebuffer_exists"] = {{"pass", render_pipeline_ != nullptr}, {"detail", "Framebuffer available"}};
  1434→    
  1435→    test["stages"] = stages;
  1436→    
  1437→    int passed = 0;
  1438→    int total = 0;
  1439→    for (auto& [key, val] : stages.items()) {
  1440→        total++;
  1441→        if (val["pass"].get<bool>()) passed++;
  1442→    }
  1443→    
  1444→    test["summary"] = {
  1445→        {"passed", passed},
  1446→        {"total", total},
  1447→        {"pass_rate", total > 0 ? (double)passed / total : 0.0},
  1448→        {"overall", passed == total ? "PASS" : (passed > total / 2 ? "PARTIAL" : "FAIL")}
  1449→    };
  1450→    
  1451→    return test;

  1454→bool ApplicationRuntime::write_all_evidence(const std::string& output_dir) {
  1455→    try {
  1456→        fs::create_directories(output_dir);
  1457→        fs::create_directories(output_dir + "/golden");
  1458→        fs::create_directories(output_dir + "/database");
  1459→        
  1460→        auto write_json = [&](const std::string& filename, const json& data) -> bool {
  1461→            std::string path = output_dir + "/" + filename;
  1462→            std::ofstream file(path);
  1463→            if (!file.is_open()) {
  1464→                std::cerr << "[ERROR] Cannot write evidence: " << path << std::endl;
  1465→                return false;
  1466→            }
  1467→            file << data.dump(2) << std::endl;
  1468→            file.close();
  1469→            return true;
  1470→        };
  1471→        
  1472→        write_json("application_runtime.json", to_json());
  1473→        write_json("runtime_component_map.json", generate_component_map());
  1474→        write_json("runtime_state_trace.json", generate_state_trace());
  1475→        write_json("failure_report.json", generate_failure_report());
  1476→        write_json("golden_end_to_end.json", generate_golden_end_to_end());
  1477→        
  1478→        if (config_.verbose) {
  1479→            std::cout << "\n[Evidence] All files written to: " << output_dir << "/" << std::endl;

  1481→        
  1482→        return true;
  1483→        
  1484→    } catch (const std::exception& e) {
