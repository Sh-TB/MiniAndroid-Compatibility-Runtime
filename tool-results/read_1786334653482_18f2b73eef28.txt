    24→std::string resource_type_to_string(ResourceType type) {
    25→    switch (type) {
    26→        case ResourceType::STRING: return "string";
    27→        case ResourceType::LAYOUT: return "layout";
    28→        case ResourceType::DRAWABLE: return "drawable";
    29→        case ResourceType::MIPMAP: return "mipmap";
    30→        case ResourceType::COLOR: return "color";
    31→        case ResourceType::DIMEN: return "dimen";
    32→        case ResourceType::ID: return "id";
    33→        case ResourceType::STYLE: return "style";
    34→        case ResourceType::ATTR: return "attr";
    35→        case ResourceType::RAW: return "raw";
    36→        default: return "unknown";
    37→    }
    38→}

    40→ResourceType string_to_resource_type(const std::string& str) {
    41→    if (str == "string") return ResourceType::STRING;
    42→    if (str == "layout") return ResourceType::LAYOUT;
    43→    if (str == "drawable") return ResourceType::DRAWABLE;
    44→    if (str == "mipmap") return ResourceType::MIPMAP;
    45→    if (str == "color") return ResourceType::COLOR;
    46→    if (str == "dimen") return ResourceType::DIMEN;
    47→    if (str == "id") return ResourceType::ID;
    48→    if (str == "style") return ResourceType::STYLE;
    49→    if (str == "attr") return ResourceType::ATTR;
    50→    if (str == "raw") return ResourceType::RAW;
    51→    return ResourceType::UNKNOWN;
    52→}

    58→json XmlNode::to_json() const {
    59→    json j;
    60→    j["tag_name"] = tag_name;
    61→    j["line_number"] = line_number;
    62→    
    63→    if (!attributes.empty()) {
    64→        j["attributes"] = json::array();
    65→        for (const auto& attr : attributes) {
    66→            j["attributes"].push_back(attr.to_json());
    67→        }
    68→    }
    69→    
    70→    if (!text_content.empty()) {
    71→        j["text_content"] = text_content;
    72→    }
    73→    
    74→    if (!children.empty()) {
    75→        j["children"] = json::array();
    76→        for (const auto& child : children) {
    77→            j["children"].push_back(child->to_json());
    78→        }
    79→    }
    80→    
    81→    return j;
    82→}

    91→std::unique_ptr<XmlNode> SimpleXmlParser::parse(const std::string& xml_content) {
    92→    std::istringstream stream(xml_content);
    93→    last_error_.clear();
    94→    
    95→    try {
    96→        auto root = parse_node(stream);
    97→        if (root && !last_error_.empty()) {
    98→            return nullptr;
    99→        }
   100→        return root;
   101→    } catch (const std::exception& e) {
   102→        last_error_ = std::string("XML parse exception: ") + e.what();
   103→        if (reporter_) {
   104→            reporter_->report_error(ResourceError::XML_PARSE_ERROR, last_error_);
   105→        }
   106→        return nullptr;
   107→    }
   108→}

   110→std::unique_ptr<XmlNode> SimpleXmlParser::parse_file(const std::string& filepath) {
   111→    std::ifstream file(filepath);
   112→    if (!file.is_open()) {
   113→        last_error_ = "Cannot open file: " + filepath;
   114→        if (reporter_) {
   115→            reporter_->report_error(ResourceError::FILE_NOT_FOUND, last_error_);
   116→        }
   117→        return nullptr;
   118→    }
   119→    
   120→    std::stringstream buffer;
   121→    buffer << file.rdbuf();
   122→    file.close();
   123→    
   124→    return parse(buffer.str());
   125→}

   127→std::unique_ptr<XmlNode> SimpleXmlParser::parse_node(std::istringstream& stream, int depth) {
   128→    // Prevent stack overflow on deeply nested/malformed XML
   129→    if (depth > 100) {
   130→        last_error_ = "XML nesting too deep (>100 levels)";
   131→        return nullptr;
   132→    }
   133→    
   134→    std::string line;
   135→    while (std::getline(stream, line)) {
   136→        line = trim(line);
   137→        
   138→        // Skip empty lines and comments
   139→        if (line.empty() || line.find("<!--") == 0) {
   140→            if (line.find("-->") != std::string::npos) {
   141→                continue; // Skip rest of comment
   142→            }
   143→            continue;
   144→        }
   145→        
   146→        // Skip XML declaration
   147→        if (line.find("<?xml") == 0) {
   148→            continue;
   149→        }
   150→        
   151→        // Look for opening tag
   152→        size_t tag_start = line.find('<');
   153→        if (tag_start == std::string::npos) {
   154→            // Pure text content - return as text node
   155→            auto node = std::make_unique<XmlNode>();
   156→            node->text_content = unescape_xml(trim(line));
   157→            return node;
   158→        }
   159→        
   160→        // Check if there's text before the tag (e.g., "text<tag>")
   161→        if (tag_start > 0) {
   162→            auto text_node = std::make_unique<XmlNode>();
   163→            text_node->text_content = unescape_xml(trim(line.substr(0, tag_start)));
   164→            
   165→            // Put back the rest for next read
   166→            // For simplicity, just return the text node
   167→            return text_node;
   168→        }
   169→        
   170→        size_t tag_end = line.find('>', tag_start);
   171→        if (tag_end == std::string::npos) {
   172→            // Multi-line tag - read more lines until we find '>'
   173→            std::string remaining;
   174→            while (tag_end == std::string::npos && std::getline(stream, remaining)) {
   175→                line += " " + trim(remaining);  // Add space and next line
   176→                tag_end = line.find('>');
   177→            }
   178→            if (tag_end == std::string::npos) {
   179→                last_error_ = "Unclosed tag";
   180→                return nullptr;
   181→            }
   182→        }
   183→        
   184→        // Extract tag content (between < and >)
   185→        std::string tag_content = line.substr(tag_start + 1, tag_end - tag_start - 1);
   186→        
   187→        // Check for closing tag
   188→        if (!tag_content.empty() && tag_content[0] == '/') {
   189→            return nullptr; // End of current element
   190→        }
   191→        
   192→        // Check for self-closing tag
   193→        bool self_closing = false;
   194→        if (!tag_content.empty() && tag_content[tag_content.size() - 1] == '/') {
   195→            self_closing = true;
   196→            tag_content = tag_content.substr(0, tag_content.size() - 1);
   197→        }
   198→        
   199→        // Parse tag name and attributes
   200→        auto node = std::make_unique<XmlNode>();
   201→        size_t space_pos = tag_content.find(' ');
   202→        if (space_pos != std::string::npos) {
   203→            node->tag_name = trim(tag_content.substr(0, space_pos));
   204→            std::string attrs_str = tag_content.substr(space_pos + 1);
   205→            node->attributes = parse_attributes(attrs_str);
   206→        } else {
   207→            node->tag_name = trim(tag_content);
   208→        }
   209→        
   210→        // Extract any text content after the closing '>' on the same line
   211→        std::string remaining_line = line.substr(tag_end + 1);
   212→        if (!remaining_line.empty()) {
   213→            // Check if there's text content before the next tag or closing tag
   214→            size_t next_tag = remaining_line.find('<');
   215→            if (next_tag != std::string::npos) {
   216→                std::string potential_text = trim(remaining_line.substr(0, next_tag));
   217→                if (!potential_text.empty()) {
   218→                    node->text_content = unescape_xml(potential_text);
   219→                }
   220→                
   221→                // Check if what follows is a closing tag
   222→                std::string after_text = remaining_line.substr(next_tag);
   223→                if (after_text.find("</" + node->tag_name) == 0) {
   224→                    // This is all the text content
   225→                    return node;
   226→                }
   227→            } else {
   228→                // No more tags on this line - might be pure text content
   229→                std::string text = trim(remaining_line);
   230→                if (!text.empty() && text.find("</" + node->tag_name) != std::string::npos) {
   231→                    // Extract text before closing tag
   232→                    size_t close_pos = text.find("</" + node->tag_name);
   233→                    node->text_content = unescape_xml(trim(text.substr(0, close_pos)));
   234→                    return node;
   235→                } else if (!text.empty()) {
   236→                    node->text_content = unescape_xml(text);
   237→                }
   238→            }
   239→        }
   240→        
   241→        // If not self-closing, parse children
   242→        if (!self_closing) {
   243→            while (true) {
   244→                auto pos = stream.tellg();
   245→                std::string child_line;
   246→                if (!std::getline(stream, child_line)) break;
   247→                
   248→                child_line = trim(child_line);
   249→                if (child_line.empty()) continue;
   250→                
   251→                // Check for closing tag
   252→                if (child_line.find("</" + node->tag_name) == 0) {
   253→                    break;
   254→                }
   255→                
   256→                // Put back and parse as child
   257→                stream.seekg(pos);
   258→                auto child = parse_node(stream, depth + 1);
   259→                if (child) {
   260→                    // If child is a text node, add to our text content
   261→                    if (child->tag_name.empty() && !child->text_content.empty()) {
   262→                        if (node->text_content.empty()) {
   263→                            node->text_content = child->text_content;
   264→                        } else {
   265→                            node->text_content += child->text_content;
   266→                        }
   267→                    } else {
   268→                        node->children.push_back(std::move(child));
   269→                    }
   270→                } else {
   271→                    break; // End of children
   272→                }
   273→            }
   274→        }
   275→        
   276→        return node;
   277→    }
   278→    
   279→    return nullptr;
   280→}

   282→std::vector<XmlAttribute> SimpleXmlParser::parse_attributes(const std::string& attr_string) {
   283→    std::vector<XmlAttribute> attributes;
   284→    
   285→    // Simple attribute parser: name="value" or namespace:name="value"
   286→    size_t pos = 0;
   287→    while (pos < attr_string.size()) {
   288→        // Skip whitespace
   289→        while (pos < attr_string.size() && std::isspace(attr_string[pos])) pos++;
   290→        if (pos >= attr_string.size()) break;
   291→        
   292→        // Find attribute name
   293→        size_t name_start = pos;
   294→        while (pos < attr_string.size() && attr_string[pos] != '=' && !std::isspace(attr_string[pos])) {
   295→            pos++;
   296→        }
   297→        std::string name = trim(attr_string.substr(name_start, pos - name_start));
   298→        
   299→        // Skip to =
   300→        while (pos < attr_string.size() && std::isspace(attr_string[pos])) pos++;
   301→        if (pos >= attr_string.size() || attr_string[pos] != '=') break;
   302→        pos++; // skip =
   303→        
   304→        // Skip whitespace after =
   305→        while (pos < attr_string.size() && std::isspace(attr_string[pos])) pos++;
   306→        if (pos >= attr_string.size()) break;
   307→        
   308→        // Find value (quoted)
   309→        char quote_char = attr_string[pos];
   310→        if (quote_char != '"' && quote_char != '\'') break;
   311→        pos++; // skip opening quote
   312→        
   313→        size_t value_start = pos;
   314→        while (pos < attr_string.size() && attr_string[pos] != quote_char) {
   315→            if (attr_string[pos] == '\\') pos++; // skip escaped char
   316→            pos++;
   317→        }
   318→        std::string value = unescape_xml(attr_string.substr(value_start, pos - value_start));
   319→        if (pos < attr_string.size()) pos++; // skip closing quote
   320→        
   321→        // Parse namespace if present
   322→        XmlAttribute attr;
   323→        size_t colon_pos = name.find(':');
   324→        if (colon_pos != std::string::npos) {
   325→            attr.namespace_uri = name.substr(0, colon_pos);
   326→            attr.name = name.substr(colon_pos + 1);
   327→        } else {
   328→            attr.name = name;
   329→        }
   330→        attr.value = value;
   331→        
   332→        attributes.push_back(attr);
   333→    }
   334→    
   335→    return attributes;
   336→}

   338→std::string SimpleXmlParser::trim(const std::string& str) {
   339→    size_t start = str.find_first_not_of(" \t\n\r");
   340→    if (start == std::string::npos) return "";
   341→    size_t end = str.find_last_not_of(" \t\n\r");
   342→    return str.substr(start, end - start + 1);
   343→}
   344→
   345→std::string SimpleXmlParser::unescape_xml(const std::string& str) {
   346→    std::string result = str;
   347→    size_t pos = 0;
   348→    while ((pos = result.find('&', pos)) != std::string::npos) {
   349→        if (result.substr(pos, 4) == "&lt;") {
   350→            result.replace(pos, 4, "<");
   351→        } else if (result.substr(pos, 4) == "&gt;") {
   352→            result.replace(pos, 4, ">");
   353→        } else if (result.substr(pos, 5) == "&amp;") {
   354→            result.replace(pos, 5, "&");
   355→        } else if (result.substr(pos, 6) == "&quot;") {
   356→            result.replace(pos, 6, "\"");
   357→        } else if (result.substr(pos, 6) == "&apos;") {
   358→            result.replace(pos, 6, "'");
   359→        } else {
   360→            pos++;
   361→            continue;
   362→        }
   363→    }
   364→    return result;
   365→}

   420→bool ResourceTable::load_from_arsc(const std::vector<uint8_t>& arsc_data) {
   421→    // Basic ARSC validation
   422→    if (arsc_data.size() < 8) {
   423→        if (reporter_) {
   424→            reporter_->report_error(ResourceError::ARSC_CORRUPTED, 
   425→                "ARSC data too small for valid header");
   426→        }
   427→        return false;
   428→    }
   429→    
   430→    // Check ARSC magic number (0x00020001 or similar)
   431→    uint32_t magic = arsc_data[0] | (arsc_data[1] << 8) | (arsc_data[2] << 16) | (arsc_data[3] << 24);
   432→    if (magic != 0x00020001 && magic != 0x00020000) {
   433→        // Not standard ARSC format - report but don't fail
   434→        if (reporter_) {
   435→            reporter_->report_unsupported("resources.arsc", "binary_format",
   436→                "Non-standard ARSC format detected (magic: 0x" + 
   437→                [&]() {
   438→                    std::ostringstream oss;
   439→                    oss << std::hex << magic;
   440→                    return oss.str();
   441→                }() + ")", "warning");
   442→        }
   443→        // Continue anyway - we'll use XML-based resources instead
   444→    }
   445→    
   446→    // For now, we'll primarily rely on XML resources
   447→    // Full ARSC parsing would require significant additional code
   448→    return true;
   449→}

   481→bool StringResources::load_from_xml(const std::string& xml_content) {
   482→    SimpleXmlParser parser(reporter_);
   483→    auto root = parser.parse(xml_content);
   484→    
   485→    if (!root) {
   486→        if (reporter_) {
   487→            reporter_->report_error(ResourceError::XML_PARSE_ERROR,
   488→                "Failed to parse strings XML: " + parser.get_last_error());
   489→        }
   490→        return false;
   491→    }
   492→    
   493→    // Expect <resources> as root
   494→    if (root->tag_name != "resources") {
   495→        if (reporter_) {
   496→            reporter_->report_error(ResourceError::PARSE_ERROR,
   497→                "strings.xml root element must be <resources>, got <" + root->tag_name + ">");
   498→        }
   499→        return false;
   500→    }
   501→    
   502→    // Parse <string> elements
   503→    for (const auto& child : root->children) {
   504→        if (child->tag_name == "string") {
   505→            StringResource str_res;
   506→            
   507→            // Get name attribute (required)
   508→            auto name_attr = child->get_attribute("name");
   509→            if (!name_attr) {
   510→                if (reporter_) {
   511→                    reporter_->report_error(ResourceError::PARSE_ERROR,
   512→                        "String resource missing 'name' attribute");
   513→                }
   514→                continue;
   515→            }
   516→            str_res.name = *name_attr;
   517→            
   518→            // Get text content or value
   519→            if (!child->text_content.empty()) {
   520→                str_res.value = child->text_content;
   521→            } else if (!child->children.empty()) {
   522→                // Handle nested elements or CDATA
   523→                for (const auto& sub_child : child->children) {
   524→                    if (sub_child->text_content.empty()) {
   525→                        str_res.value += sub_child->tag_name; // Fallback
   526→                    } else {
   527→                        str_res.value += sub_child->text_content;
   528→                    }
   529→                }
   530→            }
   531→            
   532→            // Assign a simple ID based on index
   533→            str_res.id = ResourceId::from_raw(0x7F040001 + static_cast<uint32_t>(strings_.size()));
   534→            
   535→            strings_.push_back(str_res);
   536→            name_index_[str_res.name] = strings_.size() - 1;
   537→            id_index_[str_res.id.raw_id] = strings_.size() - 1;
   538→        }
   539→    }
   540→    
   541→    return !strings_.empty();
   542→}

   618→ResourceType ResourceInventory::classify_path(const std::string& path) {
   619→    if (path.find("res/values/") != std::string::npos || path.find("res/values-") != std::string::npos) {
   620→        if (path.find("strings.xml") != std::string::npos) return ResourceType::STRING;
   621→        if (path.find("colors.xml") != std::string::npos) return ResourceType::COLOR;
   622→        if (path.find("dims.xml") != std::string::npos || path.find("dimens.xml") != std::string::npos) return ResourceType::DIMEN;
   623→        if (path.find("ids.xml") != std::string::npos) return ResourceType::ID;
   624→        if (path.find("styles.xml") != std::string::npos) return ResourceType::STYLE;
   625→        if (path.find("attrs.xml") != std::string::npos) return ResourceType::ATTR;
   626→        return ResourceType::UNKNOWN; // Other values files
   627→    }
   628→    if (path.find("res/layout/") != std::string::npos || path.find("res/layout-") != std::string::npos) {
   629→        return ResourceType::LAYOUT;
   630→    }
   631→    if (path.find("res/drawable/") != std::string::npos || path.find("res/drawable-") != std::string::npos) {
   632→        return ResourceType::DRAWABLE;
   633→    }
   634→    if (path.find("res/mipmap/") != std::string::npos || path.find("res/mipmap-") != std::string::npos) {
   635→        return ResourceType::MIPMAP;
   636→    }
   637→    if (path.find("res/raw/") != std::string::npos) {
   638→        return ResourceType::RAW;
   639→    }
   640→    if (path.find("resources.arsc") != std::string::npos) {
   641→        return ResourceType::STRING; // ARSC contains all types
   642→    }
   643→    if (path.find("assets/") != std::string::npos) {
   644→        return ResourceType::RAW;
   645→    }
   646→    
   647→    return ResourceType::UNKNOWN;
   648→}

   650→bool ResourceInventory::scan_apk_entries(const std::vector<std::string>& entries) {
   651→    entries_.clear();
   652→    
   653→    for (const auto& entry : entries) {
   654→        ResourceEntry res_entry;
   655→        res_entry.path = entry;
   656→        res_entry.type = classify_path(entry);
   657→        res_entry.is_supported = (res_entry.type != ResourceType::UNKNOWN);
   658→        
   659→        entries_.push_back(res_entry);
   660→        
   661→        // Report unsupported types
   662→        if (!res_entry.is_supported && reporter_) {
   663→            reporter_->report_unsupported(
   664→                resource_type_to_string(res_entry.type),
   665→                entry,
   666→                "Unknown resource type classification",
   667→                "info");
   668→        }
   669→        
   670→        // Report unsupported but known types
   671→        if ((res_entry.type == ResourceType::DRAWABLE || res_entry.type == ResourceType::MIPMAP ||
   672→             res_entry.type == ResourceType::STYLE || res_entry.type == ResourceType::ATTR) && reporter_) {
   673→            reporter_->report_unsupported(
   674→                resource_type_to_string(res_entry.type),
   675→                entry,
   676→                "Resource type not fully implemented in EXP-006",
   677→                "warning");
   678→        }
   679→    }
   680→    
   681→    return true;
   682→}

   780→InflateResult LayoutInflater::inflate(const XmlNode* layout_root) {
   781→    InflateResult result;
   782→    inflate_count_++;
   783→    
   784→    if (!layout_root) {
   785→        result.success = false;
   786→        result.error_message = "Layout root is null";
   787→        return result;
   788→    }
   789→    
   790→    if (!heap_) {
   791→        result.success = false;
   792→        result.error_message = "Object heap not initialized";
   793→        if (reporter_) {
   794→            reporter_->report_error(ResourceError::PARSE_ERROR, result.error_message);
   795→        }
   796→        return result;
   797→    }
   798→    
   799→    // Push an empty result for inflate_node to populate
   800→    history_.push_back(InflateResult());
   801→    
   802→    try {
   803→        // Inflate the root node (will populate history_.back())
   804→        result.root_view_id = inflate_node(layout_root, 0);
   805→        
   806→        // Copy data from populated history entry
   807→        if (!history_.empty()) {
   808→            auto& populated = history_.back();
   809→            result.created_view_ids = std::move(populated.created_view_ids);
   810→            result.view_classes = std::move(populated.view_classes);
   811→            result.layout_params = std::move(populated.layout_params);
   812→            result.view_attributes = std::move(populated.view_attributes);
   813→        }
   814→        
   815→        if (result.root_view_id == 0) {
   816→            result.success = false;
   817→            result.error_message = "Failed to inflate root view";
   818→            history_.pop_back(); // Remove failed entry
   819→            return result;
   820→        }
   821→        
   822→        // Get root view class
   823→        result.root_view_class = layout_root->tag_name;
   824→        
   825→        result.success = true;
   826→        // Update history_ with the complete result
   827→        if (!history_.empty()) {
   828→            history_.back() = result;
   829→        }
   830→        
   831→    } catch (const std::exception& e) {
   832→        result.success = false;
   833→        result.error_message = std::string("Inflate exception: ") + e.what();
   834→        if (reporter_) {
   835→            reporter_->report_error(ResourceError::XML_PARSE_ERROR, result.error_message);
   836→        }
   837→        if (!history_.empty()) {
   838→            history_.back() = result;
   839→        }
   840→    }
   841→    
   842→    return result;
   843→}

   845→InflateResult LayoutInflater::inflate_from_xml(const std::string& xml_content) {
   846→    SimpleXmlParser parser(reporter_);

   852→        result.error_message = "Failed to parse layout XML: " + parser.get_last_error();
   853→        return result;

   855→    
   856→    return inflate(root.get());

   877→uint32_t LayoutInflater::inflate_node(const XmlNode* node, uint32_t parent_id) {
   878→    if (!node) return 0;
   879→    
   880→    std::string class_name = node->tag_name;
   881→    
   882→    // Map XML tag names to View class names
   883→    if (class_name == "LinearLayout") {
   884→        class_name = "android.widget.LinearLayout";
   885→    } else if (class_name == "TextView") {
   886→        class_name = "android.widget.TextView";
   887→    } else if (class_name == "Button") {
   888→        class_name = "android.widget.Button";
   889→    } else if (class_name == "ImageView") {
   890→        class_name = "android.widget.ImageView";
   891→    }
   892→    
   893→    // Create the view object
   894→    uint32_t view_id = create_view(class_name, node->attributes);
   895→    
   896→    if (view_id == 0) {
   897→        if (reporter_) {
   898→            reporter_->report_unsupported("View", class_name,
   899→                "Unknown view type or creation failed", "error");
   900→        }
   901→        return 0;
   902→    }
   903→    
   904→    // Apply attributes to the view
   905→    apply_attributes(view_id, node->attributes);
   906→    
   907→    // Store view class mapping (use the pre-created history entry)
   908→    if (history_.empty()) {
   909→        history_.push_back(InflateResult()); // Fallback if not created
   910→    }
   911→    auto& current_result = history_.back();
   912→    current_result.created_view_ids.push_back(view_id);
   913→    current_result.view_classes[view_id] = class_name;
   914→    
   915→    // Recursively inflate children
   916→    for (const auto& child : node->children) {
   917→        uint32_t child_id = inflate_node(child.get(), view_id);
   918→        if (child_id == 0) {
   919→            // Child inflation failed but continue with siblings
   920→            continue;
   921→        }
   922→    }
   923→    
   924→    return view_id;
   925→}
   926→
   927→uint32_t LayoutInflater::create_view(const std::string& class_name, const std::vector<XmlAttribute>& attrs) {
   928→    if (!heap_) return 0;
   929→    
   930→    // Determine view type from class name
   931→    if (class_name.find("TextView") != std::string::npos || 
   932→        class_name.find("Button") != std::string::npos) {
   933→        // Create TextView
   934→        auto* tv = heap_->allocate<miniandroid::runtime::TextViewRuntimeObject>("TextView", 0);
   935→        if (tv) {
   936→            uint32_t id = tv->get_object_id();
   937→            
   938→            // Set initial properties from attributes
   939→            for (const auto& attr : attrs) {
   940→                if (attr.name == "text") {
   941→                    tv->set_text(resolve_attribute_value(attr.value));
   942→                }
   943→            }
   944→            
   945→            return id;
   946→        }
   947→    } else if (class_name.find("LinearLayout") != std::string::npos ||
   948→               class_name.find("ViewGroup") != std::string::npos) {
   949→        // Create generic ViewGroup (use View for now)
   950→        auto* view = heap_->allocate<miniandroid::runtime::ViewRuntimeObject>("ViewGroup", 0);
   951→        if (view) {
   952→            return view->get_object_id();
   953→        }
   954→    } else {
   955→        // Generic View
   956→        auto* view = heap_->allocate<miniandroid::runtime::ViewRuntimeObject>("View", 0);
   957→        if (view) {
   958→            return view->get_object_id();
   959→        }
   960→    }
   961→    
   962→    return 0;
   963→}

   965→void LayoutInflater::apply_attributes(uint32_t view_id, const std::vector<XmlAttribute>& attrs) {
   966→    if (!heap_ || view_id == 0) return;
   967→    
   968→    LayoutParams params = parse_layout_params(attrs);
   969→    
   970→    // Store layout params
   971→    if (history_.empty()) {
   972→        history_.push_back(InflateResult());
   973→    }
   974→    auto& current_result = history_.back();
   975→    current_result.layout_params[view_id] = params;
   976→    
   977→    // Store all attributes for reference
   978→    std::map<std::string, std::string> attr_map;
   979→    for (const auto& attr : attrs) {
   980→        std::string key = attr.namespace_uri.empty() ? attr.name : attr.namespace_uri + ":" + attr.name;
   981→        attr_map[key] = resolve_attribute_value(attr.value);
   982→    }
   983→    current_result.view_attributes[view_id] = attr_map;
   984→    
   985→    // Apply specific attributes to the view object
   986→    auto* view = heap_->get_as<miniandroid::runtime::ViewRuntimeObject>(view_id);
   987→    if (view) {
   988→        // Apply layout params to view bounds (simplified)
   989→        // In real Android, this happens during layout pass
   990→    }
   991→    
   992→    // Apply text to TextView
   993→    auto* tv = heap_->get_as<miniandroid::runtime::TextViewRuntimeObject>(view_id);
   994→    if (tv) {
   995→        for (const auto& attr : attrs) {
   996→            if (attr.name == "text") {
   997→                std::string resolved = resolve_attribute_value(attr.value);
   998→                if (!resolved.empty()) {
   999→                    tv->set_text(resolved);
  1000→                }
  1001→            }
  1002→        }
  1003→    }
  1004→}
  1005→
  1006→LayoutParams LayoutInflater::parse_layout_params(const std::vector<XmlAttribute>& attrs) {
  1007→    LayoutParams params;
  1008→    
  1009→    for (const auto& attr : attrs) {
  1010→        if (attr.namespace_uri == "android") {
  1011→            if (attr.name == "layout_width") {
  1012→                if (attr.value == "match_parent" || attr.value == "fill_parent") {
  1013→                    params.width = LayoutParams::MATCH_PARENT;
  1014→                } else if (attr.value == "wrap_content") {
  1015→                    params.width = LayoutParams::WRAP_CONTENT;
  1016→                } else {
  1017→                    try {
  1018→                        params.width = std::stoi(attr.value);
  1019→                    } catch (...) {}
  1020→                }
  1021→            } else if (attr.name == "layout_height") {
  1022→                if (attr.value == "match_parent" || attr.value == "fill_parent") {
  1023→                    params.height = LayoutParams::MATCH_PARENT;
  1024→                } else if (attr.value == "wrap_content") {
  1025→                    params.height = LayoutParams::WRAP_CONTENT;
  1026→                } else {
  1027→                    try {
  1028→                        params.height = std::stoi(attr.value);
  1029→                    } catch (...) {}
  1030→                }
  1031→            } else if (attr.name == "layout_margin") {
  1032→                try {
  1033→                    int margin = std::stoi(attr.value);
  1034→                    params.left_margin = margin;
  1035→                    params.top_margin = margin;
  1036→                    params.right_margin = margin;
  1037→                    params.bottom_margin = margin;
  1038→                } catch (...) {}
  1039→            } else if (attr.name == "layout_marginLeft") {
  1040→                try { params.left_margin = std::stoi(attr.value); } catch (...) {}
  1041→            } else if (attr.name == "layout_marginTop") {
  1042→                try { params.top_margin = std::stoi(attr.value); } catch (...) {}
  1043→            } else if (attr.name == "layout_marginRight") {
  1044→                try { params.right_margin = std::stoi(attr.value); } catch (...) {}
  1045→            } else if (attr.name == "layout_marginBottom") {
  1046→                try { params.bottom_margin = std::stoi(attr.value); } catch (...) {}
  1047→            } else if (attr.name == "layout_gravity") {
  1048→                // Simplified gravity handling
  1049→                if (attr.value.find("center") != std::string::npos) {
  1050→                    params.gravity = 1; // CENTER
  1051→                } else if (attr.value.find("right") != std::string::npos || attr.value.find("end") != std::string::npos) {
  1052→                    params.gravity = 2; // RIGHT/END
  1053→                }
  1054→            }
  1055→        }
  1056→    }
  1057→    
  1058→    return params;
  1059→}
  1060→
  1061→std::string LayoutInflater::resolve_attribute_value(const std::string& value) {
  1062→    // Handle resource references like @string/hello_text
  1063→    if (value.find("@string/") == 0) {
  1064→        std::string name = value.substr(8); // Remove "@string/"
  1065→        if (string_res_) {
  1066→            return string_res_->resolve_string(name, "");
  1067→        }
  1068→        return ""; // Resource not available
  1069→    }
  1070→    
  1071→    // Handle other resource references
  1072→    if (value.find("@") == 0) {
  1073→        if (reporter_) {
  1074→            reporter_->report_unsupported("resource_reference", value,
  1075→                "Non-string resource references not fully resolved", "warning");
  1076→        }
  1077→        return value; // Return as-is for now
  1078→    }
  1079→    
  1080→    return value;
  1081→}

  1091→bool ResourceManager::initialize(const std::string& apk_path) {
  1092→    apk_path_ = apk_path;
  1093→    
  1094→    // Read APK into memory
  1095→    std::ifstream file(apk_path, std::ios::binary | std::ios::ate);
  1096→    if (!file.is_open()) {
  1097→        failure_reporter_.report_error(ResourceError::FILE_NOT_FOUND,
  1098→            "Cannot open APK: " + apk_path);
  1099→        return false;
  1100→    }
  1101→    
  1102→    size_t size = file.tellg();
  1103→    file.seekg(0, std::ios::beg);
  1104→    
  1105→    apk_data_.resize(size);
  1106→    if (!file.read(reinterpret_cast<char*>(apk_data_.data()), size)) {
  1107→        failure_reporter_.report_error(ResourceError::FILE_NOT_FOUND,
  1108→            "Failed to read APK data: " + apk_path);
  1109→        return false;
  1110→    }
  1111→    file.close();
  1112→    
  1113→    state_.initialized = true;
  1114→    
  1115→    // Detect ARSC
  1116→    state_.arsc_detected = resource_table_.detect_and_load_arsc(apk_data_);
  1117→    
  1118→    // Scan inventory using APK parser
  1119→    apk::ApkParser parser;
  1120→    auto entries = parser.list_entries(apk_path);
  1121→    std::vector<std::string> entry_names;
  1122→    for (const auto& entry : entries) {
  1123→        entry_names.push_back(entry.name);
  1124→    }
  1125→    inventory_.scan_apk_entries(entry_names);
  1126→    
  1127→    return true;
  1128→}

  1144→bool ResourceManager::load_all_resources() {
  1145→    if (!state_.initialized) {
  1146→        failure_reporter_.report_error(ResourceError::PARSE_ERROR,
  1147→            "Manager not initialized");
  1148→        return false;
  1149→    }
  1150→
