   145→InstructionTrace DexInterpreterExp018::execute_exp018(
   146→    const MethodInfo& method,
   147→    const std::vector<std::string>& strings,
   148→    const Exp018Config& config
   149→) {
   150→    exp018_config_ = config;
   151→    
   152→    // Load databases if available
   153→    if (!config.api_priority_path.empty()) {
   154→        load_api_priority_database(config.api_priority_path);
   155→    }
   156→    if (!config.opcode_frequency_path.empty()) {
   157→        load_opcode_frequency_database(config.opcode_frequency_path);
   158→    }
   159→    
   160→    // Reset state
   161→    branch_trace_.clear();
   162→    loops_detected_.clear();
   163→    cf_stats_ = ControlFlowStatistics{};
   164→    return_trace_.clear();
   165→    has_pending_return_ = false;
   166→    pending_return_ = Value::make_uninitialized();
   167→    static_call_trace_.clear();
   168→    interface_trace_.clear();
   169→    pc_visit_count_.clear();
   170→    
   171→    // Execute with base interpreter first, then enhance
   172→    InstructionTrace trace = execute(method, strings, config);
   173→    
   174→    // Add EXP-018 specific data to trace
   175→    trace.branch_trace = branch_trace_;
   176→    trace.loops_detected = loops_detected_;
   177→    trace.control_flow_stats = cf_stats_.to_json();
   178→    trace.return_trace = return_trace_;
   179→    trace.static_calls = static_call_trace_;
   180→    trace.interface_calls = interface_trace_;
   181→    
   182→    return trace;
   183→}

   431→bool DexInterpreterExp018::execute_if_eqz(uint32_t pc, InstructionTraceEntry& entry) {
   432→    // Format: if-eqz vAA, +BBBB
   433→    // Branch if vAA == zero (or null for references)
   434→    
   435→    if (bytecode_.size() < pc + 3) return false;
   436→    
   437→    uint8_t regA = bytecode_[pc + 1];
   438→    int16_t offset = static_cast<int16_t>((bytecode_[pc + 2] << 8) | bytecode_[pc + 3]);
   439→    
   440→    Value val = get_register(regA);
   441→    
   442→    BranchTraceBranch branch;
   443→    branch.sequence = branch_trace_.size();
   444→    branch.pc_before = pc;
   445→    branch.opcode_name = "if-eqz";
   446→    branch.condition_description = "Branch if v" + std::to_string(regA) + " == 0/null";
   447→    branch.condition.register_a = "v" + std::to_string(regA);
   448→    branch.offset = offset;
   449→    branch.fallthrough_pc = pc + 4;  // Instruction size is 4 bytes for 22x format
   450→    
   451→    // Evaluate condition: zero or null
   452→    bool is_zero_or_null = false;
   453→    if (val.type == ValueType::NULL_REF || val.is_null) {
   454→        is_zero_or_null = true;
   455→        branch.condition.was_null_a = true;
   456→        branch.condition.value_a = 0;
   457→    } else if (val.type == ValueType::INT32) {
   458→        is_zero_or_null = (val.int_val == 0);
   459→        branch.condition.value_a = val.int_val;
   460→    } else if (val.type == ValueType::BOOLEAN) {
   461→        is_zero_or_null = !val.bool_val;
   462→        branch.condition.value_a = val.bool_val ? 1 : 0;
   463→    } else {
   464→        // Treat uninitialized as not-zero for safety
   465→        is_zero_or_null = false;
   466→        branch.condition.was_null_a = false;
   467→        branch.condition.value_a = 0;
   468→    }
   469→    
   470→    branch.condition.condition_result = is_zero_or_null;
   471→    branch.taken = is_zero_or_null;
   472→    
   473→    if (is_zero_or_null) {
   474→        branch.target_pc = pc + offset;
   475→        pc_ = branch.target_pc;
   476→        cf_stats_.branches_taken++;
   477→    } else {
   478→        branch.target_pc = pc + offset;
   479→        pc_ = branch.fallthrough_pc;
   480→        cf_stats_.branches_not_taken++;
   481→    }
   482→    
   483→    entry.status = InstructionTraceEntry::Status::SUCCESS;
   484→    entry.operands.push_back({"v" + std::to_string(regA), val.to_string(), regA});
   485→    entry.operands.push_back({"offset", std::to_string(offset), offset});
   486→    
   487→    if (exp018_config_.trace_all_branches) {
   488→        record_branch(branch);
   489→    }
   490→    
   491→    detect_loops(pc, pc_);
   492→    
   493→    return true;
   494→}

   496→bool DexInterpreterExp018::execute_if_nez(uint32_t pc, InstructionTraceEntry& entry) {
   497→    // Format: if-nez vAA, +BBBB
   498→    // Branch if vAA != zero (non-null for references)
   499→    
   500→    if (bytecode_.size() < pc + 3) return false;
   501→    
   502→    uint8_t regA = bytecode_[pc + 1];
   503→    int16_t offset = static_cast<int16_t>((bytecode_[pc + 2] << 8) | bytecode_[pc + 3]);
   504→    
   505→    Value val = get_register(regA);
   506→    
   507→    BranchTraceBranch branch;
   508→    branch.sequence = branch_trace_.size();
   509→    branch.pc_before = pc;
   510→    branch.opcode_name = "if-nez";
   511→    branch.condition_description = "Branch if v" + std::to_string(regA) + " != 0/non-null";
   512→    branch.condition.register_a = "v" + std::to_string(regA);
   513→    branch.offset = offset;
   514→    branch.fallthrough_pc = pc + 4;
   515→    
   516→    // Evaluate condition: non-zero or non-null
   517→    bool is_non_zero_non_null = false;
   518→    if (val.type == ValueType::NULL_REF || val.is_null) {
   519→        is_non_zero_non_null = false;
   520→        branch.condition.was_null_a = true;
   521→        branch.condition.value_a = 0;
   522→    } else if (val.type == ValueType::INT32) {
   523→        is_non_zero_non_null = (val.int_val != 0);
   524→        branch.condition.value_a = val.int_val;
   525→    } else if (val.type == ValueType::BOOLEAN) {
   526→        is_non_zero_non_null = val.bool_val;
   527→        branch.condition.value_a = val.bool_val ? 1 : 0;
   528→    } else if (val.type == ValueType::OBJECT_REF || val.type == ValueType::STRING_REF) {
   529→        is_non_zero_non_null = true;  // Valid reference is always "non-null"
   530→        branch.condition.value_a = val.reference_id;
   531→    } else {
   532→        is_non_zero_non_null = false;
   533→    }
   534→    
   535→    branch.condition.condition_result = is_non_zero_non_null;
   536→    branch.taken = is_non_zero_non_null;
   537→    
   538→    if (is_non_zero_non_null) {
   539→        branch.target_pc = pc + offset;
   540→        pc_ = branch.target_pc;
   541→        cf_stats_.branches_taken++;
   542→    } else {
   543→        branch.target_pc = pc + offset;
   544→        pc_ = branch.fallthrough_pc;
   545→        cf_stats_.branches_not_taken++;
   546→    }
   547→    
   548→    entry.status = InstructionTraceEntry::Status::SUCCESS;
   549→    entry.operands.push_back({"v" + std::to_string(regA), val.to_string(), regA});
   550→    entry.operands.push_back({"offset", std::to_string(offset), offset});
   551→    
   552→    if (exp018_config_.trace_all_branches) {
   553→        record_branch(branch);

   561→bool DexInterpreterExp018::execute_if_eq(uint32_t pc, InstructionTraceEntry& entry) {
   562→    // Format: if-eq vAA, vBB, +CCCC
   563→    // Branch if vAA == vBB
   564→    
   565→    if (bytecode_.size() < pc + 4) return false;
   566→    
   567→    uint8_t regA = bytecode_[pc + 1];  // AA in low 8 bits
   568→    uint8_t regB = bytecode_[pc + 2];  // BB
   569→    int16_t offset = static_cast<int16_t>((bytecode_[pc + 3] << 8) | bytecode_[pc + 4]);
   570→    
   571→    Value valA = get_register(regA);
   572→    Value valB = get_register(regB);
   573→    
   574→    BranchTraceBranch branch;
   575→    branch.sequence = branch_trace_.size();
   576→    branch.pc_before = pc;
   577→    branch.opcode_name = "if-eq";
   578→    branch.condition_description = "Branch if v" + std::to_string(regA) + " == v" + std::to_string(regB);
   579→    branch.condition.register_a = "v" + std::to_string(regA);
   580→    branch.condition.register_b = "v" + std::to_string(regB);
   581→    branch.offset = offset;
   582→    branch.fallthrough_pc = pc + 5;
   583→    
   584→    // Extract integer values for comparison
   585→    int32_t a = 0, b = 0;
   586→    bool can_compare = true;
   587→    
   588→    if (valA.type == ValueType::INT32) { a = valA.int_val; }
   589→    else if (valA.type == ValueType::BOOLEAN) { a = valA.bool_val ? 1 : 0; }
   590→    else if (valA.type == ValueType::NULL_REF) { a = 0; }
   591→    else { can_compare = false; branch.condition.was_null_a = true; }
   592→    
   593→    if (valB.type == ValueType::INT32) { b = valB.int_val; }
   594→    else if (valB.type == ValueType::BOOLEAN) { b = valB.bool_val ? 1 : 0; }
   595→    else if (valB.type == ValueType::NULL_REF) { b = 0; }
   596→    else { can_compare = false; branch.condition.was_null_b = true; }
   597→    
   598→    branch.condition.value_a = a;
   599→    branch.condition.value_b = b;
   600→    
   601→    bool is_equal = can_compare && (a == b);
   602→    branch.condition.condition_result = is_equal;
   603→    branch.taken = is_equal;
   604→    
   605→    if (is_equal) {
   606→        branch.target_pc = pc + offset;
   607→        pc_ = branch.target_pc;
   608→        cf_stats_.branches_taken++;
   609→    } else {
   610→        branch.target_pc = pc + offset;
   611→        pc_ = branch.fallthrough_pc;
   612→        cf_stats_.branches_not_taken++;
   613→    }
   614→    
   615→    entry.status = InstructionTraceEntry::Status::SUCCESS;
   616→    entry.operands.push_back({"v" + std::to_string(regA), valA.to_string(), regA});
   617→    entry.operands.push_back({"v" + std::to_string(regB), valB.to_string(), regB});
   618→    entry.operands.push_back({"offset", std::to_string(offset), offset});
   619→    
   620→    if (exp018_config_.trace_all_branches) {
   621→        record_branch(branch);
   622→    }
   623→    
   624→    detect_loops(pc, pc_);
   625→    
   626→    return true;

  1106→void DexInterpreterExp018::detect_loops(uint32_t from_pc, uint32_t to_pc) {
  1107→    // Simple backward edge detection: if we're jumping to an earlier PC, it's likely a loop
  1108→    if (to_pc < from_pc) {
  1109→        // Check if this loop was already detected
  1110→        bool found = false;
  1111→        for (auto& loop : loops_detected_) {
  1112→            if (loop.loop_start_pc <= to_pc && loop.loop_end_pc >= from_pc) {
  1113→                loop.iteration_count++;
  1114→                loop.back_edge_pcs.push_back(from_pc);
  1115→                found = true;
  1116→                break;
  1117→            }
  1118→        }
  1119→        
  1120→        if (!found) {
  1121→            LoopDetectionRecord loop;
  1122→            loop.loop_start_pc = to_pc;
  1123→            loop.loop_end_pc = from_pc;
  1124→            loop.loop_type = "detected-back-edge";
  1125→            loop.iteration_count = 1;
  1126→            loop.back_edge_pcs.push_back(from_pc);
  1127→            loops_detected_.push_back(loop);
  1128→            cf_stats_.loops_detected++;
  1129→        }
  1130→    }
  1131→}

  1133→bool DexInterpreterExp018::check_loop_protection(uint32_t target_pc) {
  1134→    if (!exp018_config_.enable_loop_detection) return true;
  1135→    
  1136→    auto it = pc_visit_count_.find(target_pc);

  1138→        return false;  // Block the jump - infinite loop protection
  1139→    }
  1140→    return true;
  1141→}

  1730→bool DexInterpreterExp018::execute_invoke_static(uint32_t pc, InstructionTraceEntry& entry, 
  1731→                                                  const Exp018Config& config) {
  1732→    // Format: invoke-static {vC, vD, ...}, method@BBBB
  1733→    // Call a static method
  1734→    
  1735→    if (bytecode_.size() < pc + 3) return false;
  1736→    
  1737→    uint16_t method_idx = (bytecode_[pc + 2] << 8) | bytecode_[pc + 3];
  1738→    
  1739→    // Decode register count and argument registers
  1740→    uint8_t arg_info = bytecode_[pc + 1];
  1741→    uint8_t arg_count = (arg_info >> 4) & 0x0F;
  1742→    
  1743→    // Extract argument registers (simplified - assuming contiguous vC..)
  1744→    std::vector<uint8_t> arg_registers;
  1745→    std::vector<Value> args;
  1746→    
  1747→    for (uint8_t i = 0; i < arg_count && (pc + 4 + i) < bytecode_.size(); ++i) {
  1748→        uint8_t reg = bytecode_[pc + 4 + i];
  1749→        arg_registers.push_back(reg);
  1750→        args.push_back(get_register(reg));
  1751→    }
  1752→    
  1753→    // Look up method in string table (simplified resolution)
  1754→    // In real DEX, method_idx refers to method_ids[] which has class, method, proto refs
  1755→    std::string class_name = "unknown";
  1756→    std::string method_name = "unknown";
  1757→    std::string descriptor = "()V";
  1758→    
  1759→    // Try to resolve from dex_report if available
  1760→    // For now, use heuristic based on common patterns
  1761→    
  1762→    ApiCallRecord call_record;
  1763→    call_record.sequence = api_call_sequence_++;
  1764→    call_record.pc = pc;
  1765→    call_record.dispatch_type = DispatchType::STATIC;
  1766→    call_record.was_dispatched_from_dex = true;
  1767→    call_record.dispatch_path = "DEX_INVOKE_STATIC_EXP018";
  1768→    call_record.arguments = args;
  1769→    
  1770→    // Look up in static registry
  1771→    StaticMethodEntry* method_impl = nullptr;
  1772→    
  1773→    // Try exact match first
  1774→    for (auto& entry : static_registry_) {
  1775→        if (entry.method_name == method_name && entry.class_name == class_name) {
  1776→            method_impl = &entry;
  1777→            break;
  1778→        }
  1779→    }
  1780→    
  1781→    // If not found by index, try common patterns
  1782→    if (!method_impl) {
  1783→        // Search by method name only (heuristic)
  1784→        for (auto& reg_entry : static_registry_) {
  1785→            if (reg_entry.method_name == method_name) {
  1786→                method_impl = &reg_entry;
  1787→                class_name = reg_entry.class_name;  // Update for trace
  1788→                descriptor = reg_entry.descriptor;
  1789→                break;
  1790→            }
  1791→        }
  1792→    }
  1793→    
  1794→    call_record.class_name = class_name;
  1795→    call_record.method_name = method_name;
  1796→    call_record.descriptor = descriptor;
  1797→    
  1798→    if (method_impl && config.enable_static_dispatch) {
  1799→        // Execute the static method
  1800→        method_impl->call_count++;
  1801→        Value result = method_impl->implementation(args, nullptr);
  1802→        
  1803→        call_record.success = true;
  1804→        call_record.result_value = result;
  1805→        call_record.result_summary = "Static method executed: " + class_name + "." + method_name;
  1806→        
  1807→        // Set pending return for move-result/move-result-object
  1808→        if (descriptor.find(")V") == std::string::npos) {
  1809→            // Non-void method - set pending return
  1810→            set_pending_return(result);
  1811→        }
  1812→        
  1813→        entry.status = InstructionTraceEntry::Status::API_CALL_DISPATCHED;
  1814→        entry.api_call_made = call_record;
  1815→        
  1816→    } else {
  1817→        // Method not implemented
  1818→        call_record.success = false;
  1819→        call_record.error_message = "Static method not in registry: " + class_name + "." + method_name;
  1820→        call_record.result_summary = "UNIMPLEMENTED_STATIC";
  1821→        
  1822→        entry.status = InstructionTraceEntry::Status::HALT_UNIMPLEMENTED;
  1823→        entry.halt_reason = "invoke-static not implemented: " + class_name + "." + method_name;
  1824→        entry.api_call_made = call_record;
  1825→        
  1826→        if (config.stop_on_unimplemented) {
  1827→            halt_reason_ = entry.halt_reason.value();
  1828→            halted_ = true;
  1829→        }
  1830→    }
  1831→    
  1832→    static_call_trace_.push_back(call_record);
  1833→    api_calls_.push_back(call_record);
  1834→    
  1835→    // Advance PC past the instruction
  1836→    // invoke-static format: 35c or 3rc (5 bytes + 2 bytes per 5 regs over 4, rounded up)
  1837→    uint32_t instr_size = 4 + ((arg_count + 3) / 4) * 2;  // Simplified calculation
  1838→    if (arg_count <= 5) {
  1839→        instr_size = 5;  // 35c format
  1840→    } else {
  1841→        instr_size = 5 + ((arg_count - 5) / 4 + 1) * 2;  // Extended format
  1842→    }
  1843→    pc_ += instr_size;
  1844→    
  1845→    entry.operands.push_back({"method", class_name + "." + method_name + descriptor, method_idx});
  1846→    entry.operands.push_back({"args", std::to_string(arg_count), arg_count});
  1847→    
  1848→    return true;
  1849→}

  1893→bool DexInterpreterExp018::execute_invoke_interface(uint32_t pc, InstructionTraceEntry& entry,
  1894→                                                     const Exp018Config& config) {

  1896→    // Call an interface method on an object
  1897→    
  1898→    if (bytecode_.size() < pc + 3) return false;
  1899→    
  1900→    uint16_t method_idx = (bytecode_[pc + 2] << 8) | bytecode_[pc + 3];
  1901→    uint8_t arg_info = bytecode_[pc + 1];
  1902→    uint8_t arg_count = (arg_info >> 4) & 0x0F;
  1903→    
  1904→    // First argument is 'this' for interface calls
  1905→    std::vector<uint8_t> arg_registers;
  1906→    std::vector<Value> args;
  1907→    
  1908→    for (uint8_t i = 0; i < arg_count && (pc + 4 + i) < bytecode_.size(); ++i) {
  1909→        uint8_t reg = bytecode_[pc + 4 + i];
  1910→        arg_registers.push_back(reg);
  1911→        args.push_back(get_register(reg));
  1912→    }
  1913→    
  1914→    InterfaceDispatchRecord dispatch;
  1915→    dispatch.sequence = interface_trace_.size();
  1916→    dispatch.pc = pc;
  1917→    dispatch.target_object = args.empty() ? Value::make_null() : args[0];
  1918→    dispatch.actual_class = dispatch.target_object.object_class;
  1919→    
  1920→    // Resolve interface method (simplified)
  1921→    std::string interface_name = "android.view.View$OnClickListener";  // Most common
  1922→    std::string method_name = "onClick";
  1923→    std::string descriptor = "(Landroid/view/View;)V";
  1924→    
  1925→    dispatch.interface_name = interface_name;
  1926→    dispatch.method_name = method_name;
  1927→    dispatch.descriptor = descriptor;
  1928→    dispatch.is_on_click_listener = (method_name == "onClick");
  1929→    
  1930→    // Look up implementation
  1931→    auto iface_it = interface_implementations_.find(interface_name);
  1932→    bool found = false;
  1933→    
  1934→    if (iface_it != interface_implementations_.end()) {
  1935→        auto method_it = iface_it->second.find(method_name);
  1936→        if (method_it != iface_it->second.end()) {
  1937→            // Execute interface method
  1938→            Value result = method_it->second(args, nullptr);
  1939→            
  1940→            dispatch.resolved = true;
  1941→            dispatch.resolved_method = interface_name + "." + method_name;
  1942→            dispatch.result = result;
  1943→            
  1944→            entry.status = InstructionTraceEntry::Status::API_CALL_DISPATCHED;
  1945→            found = true;
  1946→            
  1947→            log("[invoke-interface] " + interface_name + "." + method_name + " dispatched");
  1948→        }
  1949→    }
  1950→    
  1951→    if (!found) {
  1952→        dispatch.resolved = false;
  1953→        dispatch.resolved_method = "UNIMPLEMENTED";
  1954→        
  1955→        entry.status = config.strict_real_mode ? 
  1956→            InstructionTraceEntry::Status::HALT_UNIMPLEMENTED :
  1957→            InstructionTraceEntry::Status::SUCCESS;
  1958→        entry.halt_reason = "invoke-interface not implemented: " + interface_name + "." + method_name;
  1959→        
  1960→        if (config.stop_on_unimplemented) {
  1961→            halt_reason_ = entry.halt_reason.value();
