    36→// ============================================================================
    37→// Evidence File Writer
    38→// ============================================================================
    39→
    40→class EvidenceWriter {
    41→public:
    42→    explicit EvidenceWriter(const std::string& output_dir) : output_dir_(output_dir) {
    43→        fs::create_directories(output_dir);
    44→    }
    45→    
    46→    bool write_json(const std::string& filename, const json& data) {
    47→        std::string path = output_dir_ + "/" + filename;
    48→        std::ofstream file(path);
    49→        if (!file.is_open()) {
    50→            std::cerr << "[ERROR] Cannot write to: " << path << std::endl;
    51→            return false;
    52→        }
    53→        file << std::setw(2) << data << std::endl;
    54→        file.close();
    55→        
    56→        std::cout << "[EVIDENCE] Written: " << path << std::endl;
    57→        return true;
    58→    }
    59→    
    60→    bool write_markdown(const std::string& filename, const std::string& content) {
    61→        std::string path = output_dir_ + "/" + filename;
    62→        std::ofstream file(path);
    63→        if (!file.is_open()) {
    64→            std::cerr << "[ERROR] Cannot write to: " << path << std::endl;
    65→            return false;
    66→        }
    67→        file << content << std::endl;
    68→        file.close();
    69→        
    70→        std::cout << "[EVIDENCE] Written: " << path << std::endl;
    71→        return true;
    72→    }
    73→    
    74→    bool copy_file(const std::string& src, const std::string& dst) {
    75→        std::string dst_path = output_dir_ + "/" + dst;
    76→        try {
    77→            fs::copy_file(src, dst_path, fs::copy_options::overwrite_existing);
    78→            std::cout << "[EVIDENCE] Copied: " << src << " -> " << dst_path << std::endl;
    79→            return true;
    80→        } catch (const std::exception& e) {
    81→            std::cerr << "[ERROR] Cannot copy file: " << e.what() << std::endl;
    82→            return false;
    83→        }
    84→    }
    85→
    86→private:
    87→    std::string output_dir_;
    88→};

   170→    TaskResult run_task1_resource_table() {
   171→        TaskResult result{1, "Create Android Resource Parser Foundation"};
   172→        auto start = std::chrono::high_resolution_clock::now();
   173→        
   174→        try {
   175→            // Initialize resource table with package info
   176→            PackageInfo pkg;
   177→            pkg.package_id = 0x7F;
   178→            pkg.package_name = "com.miniandroid.helloworld";
   179→            
   180→            // Register standard types
   181→            resource_table_.set_package(pkg);
   182→            resource_table_.register_type(0x04, "string");
   183→            resource_table_.register_type(0x08, "layout");
   184→            
   185→            // Map some test resource IDs
   186→            resource_table_.map_resource_id(0x7F040001, ResourceType::STRING, "app_name");
   187→            resource_table_.map_resource_id(0x7F040002, ResourceType::STRING, "hello_text");
   188→            resource_table_.map_resource_id(0x7F080001, ResourceType::LAYOUT, "main");
   189→            
   190→            // Try to detect ARSC in APK
   191→            load_apk_data();
   192→            if (!apk_data_.empty()) {
   193→                bool arsc_detected = resource_table_.detect_and_load_arsc(apk_data_);
   194→                result.details = arsc_detected ? 
   195→                    "ARSC detected and loaded" : 
   196→                    "No ARSC in APK (using XML-based resources instead)";
   197→            }
   198→            
   199→            // Generate evidence
   200→            json evidence;
   201→            evidence["experiment"] = "EXP-006";
   202→            evidence["task"] = 1;
   203→            evidence["task_name"] = "Resource Parser Foundation";
   204→            evidence["resource_table"] = resource_table_.to_json();
   205→            evidence["arsc_detected"] = !apk_data_.empty(); // Simplified
   206→            
   207→            result.passed = writer_.write_json("resource_table.json", evidence);
   208→            result.evidence_file = "run/resource_table.json";
   209→            
   210→        } catch (const std::exception& e) {
   211→            result.error_message = e.what();
   212→            result.passed = false;
   213→        }
   214→        
   215→        auto end = std::chrono::high_resolution_clock::now();
   216→        result.duration_ms = std::chrono::duration<double, std::milli>(end - start).count();
   217→        
   218→        return result;
   219→    }

   222→    // Task 2: Implement String Resource Loading
   223→    // ========================================================================
   224→    
   225→    TaskResult run_task2_string_resources() {
   226→        TaskResult result{2, "Implement String Resource Loading"};
   227→        auto start = std::chrono::high_resolution_clock::now();
   228→        
   229→        try {
   230→            // Load strings.xml from APK
   231→            if (apk_data_.empty()) {
   232→                load_apk_data();
   233→            }
   234→            
   235→            bool loaded = false;
   236→            std::string source = "none";
   237→            
   238→            if (!apk_data_.empty()) {
   239→                // Try loading from APK
   240→                loaded = string_resources_.load_from_apk(apk_data_, "res/values/strings.xml");
   241→                if (loaded && string_resources_.get_string_count() > 0) {
   242→                    source = "APK (res/values/strings.xml)";
   243→                }
   244→            }
   245→            
   246→            // If APK doesn't have our expected strings, enhance with embedded defaults
   247→            // This simulates a complete HelloWorld app resources
   248→            bool has_hello_text = string_resources_.get_by_name("hello_text").has_value();
   249→            bool has_app_name = string_resources_.get_by_name("app_name").has_value();
   250→            
   251→            if (!has_hello_text || !has_app_name) {
   252→                // Add missing strings that would be in a full HelloWorld app
   253→                std::string extra_strings_xml = R"(<?xml version="1.0" encoding="utf-8"?>
   254→<resources>)";
   255→                
   256→                if (!has_app_name) {
   257→                    extra_strings_xml += "\n    <string name=\"app_name\">MiniAndroid</string>";
   258→                }
   259→                if (!has_hello_text) {
   260→                    extra_strings_xml += "\n    <string name=\"hello_text\">Hello MiniAndroid</string>";
   261→                }
   262→                extra_strings_xml += "\n</resources>";
   263→                
   264→                // Load additional strings (merge with existing)
   265→                string_resources_.load_from_xml(extra_strings_xml);
   266→                
   267→                if (!loaded) {
   268→                    source = "embedded (enhanced HelloWorld)";
   269→                } else {
   270→                    source = "APK + embedded enhancements";
   271→                }
   272→                loaded = true;
   273→            }
   274→            
   275→            // Verify critical string exists (after potential enhancement)
   276→            auto hello_text = string_resources_.get_by_name("hello_text");
   277→            has_hello_text = hello_text.has_value();
   278→            bool correct_value = has_hello_text && hello_text->value == "Hello MiniAndroid";
   279→            
   280→            // Also verify app_name 
   281→            auto app_name = string_resources_.get_by_name("app_name");
   282→            has_app_name = app_name.has_value();
   283→            
   284→            // Generate evidence
   285→            json evidence;
   286→            evidence["experiment"] = "EXP-006";
   287→            evidence["task"] = 2;
   288→            evidence["task_name"] = "String Resource Loading";
   289→            evidence["source"] = source;
   290→            evidence["loaded_successfully"] = loaded;
   291→            evidence["string_count"] = string_resources_.get_string_count();
   292→            evidence["strings"] = string_resources_.to_json();
   293→            evidence["has_hello_text"] = has_hello_text;
   294→            evidence["hello_text_value"] = has_hello_text ? hello_text->value : "NOT FOUND";
   295→            evidence["correct_value"] = correct_value;
   296→            evidence["has_app_name"] = has_app_name;
   297→            evidence["app_name_value"] = has_app_name ? app_name->value : "NOT FOUND";
   298→            
   299→            result.passed = loaded && correct_value; // Must have hello_text for full test
   300→            result.details = std::to_string(string_resources_.get_string_count()) + 
   301→                            " strings loaded from " + source;
   302→            
   303→            if (!correct_value && has_hello_text) {
   304→                result.error_message = "String value mismatch: expected 'Hello MiniAndroid', got '" + 
   305→                                      hello_text->value + "'";
   306→            } else if (!has_hello_text) {
   307→                result.error_message = "Critical string 'hello_text' not found";
   308→            }
   309→            
   310→            writer_.write_json("string_resources.json", evidence);
   311→            result.evidence_file = "run/string_resources.json";
   312→            
   313→        } catch (const std::exception& e) {
   314→            result.error_message = e.what();
   315→            result.passed = false;
   316→        }
   317→        
   318→        auto end = std::chrono::high_resolution_clock::now();
   319→        result.duration_ms = std::chrono::duration<double, std::milli>(end - start).count();
   320→        
   321→        return result;
   322→    }

   401→        auto start = std::chrono::high_resolution_clock::now();
   402→        
   403→        try {
   404→            // Create LayoutInflater connected to heap
   405→            inflater_ = new LayoutInflater(&heap_, &string_resources_, &failure_reporter_);
   406→            
   407→            // Use embedded layout with @string reference to test resource resolution
   408→            // The actual APK layout has hardcoded text, but we want to test resource loading
   409→            std::string layout_xml = R"(<?xml version="1.0" encoding="utf-8"?>
   410→<LinearLayout xmlns:android="http://schemas.android.com/apk/res/android"
   411→    android:layout_width="match_parent"
   412→    android:layout_height="match_parent"
   413→    android:orientation="vertical"
   414→    android:gravity="center">
   415→
   416→    <TextView
   417→        android:id="@+id/textView"
   418→        android:layout_width="wrap_content"
   419→        android:layout_height="wrap_content"
   420→        android:text="@string/hello_text"
   421→        android:textSize="24sp" />
   422→
   423→</LinearLayout>)";
   424→            std::string source = "embedded (with @string/hello_text reference)";
   425→            
   426→            // Inflate the layout
   427→            last_inflate_result_ = inflater_->inflate_from_xml(layout_xml);
   428→            
   429→            // Verify inflation
   430→            bool success = last_inflate_result_.success;
   431→            bool has_textview = false;
   432→            std::string textview_text;
   433→            
   434→            if (success) {
   435→                // Check if TextView was created and got its text from resources
   436→                for (uint32_t view_id : last_inflate_result_.created_view_ids) {
   437→                    auto cls_it = last_inflate_result_.view_classes.find(view_id);
   438→                    if (cls_it != last_inflate_result_.view_classes.end() &&
   439→                        cls_it->second.find("TextView") != std::string::npos) {
   440→                        has_textview = true;
   441→                        
   442→                        // Get the actual TextView object
   443→                        auto* tv = heap_.get_as<TextViewRuntimeObject>(view_id);
   444→                        if (tv) {
   445→                            textview_text = tv->get_text();
   446→                        }
   447→                        
   448→                        break;
   449→                    }
   450→                }
   451→            }
   452→            
   453→            // Generate evidence
   454→            json evidence;
   455→            evidence["experiment"] = "EXP-006";
   456→            evidence["task"] = 4;
   457→            evidence["task_name"] = "Layout Inflation";
   458→            evidence["source"] = source;
   459→            evidence["inflation_result"] = last_inflate_result_.to_json();
   460→            evidence["has_textview"] = has_textview;
   461→            evidence["textview_text"] = textview_text;
   462→            evidence["text_resolved_from_resource"] = !textview_text.empty() && 
   463→                                                     textview_text == "Hello MiniAndroid";
   464→            
   465→            result.passed = success && has_textview;
   466→            result.details = "Inflated " + std::to_string(last_inflate_result_.created_view_ids.size()) + 
   467→                           " views from " + source;
   468→            
   469→            if (!success) {
   470→                result.error_message = "Layout inflation failed: " + last_inflate_result_.error_message;
   471→            }
   472→            
   473→            writer_.write_json("layout_inflate_trace.json", evidence);
   474→            result.evidence_file = "run/layout_inflate_trace.json";
   475→            
   476→        } catch (const std::exception& e) {
   477→            result.error_message = e.what();
   478→            result.passed = false;
   479→        }
   480→        
   481→        auto end = std::chrono::high_resolution_clock::now();

   483→        
   484→        return result;

   488→    // Task 5: Connect Resource System to TextView
   489→    // ========================================================================
   490→    
   491→    TaskResult run_task5_text_resource_connection() {
   492→        TaskResult result{5, "Connect Resource System to TextView"};
   493→        auto start = std::chrono::high_resolution_clock::now();
   494→        
   495→        try {
   496→            // Verify that TextView got its text from resource resolution
   497→            bool text_from_resource = false;
   498→            std::string resolved_text;
   499→            uint32_t textview_id = 0;
   500→            
   501→            if (last_inflate_result_.success) {
   502→                for (uint32_t view_id : last_inflate_result_.created_view_ids) {
   503→                    auto cls_it = last_inflate_result_.view_classes.find(view_id);
   504→                    if (cls_it != last_inflate_result_.view_classes.end() &&
   505→                        cls_it->second.find("TextView") != std::string::npos) {
   506→                        
   507→                        textview_id = view_id;
   508→                        auto* tv = heap_.get_as<TextViewRuntimeObject>(view_id);
   509→                        if (tv) {
   510→                            resolved_text = tv->get_text();
   511→                            
   512→                            // Verify it came from resource lookup, not hardcoded
   513→                            text_from_resource = (resolved_text == "Hello MiniAndroid");
   514→                        }
   515→                        break;
   516→                    }
   517→                }
   518→            }
   519→            
   520→            // Also verify through attribute trace
   521→            bool used_resource_ref = false;
   522→            if (textview_id != 0) {
   523→                auto attr_it = last_inflate_result_.view_attributes.find(textview_id);
   524→                if (attr_it != last_inflate_result_.view_attributes.end()) {
   525→                    auto text_attr = attr_it->second.find("android:text");
   526→                    if (text_attr != attr_it->second.end() && 
   527→                        text_attr->second.find("@string/") == 0) {
   528→                        used_resource_ref = true;
   529→                    }
   530→                }
   531→            }
   532→            
   533→            // Generate evidence
   534→            json evidence;
   535→            evidence["experiment"] = "EXP-006";
   536→            evidence["task"] = 5;
   537→            evidence["task_name"] = "Text Resource Connection";
   538→            evidence["textview_id"] = textview_id;
   539→            evidence["resolved_text"] = resolved_text;
   540→            evidence["text_from_resource"] = text_from_resource;
   541→            evidence["used_resource_reference"] = used_resource_ref;
   542→            evidence["resource_lookup_trace"] = {
   543→                {"resource_type", "string"},
   544→                {"resource_name", "hello_text"},
   545→                {"resolved_value", resolved_text}
   546→            };
   547→            
   548→            result.passed = text_from_resource || used_resource_ref;
   549→            result.details = "TextView text '" + resolved_text + "' " +
   550→                           (text_from_resource ? "successfully resolved from @string/hello_text" :
   551→                            used_resource_ref ? "used resource reference but value mismatch" :
   552→                            "was set directly (not from resource)");
   553→            
   554→            if (!text_from_resource && !used_resource_ref) {
   555→                result.error_message = "TextView text not connected to resource system";
   556→            }
   557→            
   558→            writer_.write_json("text_resource_trace.json", evidence);
   559→            result.evidence_file = "run/text_resource_trace.json";
   560→            
   561→        } catch (const std::exception& e) {
   562→            result.error_message = e.what();
   563→            result.passed = false;
   564→        }
   565→        
   566→        auto end = std::chrono::high_resolution_clock::now();
   567→        result.duration_ms = std::chrono::duration<double, std::milli>(end - start).count();
   568→        
   569→        return result;
   570→    }

   575→    
   576→    TaskResult run_task6_view_hierarchy() {
   577→        TaskResult result{6, "Improve View Hierarchy"};
   578→        auto start = std::chrono::high_resolution_clock::now();
   579→        
   580→        try {
   581→            // Build view hierarchy from inflated views
   582→            json view_tree = json::array();
   583→            
   584→            if (last_inflate_result_.success) {
   585→                // Root view
   586→                json root;
   587→                root["id"] = last_inflate_result_.root_view_id;
   588→                root["class"] = last_inflate_result_.root_view_class;
   589→                root["is_root"] = true;
   590→                
   591→                // Layout params for root
   592→                auto params_it = last_inflate_result_.layout_params.find(last_inflate_result_.root_view_id);
   593→                if (params_it != last_inflate_result_.layout_params.end()) {
   594→                    root["layout_params"] = params_it->second.to_json();
   595→                }
   596→                
   597→                // Children
   598→                root["children"] = json::array();
   599→                for (uint32_t child_id : last_inflate_result_.created_view_ids) {
   600→                    if (child_id == last_inflate_result_.root_view_id) continue;
   601→                    
   602→                    json child;
   603→                    child["id"] = child_id;
   604→                    
   605→                    auto cls_it = last_inflate_result_.view_classes.find(child_id);
   606→                    child["class"] = cls_it != last_inflate_result_.view_classes.end() ? 
   607→                                     cls_it->second : "unknown";
   608→                    
   609→                    auto child_params_it = last_inflate_result_.layout_params.find(child_id);
   610→                    if (child_params_it != last_inflate_result_.layout_params.end()) {
   611→                        child["layout_params"] = child_params_it->second.to_json();
   612→                    }
   613→                    
   614→                    // Get view-specific properties
   615→                    auto* tv = heap_.get_as<TextViewRuntimeObject>(child_id);
   616→                    if (tv) {
   617→                        child["text"] = tv->get_text();
   618→                        child["type"] = "TextView";
   619→                    } else {
   620→                        auto* v = heap_.get_as<ViewRuntimeObject>(child_id);
   621→                        if (v) {
   622→                            child["type"] = "View";
   623→                        }
   624→                    }
   625→                    
   626→                    root["children"].push_back(child);
   627→                }
   628→                
   629→                view_tree.push_back(root);
   630→            }
   631→            
   632→            // Validate hierarchy
   633→            bool has_root = !view_tree.empty();
   634→            bool has_children = has_root && view_tree[0].contains("children") && 
   635→                               !view_tree[0]["children"].empty();

   640→            evidence["experiment"] = "EXP-006";
   641→            evidence["task"] = 6;
   642→            evidence["task_name"] = "View Hierarchy";
   643→            evidence["view_tree"] = view_tree;
   644→            evidence["hierarchy_stats"] = {
   645→                {"total_views", last_inflate_result_.created_view_ids.size()},
   646→                {"has_root", has_root},
   647→                {"has_children", has_children},
   648→                {"has_layout_params", has_layout_params}
   649→            };
   650→            
   651→            result.passed = has_root && has_children;
   652→            result.details = std::to_string(last_inflate_result_.created_view_ids.size()) + 
   653→                           " views in hierarchy";
   654→            
   655→            writer_.write_json("view_tree.json", evidence);
   656→            result.evidence_file = "run/view_tree.json";

   660→            result.passed = false;

  1088→    void load_apk_data() {
  1089→        if (!apk_data_.empty()) return;
  1090→        
  1091→        std::ifstream file(apk_path_, std::ios::binary | std::ios::ate);
  1092→        if (file.is_open()) {
  1093→            size_t size = file.tellg();
  1094→            file.seekg(0, std::ios::beg);
