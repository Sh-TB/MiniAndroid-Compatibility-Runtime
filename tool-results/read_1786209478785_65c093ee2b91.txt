   240→void BitmapFont::initialize_ascii_glyphs() {
   241→    // Space (32)
   242→    glyphs_[0] = {' ', 4, 16, 6, space_bitmap};
   243→    
   244→    // Characters needed for "Hello MiniAndroid"
   245→    glyphs_[33] = {'H', 7, 16, 8, H_bitmap};   // H (72-32=40) - actually index 40
   246→    glyphs_[37] = {'e', 6, 16, 7, e_bitmap};    // e (101-32=69)
   247→    glyphs_[44] = {'l', 2, 16, 3, l_bitmap};    // l (108-32=76)
   248→    glyphs_[47] = {'o', 6, 16, 7, o_bitmap};    // o (111-32=79)
   249→    glyphs_[45] = {'M', 8, 16, 9, M_bitmap};    // M (77-32=45)
   250→    glyphs_[41] = {'i', 2, 16, 3, i_bitmap};    // i (105-32=73)
   251→    glyphs_[46] = {'n', 6, 16, 7, n_bitmap};    // n (110-32=78)
   252→    glyphs_[36] = {'d', 6, 16, 7, d_bitmap};    // d (100-32=68)
   253→    glyphs_[50] = {'r', 4, 16, 5, r_bitmap};    // r (114-32=82)
   254→    glyphs_[33] = {'A', 7, 16, 8, A_bitmap};    // A (65-32=33)
   255→    glyphs_[52] = {'t', 4, 16, 5, t_bitmap};    // t (116-32=84)
   256→    
   257→    // Fix H at correct position (index 40 = 'H' - 32)
   258→    static const uint8_t default_glyph_bitmap[16] = {
   259→        0x00, 0x00, 0x00, 0x00, 0x18, 0x18, 0x18, 0x18,
   260→        0x18, 0x18, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00
   261→    };
   262→    
   263→    fill_remaining_glyphs();
   264→}
   265→
   266→void BitmapFont::fill_remaining_glyphs() {
   267→    static const uint8_t default_bitmap[16] = {
   268→        0x00, 0x00, 0x00, 0x00, 0x18, 0x18, 0x18, 0x18,
   269→        0x18, 0x18, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00
   270→    };
   271→    
   272→    for (int i = 0; i < 95; i++) {
   273→        if (glyphs_[i].character == '\0') {
   274→            char c = static_cast<char>(i + 32);
   275→            glyphs_[i] = {c, 5, 16, 6, default_bitmap};
   276→        }
   277→    }
   278→}
   279→
   280→const BitmapFont::Glyph* BitmapFont::get_glyph(char c) const {
   281→    if (c >= 32 && c <= 126) {
   282→        return &glyphs_[c - 32];
   283→    }
   284→    return &glyphs_[0];  // Return space for unknown chars
   285→}
   286→
   287→BitmapFont::TextMetrics BitmapFont::measure_text(const std::string& text) const {
   288→    TextMetrics metrics{0, 16, 12, 4};
   289→    
   290→    for (char c : text) {
   291→        const Glyph* glyph = get_glyph(c);
   292→        metrics.width += glyph->advance;
   293→    }
   294→    
   295→    return metrics;
   296→}

   344→void SoftwareCanvas::draw_rect(float left, float top, float right, float bottom, RGBA color) {
   345→    int x1 = static_cast<int>(std::round(left));
   346→    int y1 = static_cast<int>(std::round(top));
   347→    int x2 = static_cast<int>(std::round(right));
   348→    int y2 = static_cast<int>(std::round(bottom));
   349→    
   350→    for (int y = y1; y < y2 && y < framebuffer_->get_height(); y++) {
   351→        for (int x = x1; x < x2 && x < framebuffer_->get_width(); x++) {
   352→            if (x >= 0 && y >= 0) {
   353→                framebuffer_->set_pixel(x, y, color);
   354→            }
   355→        }
   356→    }
   357→    
   358→    CanvasCommand cmd;
   359→    cmd.type = "drawRect";
   360→    cmd.sequence = ++command_sequence_;
   361→    cmd.params["rect"] = {{"left", x1}, {"top", y1}, {"right", x2}, {"bottom", y2}};
   362→    cmd.params["color"] = color.to_json();
   363→    commands_.push_back(cmd);
   364→}

   366→void SoftwareCanvas::draw_text(const std::string& text, float x, float y, RGBA color,
   367→                               const BitmapFont* font) {
   368→    const BitmapFont* use_font = font ? font : &default_font_;
   369→    
   370→    auto metrics = use_font->measure_text(text);
   371→    int start_x = static_cast<int>(std::round(x));
   372→    int start_y = static_cast<int>(std::round(y)) - use_font->get_baseline_offset();
   373→    
   374→    // Render each character
   375→    int current_x = start_x;
   376→    for (char c : text) {
   377→        const BitmapFont::Glyph* glyph = use_font->get_glyph(c);
   378→        
   379→        // Render glyph bitmap
   380→        for (int row = 0; row < glyph->height; row++) {
   381→            uint8_t bitmap_row = glyph->bitmap[row];
   382→            for (int col = 0; col < 8; col++) {
   383→                if (bitmap_row & (0x80 >> col)) {
   384→                    framebuffer_->set_pixel(current_x + col, start_y + row, color);
   385→                }
   386→            }
   387→        }
   388→        
   389→        current_x += glyph->advance;
   390→    }
   391→    
   392→    CanvasCommand cmd;
   393→    cmd.type = "drawText";
   394→    cmd.sequence = ++command_sequence_;
   395→    cmd.params["text"] = text;
   396→    cmd.params["position"] = {{"x", start_x}, {"y", start_y}};
   397→    cmd.params["color"] = color.to_json();
   398→    cmd.params["measured_width"] = metrics.width;
   399→    cmd.params["measured_height"] = metrics.height;
   400→    commands_.push_back(cmd);
   401→}

   452→RenderPipeline::RenderPipeline(int width, int height)
   453→    : framebuffer_(width, height), canvas_(&framebuffer_)
   454→    , stage_(RenderStage::IDLE), frame_number_(0) {}
   455→
   456→bool RenderPipeline::render(runtime::EnhancedObjectHeap& heap, const std::string& activity_text) {
   457→    auto total_start = std::chrono::high_resolution_clock::now();
   458→    
   459→    stage_ = RenderStage::LAYOUT;
   460→    
   461→    // Stage 1: Layout
   462→    auto layout_start = std::chrono::high_resolution_clock::now();
   463→    bool layout_ok = perform_layout(heap);
   464→    auto layout_end = std::chrono::high_resolution_clock::now();
   465→    stats_.layout_time_ms = std::chrono::duration<double, std::milli>(layout_end - layout_start).count();
   466→    
   467→    if (!layout_ok) {
   468→        stage_ = RenderStage::ERROR;
   469→        return false;
   470→    }
   471→    
   472→    // Stage 2: Measure
   473→    stage_ = RenderStage::MEASURE;
   474→    auto measure_start = std::chrono::high_resolution_clock::now();
   475→    bool measure_ok = perform_measurement();
   476→    auto measure_end = std::chrono::high_resolution_clock::now();
   477→    stats_.measure_time_ms = std::chrono::duration<double, std::milli>(measure_end - measure_start).count();
   478→    
   479→    if (!measure_ok) {
   480→        stage_ = RenderStage::ERROR;
   481→        return false;
   482→    }
   483→    
   484→    // Stage 3: Draw
   485→    stage_ = RenderStage::DRAW;
   486→    auto draw_start = std::chrono::high_resolution_clock::now();
   487→    bool draw_ok = perform_draw(heap, activity_text);
   488→    auto draw_end = std::chrono::high_resolution_clock::now();
   489→    stats_.draw_time_ms = std::chrono::duration<double, std::milli>(draw_end - draw_start).count();
   490→    
   491→    if (!draw_ok) {
   492→        stage_ = RenderStage::ERROR;
   493→        return false;
   494→    }
   495→    
   496→    // Complete
   497→    auto total_end = std::chrono::high_resolution_clock::now();
   498→    stats_.total_time_ms = std::chrono::duration<double, std::milli>(total_end - total_start).count();
   499→    stats_.pixels_written = framebuffer_.get_draw_count();
   500→    stats_.total_commands_issued = static_cast<int>(canvas_.get_command_count());
   501→    
   502→    stage_ = RenderStage::COMPLETE;
   503→    frame_number_++;
   504→    
   505→    return true;
   506→}
   507→
   508→bool RenderPipeline::perform_layout(runtime::EnhancedObjectHeap& heap) {
   509→    // Find Activity and build layout tree from it
   510→    auto activity_ids = heap.get_objects_by_class("android.app.Activity");
   511→    
   512→    if (activity_ids.empty()) {
   513→        // No Activity found, try to build from all objects
   514→        auto all_ids = heap.get_all_ids();
   515→        if (!all_ids.empty()) {
   516→            layout_tree_ = build_layout_tree(all_ids[0], heap);
   517→            return true;
   518→        }
   519→        return false;
   520→    }
   521→    
   522→    layout_tree_ = build_layout_tree(activity_ids[0], heap);
   523→    
   524→    // Apply initial layout to root (full screen)
   525→    layout_view(layout_tree_, 0, 0, framebuffer_.get_width(), framebuffer_.get_height());
   526→    
   527→    stats_.total_views_processed++;
   528→    
   529→    return true;
   530→}

   538→bool RenderPipeline::perform_draw(runtime::EnhancedObjectHeap& heap, const std::string& activity_text) {
   539→    // Clear background
   540→    canvas_.draw_color(Colors::GREY_200);  // Light grey background like Android
   541→    
   542→    // Draw Activity background (optional - could be transparent)
   543→    // For now, just draw the TextView with the activity text
   544→    
   545→    // Find TextView objects and render them
   546→    auto tv_ids = heap.get_objects_by_class("android.widget.TextView");
   547→    
   548→    for (uint32_t tv_id : tv_ids) {
   549→        auto* tv = heap.get_as<runtime::TextViewRuntimeObject>(tv_id);
   550→        if (tv) {
   551→            std::string text_to_render = tv->get_text().empty() ? activity_text : tv->get_text();
   552→            
   553→            // Get bounds from layout tree or use defaults
   554→            int text_x = 50;  // Margin from left
   555→            int text_y = 100; // Position from top
   556→            
   557→            // Draw text
   558→            canvas_.draw_text(text_to_render, text_x, text_y + font_.get_line_height(), 
   559→                            Colors::GREY_800, &font_);
   560→            
   561→            stats_.text_views_rendered++;
   562→        }
   563→    }
   564→    
   565→    // If no TextView found, still render the text (for demo purposes)
   566→    if (stats_.text_views_rendered == 0 && !activity_text.empty()) {
   567→        canvas_.draw_text(activity_text, 50, 100 + font_.get_line_height(), 
   568→                        Colors::GREY_800, &font_);
   569→        stats_.text_views_rendered = 1;
   570→    }
   571→    
   572→    return true;
   573→}

   575→LayoutNode RenderPipeline::build_layout_tree(uint32_t object_id, runtime::EnhancedObjectHeap& heap, int depth) {
   576→    LayoutNode node;
   577→    node.object_id = object_id;
   578→    
   579→    auto* obj = heap.get(object_id);
   580→    if (obj) {
   581→        node.view_class = obj->get_runtime_class();
   582→        
   583→        // Try to get geometry from View/TextView objects
   584→        auto* view_obj = dynamic_cast<const runtime::ViewRuntimeObject*>(obj);
   585→        if (view_obj) {
   586→            auto geom = view_obj->get_geometry();
   587→            node.bounds.set(geom.left, geom.top, geom.width, geom.height);
   588→        }
   589→        
   590→        // For Activity, find child TextViews and add them as children
   591→        if (node.view_class == "android.app.Activity") {
   592→            auto tv_ids = heap.get_objects_by_class("android.widget.TextView");
   593→            for (uint32_t tv_id : tv_ids) {
   594→                LayoutNode child_node = build_layout_tree(tv_id, heap, depth + 1);
   595→                node.children.push_back(child_node);
   596→            }
   597→        }
   598→    }
   599→    
   600→    // Limit depth to prevent infinite loops
   601→    if (depth > 10) {
   602→        return node;
   603→    }
   604→    
   605→    return node;
   606→}

   608→void RenderPipeline::layout_view(LayoutNode& node, int parent_left, int parent_top, 
   609→                                 int parent_width, int parent_height) {
   610→    if (node.bounds.is_empty()) {
   611→        // Default: use full parent size for Activity, reasonable size for others
   612→        if (node.view_class == "android.app.Activity") {
   613→            node.bounds.set(parent_left, parent_top, parent_width, parent_height);
   614→        } else if (node.view_class.find("TextView") != std::string::npos) {
   615→            // TextView gets centered area with padding
   616→            int padding = 40;
   617→            int text_height = font_.get_line_height() + 20;
   618→            node.bounds.set(parent_left + padding, parent_top + 100, 
   619→                          parent_width - 2 * padding, text_height);
   620→        } else {
   621→            node.bounds.set(parent_left + 10, parent_top + 10, 
   622→                          parent_width - 20, parent_height - 20);
   623→        }
   624→    }
   625→    
   626→    // Layout children
   627→    for (auto& child : node.children) {
   628→        layout_view(child, node.bounds.left, node.bounds.top, 
   629→                   node.bounds.width(), node.bounds.height());
   630→    }
   631→}

   768→std::vector<uint8_t> PNGWriter::compress_data(const uint8_t* data, size_t length) {
   769→    // Simple uncompressed deflate block for small images
   770→    // This is a minimal implementation - not optimal but works
   771→    
   772→    std::vector<uint8_t> compressed;
   773→    
   774→    // Deflate header: final block, no compression (type 00)
   775→    compressed.push_back(0x01);  // Final block, no compression
   776→    compressed.push_back(0x00);  // LEN low byte
   777→    compressed.push_back(0x00);  // LEN high byte  
   778→    compressed.push_back(0xFF);  // NLEN (one's complement of LEN)
   779→    compressed.push_back(0xFF);
   780→    
   781→    // Copy raw data
   782→    compressed.insert(compressed.end(), data, data + length);
   783→    
   784→    return compressed;
   785→}

   787→bool PNGWriter::write_png(const std::string& filename, const FrameBuffer& fb) {
   788→    std::ofstream file(filename, std::ios::binary);
   789→    if (!file.is_open()) {
   790→        return false;
   791→    }
   792→    
   793→    int width = fb.get_width();
   794→    int height = fb.get_height();
   795→    
   796→    // PNG signature
   797→    const uint8_t png_signature[8] = {137, 80, 78, 71, 13, 10, 26, 10};
   798→    file.write(reinterpret_cast<const char*>(png_signature), 8);
   799→    
   800→    // Helper lambda to write a chunk
   801→    auto write_chunk = [&](const char* type, const uint8_t* data, size_t length) {
   802→        uint32_t length_be = __builtin_bswap32(static_cast<uint32_t>(length));
   803→        file.write(reinterpret_cast<const char*>(&length_be), 4);
   804→        file.write(type, 4);
   805→        file.write(reinterpret_cast<const char*>(data), length);
   806→        
   807→        uint32_t crc_val = crc32(reinterpret_cast<const uint8_t*>(type), 4);
   808→        if (length > 0) {
   809→            crc_val = crc32(data, length, crc_val);
   810→        }
   811→        uint32_t crc_be = __builtin_bswap32(crc_val);
   812→        file.write(reinterpret_cast<const char*>(&crc_be), 4);
   813→    };
   814→    
   815→    // IHDR chunk
   816→    uint8_t ihdr[13];
   817→    ihdr[0] = (width >> 24) & 0xFF;  // Width (big-endian)
   818→    ihdr[1] = (width >> 16) & 0xFF;
   819→    ihdr[2] = (width >> 8) & 0xFF;
   820→    ihdr[3] = width & 0xFF;
   821→    ihdr[4] = (height >> 24) & 0xFF;  // Height (big-endian)
   822→    ihdr[5] = (height >> 16) & 0xFF;
   823→    ihdr[6] = (height >> 8) & 0xFF;
   824→    ihdr[7] = height & 0xFF;
   825→    ihdr[8] = 8;   // Bit depth
   826→    ihdr[9] = 2;   // Color type: RGB (we'll convert RGBA to RGB)
   827→    ihdr[10] = 0;  // Compression method
   828→    ihdr[11] = 0;  // Filter method
   829→    ihdr[12] = 0;  // Interlace method
   830→    write_chunk("IHDR", ihdr, 13);
   831→    
   832→    // IDAT chunk (image data)
   833→    // Convert RGBA to RGB and add filter bytes (none = 0)
   834→    size_t row_size = width * 3;  // RGB
   835→    std::vector<uint8_t> image_data;
   836→    image_data.reserve((row_size + 1) * height);
   837→    
   838→    for (int y = 0; y < height; y++) {
   839→        image_data.push_back(0);  // Filter type: None
   840→        
   841→        for (int x = 0; x < width; x++) {
   842→            RGBA pixel = fb.get_pixel(x, y);
   843→            image_data.push_back(pixel.r);
   844→            image_data.push_back(pixel.g);
   845→            image_data.push_back(pixel.b);
   846→        }
   847→    }
   848→    
   849→    std::vector<uint8_t> compressed = compress_data(image_data.data(), image_data.size());
   850→    write_chunk("IDAT", compressed.data(), compressed.size());
   851→    
   852→    // IEND chunk
   853→    write_chunk("IEND", nullptr, 0);
   854→    
   855→    file.close();
   856→    return true;
   857→}
