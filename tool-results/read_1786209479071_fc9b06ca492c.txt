    28→// Evidence File Writer
    29→// ============================================================================
    30→
    31→class EvidenceWriter {
    32→public:
    33→    explicit EvidenceWriter(const std::string& output_dir) : output_dir_(output_dir) {
    34→        fs::create_directories(output_dir);
    35→    }
    36→    
    37→    bool write_json(const std::string& filename, const json& data) {
    38→        std::string path = output_dir_ + "/" + filename;
    39→        std::ofstream file(path);
    40→        if (!file.is_open()) {
    41→            std::cerr << "[ERROR] Cannot write to: " << path << std::endl;
    42→            return false;
    43→        }
    44→        file << std::setw(2) << data << std::endl;
    45→        file.close();
    46→        
    47→        std::cout << "[EVIDENCE] Written: " << path << std::endl;
    48→        return true;
    49→    }
    50→    
    51→    bool write_markdown(const std::string& filename, const std::string& content) {
    52→        std::string path = output_dir_ + "/" + filename;
    53→        std::ofstream file(path);
    54→        if (!file.is_open()) {
    55→            std::cerr << "[ERROR] Cannot write to: " << path << std::endl;
    56→            return false;
    57→        }
    58→        file << content << std::endl;
    59→        file.close();
    60→        
    61→        std::cout << "[EVIDENCE] Written: " << path << std::endl;
    62→        return true;
    63→    }
    64→
    65→private:
    66→    std::string output_dir_;
    67→};

   196→    void run_task1_framebuffer() {
   197→        std::cout << "\n--- Task 1: FrameBuffer System ---\n";
   198→        
   199→        json evidence;
   200→        evidence["task"] = "FrameBuffer System";
   201→        evidence["status"] = "RUNNING";
   202→        
   203→        try {
   204→            // Get framebuffer from pipeline
   205→            const FrameBuffer& fb = pipeline_.get_framebuffer();
   206→            
   207→            evidence["framebuffer_info"] = fb.to_info_json();
   208→            evidence["width"] = fb.get_width();
   209→            evidence["height"] = fb.get_height();
   210→            evidence["pixel_count"] = static_cast<int>(fb.get_pixel_count());
   211→            evidence["initial_color"] = fb.get_pixel(0, 0).to_json();  // Should be white
   212→            
   213→            // Test clear operation
   214→            FrameBuffer test_fb(100, 100);
   215→            test_fb.clear(RGBA(255, 0, 0));  // Red
   216→            evidence["clear_test_color"] = test_fb.get_pixel(50, 50).to_json();
   217→            evidence["clear_works"] = (test_fb.get_pixel(50, 50).r == 255 &&
   218→                                       test_fb.get_pixel(50, 50).g == 0 &&
   219→                                       test_fb.get_pixel(50, 50).b == 0);
   220→            
   221→            // Test set pixel
   222→            test_fb.set_pixel(10, 10, RGBA(0, 255, 0));  // Green
   223→            evidence["set_pixel_test"] = test_fb.get_pixel(10, 10).to_json();
   224→            evidence["set_pixel_works"] = (test_fb.get_pixel(10, 10).g == 255);
   225→            
   226→            // Bounds checking
   227→            test_fb.set_pixel(-1, -1, RGBA(255, 255, 255));  // Out of bounds
   228→            evidence["bounds_check"] = true;  // Should not crash
   229→            
   230→            evidence["status"] = "COMPLETE";
   231→            
   232→            result_.task1_framebuffer = evidence.value("clear_works", false) &&
   233→                                      evidence.value("set_pixel_works", false) &&
   234→                                      evidence.value("bounds_check", false);
   235→            if (result_.task1_framebuffer) result_.total_passed++;
   236→            
   237→        } catch (const std::exception& e) {
   238→            evidence["status"] = "ERROR";
   239→            evidence["error_message"] = e.what();
   240→        }
   241→        
   242→        writer_.write_json("framebuffer_info.json", evidence);
   243→        std::cout << "Task 1: " << (result_.task1_framebuffer ? "PASS" : "FAIL") << "\n";

   305→    void run_task3_textview_measurement() {

   307→        
   308→        json evidence;
   309→        evidence["task"] = "TextView Measurement";
   310→        evidence["status"] = "RUNNING";
   311→        evidence["target_text"] = TARGET_TEXT;
   312→        
   313→        try {
   314→            BitmapFont font;
   315→            
   316→            // Measure the target text
   317→            auto metrics = font.measure_text(TARGET_TEXT);
   318→            
   319→            evidence["measured_width"] = metrics.width;
   320→            evidence["measured_height"] = metrics.height;
   321→            evidence["ascent"] = metrics.ascent;
   322→            evidence["descent"] = metrics.descent;
   323→            evidence["line_height"] = font.get_line_height();
   324→            evidence["baseline_offset"] = font.get_baseline_offset();
   325→            
   326→            // Verify measurements are reasonable
   327→            bool width_reasonable = (metrics.width > 0 && metrics.width < 500);
   328→            bool height_reasonable = (metrics.height > 0 && metrics.height <= font.get_line_height());
   329→            
   330→            evidence["measurements_valid"] = width_reasonable && height_reasonable;
   331→            
   332→            // Character-by-character breakdown
   333→            json char_measurements = json::array();
   334→            int total_advance = 0;
   335→            for (char c : TARGET_TEXT) {
   336→                const BitmapFont::Glyph* glyph = font.get_glyph(c);
   337→                json cm;
   338→                cm["character"] = std::string(1, c);
   339→                cm["advance"] = glyph->advance;
   340→                cm["cumulative_x"] = total_advance;
   341→                total_advance += glyph->advance;
   342→                char_measurements.push_back(cm);
   343→            }
   344→            evidence["character_measurements"] = char_measurements;
   345→            evidence["total_advance"] = total_advance;
   346→            evidence["matches_measured_width"] = (total_advance == metrics.width);
   347→            
   348→            // Font info
   349→            evidence["font_info"] = font.to_info_json();
   350→            
   351→            evidence["status"] = "COMPLETE";
   352→            
   353→            result_.task3_measurement = evidence.value("measurements_valid", false) &&
   354→                                     evidence.value("matches_measured_width", false);
   355→            if (result_.task3_measurement) result_.total_passed++;
   356→            
   357→        } catch (const std::exception& e) {
   358→            evidence["status"] = "ERROR";
   359→            evidence["error_message"] = e.what();
   360→        }
   361→        
   362→        writer_.write_json("measure_trace.json", evidence);
   363→        std::cout << "Task 3: " << (result_.task3_measurement ? "PASS" : "FAIL") << "\n";
   364→    }

   473→    void run_task6_text_rendering() {
   474→        std::cout << "--- Task 6: Text Rendering ---\n";
   475→        
   476→        json evidence;
   477→        evidence["task"] = "Text Rendering";
   478→        evidence["status"] = "RUNNING";
   479→        evidence["target"] = TARGET_TEXT;
   480→        
   481→        try {
   482→            // Get detailed text render trace
   483→            evidence["text_render_trace"] = pipeline_.get_text_render_trace();
   484→            
   485→            // Verify text content in canvas commands
   486→            const SoftwareCanvas& canvas = pipeline_.get_canvas();
   487→            bool found_target_text = false;
   488→            
   489→            for (const auto& cmd : canvas.get_commands()) {
   490→                if (cmd.type == "drawText" && cmd.params.value("text", "") == TARGET_TEXT) {
   491→                    found_target_text = true;
   492→                    
   493→                    evidence["drawText_command_details"] = cmd.to_json();
   494→                    evidence["position_x"] = cmd.params["position"]["x"];
   495→                    evidence["position_y"] = cmd.params["position"]["y"];
   496→                    evidence["color_used"] = cmd.params["color"];
   497→                    evidence["measured_width"] = cmd.params["measured_width"];
   498→                    evidence["measured_height"] = cmd.params["measured_height"];
   499→                    break;
   500→                }
   501→            }
   502→            
   503→            evidence["target_text_found_in_commands"] = found_target_text;
   504→            
   505→            // Sample pixels from framebuffer where text should be
   506→            const FrameBuffer& fb = pipeline_.get_framebuffer();
   507→            
   508→            // Background should be grey
   509→            RGBA bg_pixel = fb.get_pixel(10, 10);
   510→            evidence["background_sample"] = bg_pixel.to_json();
   511→            evidence["background_is_greyish"] = (bg_pixel.r > 200 && bg_pixel.g > 200 && bg_pixel.b > 200);
   512→            
   513→            // Find a pixel that should have text (dark color)
   514→            bool found_dark_pixel = false;
   515→            for (int y = 90; y < 120; y++) {  // Around where text should be
   516→                for (int x = 45; x < 300; x++) {
   517→                    RGBA p = fb.get_pixel(x, y);
   518→                    if (p.r < 100 && p.g < 100 && p.b < 100) {  // Dark pixel = text
   519→                        evidence["text_pixel_sample"] = p.to_json();
   520→                        evidence["text_pixel_location"] = {{"x", x}, {"y", y}};
   521→                        found_dark_pixel = true;
   522→                        goto done_search;
   523→                    }
   524→                }
   525→            }
   526→            done_search:
   527→            
   528→            evidence["found_text_pixels"] = found_dark_pixel;
   529→            
   530→            evidence["status"] = "COMPLETE";
   531→            
   532→            result_.task6_text_rendering = found_target_text && found_dark_pixel;
   533→            if (result_.task6_text_rendering) result_.total_passed++;
   534→            
   535→        } catch (const std::exception& e) {
   536→            evidence["status"] = "ERROR";
   537→            evidence["error_message"] = e.what();
   538→        }
   539→        
   540→        writer_.write_json("text_render_trace.json", evidence);
   541→        std::cout << "Task 6: " << (result_.task6_text_rendering ? "PASS" : "FAIL") << "\n";
   542→    }

   567→            uint64_t file_size = 0;
   568→            if (file_exists) {
   569→                file_size = fs::file_size(png_path);
   570→            }
   571→            
   572→            evidence["file_exists"] = file_exists;
   573→            evidence["file_size_bytes"] = file_size;
   574→            evidence["file_size_reasonable"] = (file_size > 1000 && file_size < 10000000);  // Between 1KB and 10MB
   575→            
   576→            // Verify it's a valid PNG by checking signature
   577→            bool valid_png_signature = false;
   578→            if (file_exists && file_size >= 8) {
   579→                std::ifstream png_file(png_path, std::ios::binary);
   580→                uint8_t signature[8];
   581→                png_file.read(reinterpret_cast<char*>(signature), 8);
   582→                
   583→                // PNG signature: 89 50 4E 47 0D 0A 1A 0A
   584→                const uint8_t expected_sig[8] = {137, 80, 78, 71, 13, 10, 26, 10};
   585→                valid_png_signature = (memcmp(signature, expected_sig, 8) == 0);
   586→                
   587→                png_file.close();
   588→            }
   589→            
   590→            evidence["valid_png_signature"] = valid_png_signature;
   591→            
   592→            evidence["status"] = "COMPLETE";
   593→            
   594→            result_.task7_screenshot_png = write_ok && file_exists && 
   595→                                         evidence.value("file_size_reasonable", false) &&
   596→                                         valid_png_signature;
   597→            if (result_.task7_screenshot_png) result_.total_passed++;
   598→            
   599→        } catch (const std::exception& e) {
   600→            evidence["status"] = "ERROR";
   601→            evidence["error_message"] = e.what();
   602→        }
   603→        
   604→        writer_.write_json("screenshot_generation_evidence.json", evidence);
   605→        std::cout << "Task 7: " << (result_.task7_screenshot_png ? "PASS" : "FAIL") << "\n";
   606→    }
