    59→    std::string descriptor;  // e.g., "(Ljava/lang/String;)V"

    91→class ClassMetadata {
    92→public:
    93→    ClassMetadata() {}
    94→    
    95→    ClassMetadata(const std::string& class_name, 
    96→                  const std::string& parent_class = "",
    97→                  const std::string& descriptor = "")
    98→        : class_name_(class_name)
    99→        , parent_class_(parent_class)
   100→        , descriptor_(descriptor) 
   101→    {}
   102→    
   103→    // Identity
   104→    const std::string& get_class_name() const { return class_name_; }
   105→    const std::string& get_parent_class() const { return parent_class_; }
   106→    const std::string& get_descriptor() const { return descriptor_; }
   107→    
   108→    void set_class_name(const std::string& name) { class_name_ = name; }
   109→    void set_parent_class(const std::string& parent) { parent_class_ = parent; }
   110→    void set_descriptor(const std::string& desc) { descriptor_ = desc; }
   111→    
   112→    // Methods management
   113→    void add_method(const MethodMetadata& method) {
   114→        methods_[method.name] = method;
   115→        method_list_.push_back(method.name);
   116→    }
   117→    
   118→    void add_method(const std::string& name, const std::string& descriptor,
   119→                    bool is_ctor = false, bool is_virtual = true) {
   120→        MethodMetadata m;
   121→        m.name = name;
   122→        m.descriptor = descriptor;
   123→        m.is_constructor = is_ctor;
   124→        m.is_virtual = is_virtual;
   125→        add_method(m);
   126→    }
   127→    
   128→    bool has_method(const std::string& name) const {
   129→        return methods_.count(name) > 0;
   130→    }
   131→    
   132→    std::optional<MethodMetadata> get_method(const std::string& name) const {
   133→        auto it = methods_.find(name);
   134→        if (it != methods_.end()) {
   135→            return it->second;
   136→        }
   137→        return std::nullopt;
   138→    }
   139→    
   140→    const std::vector<std::string>& get_method_list() const { return method_list_; }
   141→    size_t get_method_count() const { return methods_.size(); }
   142→    
   143→    // Fields management
   144→    void add_field(const FieldMetadata& field) {
   145→        fields_[field.name] = field;
   146→        field_list_.push_back(field.name);
   147→    }
   148→    
   149→    void add_field(const std::string& name, const std::string& type,
   150→                   bool is_static = false, bool is_public = true) {
   151→        FieldMetadata f;
   152→        f.name = name;
   153→        f.type = type;
   154→        f.is_static = is_static;
   155→        f.is_public = is_public;
   156→        add_field(f);
   157→    }
   158→    
   159→    bool has_field(const std::string& name) const {
   160→        return fields_.count(name) > 0;
   161→    }
   162→    
   163→    std::optional<FieldMetadata> get_field(const std::string& name) const {
   164→        auto it = fields_.find(name);
   165→        if (it != fields_.end()) {
   166→            return it->second;
   167→        }
   168→        return std::nullopt;
   169→    }
   170→    
   171→    const std::vector<std::string>& get_field_list() const { return field_list_; }
   172→    size_t get_field_count() const { return fields_.size(); }
   173→    
   174→    // Inheritance helpers
   175→    bool has_parent() const { return !parent_class_.empty(); }
   176→    bool is_derived_from(const std::string& base_class) const {
   177→        if (class_name_ == base_class) return true;
   178→        if (parent_class_.empty()) return false;
   179→        // Would need registry for full check
   180→        return parent_class_ == base_class;
   181→    }
   182→    
   183→    // Serialization
   184→    json to_json() const {
   185→        json j;
   186→        j["class_name"] = class_name_;
   187→        j["descriptor"] = descriptor_;
   188→        j["parent_class"] = parent_class_;
   189→        
   190→        json methods_arr = json::array();
   191→        for (const auto& name : method_list_) {
   192→            if (methods_.count(name)) {
   193→                methods_arr.push_back(methods_.at(name).to_json());
   194→            }
   195→        }
   196→        j["methods"] = methods_arr;
   197→        
   198→        json fields_arr = json::array();
   199→        for (const auto& name : field_list_) {
   200→            if (fields_.count(name)) {
   201→                fields_arr.push_back(fields_.at(name).to_json());
   202→            }
   203→        }
   204→        j["fields"] = fields_arr;
   205→        
   206→        j["method_count"] = get_method_count();
   207→        j["field_count"] = get_field_count();
   208→        
   209→        return j;
   210→    }
   211→
   212→private:
   213→    std::string class_name_;
   214→    std::string parent_class_;
   215→    std::string descriptor_;
   216→    
   217→    std::map<std::string, MethodMetadata> methods_;
   218→    std::vector<std::string> method_list_;
   219→    
   220→    std::map<std::string, FieldMetadata> fields_;
   221→    std::vector<std::string> field_list_;
   222→};

   228→/**
   229→ * Base class for all runtime objects in MiniAndroid.
   230→ * Every object created during DEX execution derives from this.
   231→ */
   232→class RuntimeObject {
   233→public:
   234→    virtual ~RuntimeObject() = default;
   235→    
   236→    // Object identity
   237→    uint32_t get_object_id() const { return object_id_; }
   238→    const std::string& get_runtime_class() const { return runtime_class_; }
   239→    const std::string& get_class_descriptor() const { return class_descriptor_; }
   240→    
   241→    // Lifetime management
   242→    ObjectLifetime get_lifetime() const { return lifetime_; }
   243→    void set_lifetime(ObjectLifetime state) { lifetime_ = state; }
   244→    
   245→    bool is_active() const { return lifetime_ == ObjectLifetime::ACTIVE; }
   246→    bool is_initialized() const { 
   247→        return lifetime_ == ObjectLifetime::ACTIVE || 
   248→               lifetime_ == ObjectLifetime::FINALIZING; 
   249→    }
   250→    
   251→    // Creation metadata
   252→    uint64_t get_creation_time() const { return creation_timestamp_; }
   253→    uint32_t get_creator_pc() const { return creator_pc_; }
   254→    uint64_t get_creation_sequence() const { return creation_sequence_; }
   255→    
   256→    // Type checking
   257→    virtual bool is_instance_of(const std::string& class_name) const {
   258→        return runtime_class_ == class_name;
   259→    }
   260→    
   261→    virtual bool is_kind_of(const std::string& base_class) const {
   262→        // Override in derived classes for proper inheritance
   263→        return runtime_class_ == base_class;
   264→    }
   265→    
   266→    // String representation
   267→    virtual std::string to_string() const {
   268→        std::ostringstream oss;
   269→        oss << "RuntimeObject[id=" << object_id_ 
   270→            << ", class=" << runtime_class_
   271→            << ", state=" << lifetime_to_string(lifetime_)
   272→            << "]";
   273→        return oss.str();
   274→    }
   275→    
   276→    // JSON serialization for evidence
   277→    virtual json to_json() const {
   278→        json obj;
   279→        obj["object_id"] = object_id_;
   280→        obj["runtime_class"] = runtime_class_;
   281→        obj["class_descriptor"] = class_descriptor_;
   282→        obj["lifetime_state"] = lifetime_to_string(lifetime_);
   283→        obj["creation_timestamp"] = creation_timestamp_;
   284→        obj["creator_pc"] = creator_pc_;
   285→        obj["creation_sequence"] = creation_sequence_;
   286→        obj["is_initialized"] = is_initialized();
   287→        return obj;
   288→    }
   289→    
   290→protected:
   291→    RuntimeObject(uint32_t id, const std::string& cls, const std::string& desc,
   292→                 uint32_t pc, uint64_t seq)
   293→        : object_id_(id)
   294→        , runtime_class_(cls)
   295→        , class_descriptor_(desc)
   296→        , creator_pc_(pc)
   297→        , creation_sequence_(seq)
   298→        , lifetime_(ObjectLifetime::ALLOCATED)
   299→    {
   300→        creation_timestamp_ = static_cast<uint64_t>(
   301→            std::chrono::duration_cast<std::chrono::milliseconds>(
   302→                std::chrono::system_clock::now().time_since_epoch()
   303→            ).count()
   304→        );
   305→    }
   306→    
   307→    void set_runtime_class(const std::string& cls) { runtime_class_ = cls; }
   308→    
   309→private:
   310→    uint32_t object_id_;
   311→    std::string runtime_class_;
   312→    std::string class_descriptor_;
   313→    
   314→    ObjectLifetime lifetime_;
   315→    uint64_t creation_timestamp_;
   316→    uint32_t creator_pc_;
   317→    uint64_t creation_sequence_;
   318→};

   323→
   324→class ObjectIdManager {
   325→public:
   326→    ObjectIdManager() : next_id_(1), total_allocated_(0), total_freed_(0) {}
   327→    
   328→    uint32_t allocate_id() {
   329→        uint32_t id = next_id_++;
   330→        total_allocated_++;
   331→        active_ids_.insert(id);
   332→        allocation_log_.push_back({id, "ALLOCATED"});
   333→        return id;
   334→    }
   335→    
   336→    void free_id(uint32_t id) {
   337→        if (active_ids_.erase(id) > 0) {
   338→            total_freed_++;
   339→            allocation_log_.push_back({id, "FREED"});
   340→        }
   341→    }
   342→    
   343→    bool is_active(uint32_t id) const {
   344→        return active_ids_.count(id) > 0;
   345→    }
   346→    
   347→    size_t get_active_count() const { return active_ids_.size(); }
   348→    uint32_t get_total_allocated() const { return total_allocated_; }
   349→    uint32_t get_total_freed() const { return total_freed_; }
   350→    
   351→    std::vector<uint32_t> get_all_active_ids() const {
   352→        return std::vector<uint32_t>(active_ids_.begin(), active_ids_.end());
   353→    }
   354→    
   355→    json to_json() const {
   356→        json j;
   357→        j["next_id"] = next_id_;
   358→        j["total_allocated"] = total_allocated_;
   359→        j["total_freed"] = total_freed_;
   360→        j["active_count"] = get_active_count();
   361→        
   362→        json log = json::array();
   363→        for (const auto& entry : allocation_log_) {
   364→            json e;
   365→            e["object_id"] = entry.first;
   366→            e["action"] = entry.second;
   367→            log.push_back(e);
   368→        }
   369→        j["allocation_log"] = log;
   370→        
   371→        return j;
   372→    }
   373→
   374→private:
   375→    uint32_t next_id_;
   376→    uint32_t total_allocated_;
   377→    uint32_t total_freed_;
   378→    std::set<uint32_t> active_ids_;
   379→    std::vector<std::pair<uint32_t, std::string>> allocation_log_;
   380→};

   386→struct AllocationRecord {
   387→    uint32_t object_id;
   388→    std::string class_descriptor;
   389→    uint32_t pc;
   390→    uint64_t sequence;
   391→    uint64_t timestamp_ms;
   392→    ObjectLifetime final_state;
   393→    
   394→    json to_json() const {
   395→        json j;
   396→        j["object_id"] = object_id;
   397→        j["class_descriptor"] = class_descriptor;
   398→        j["pc_hex"] = "0x" + [&]() {
   399→            std::ostringstream oss;
   400→            oss << std::hex << pc;
   401→            return oss.str();
   402→        }();
   403→        j["sequence"] = sequence;
   404→        j["timestamp_ms"] = timestamp_ms;
   405→        j["final_state"] = lifetime_to_string(final_state);
   406→        return j;
   407→    }
   408→};

   410→class EnhancedObjectHeap {
   411→public:
   412→    EnhancedObjectHeap() : allocation_sequence_(0) {}
   413→    
   414→    /**
   415→     * Allocate a new runtime object
   416→     */
   417→    template<typename T, typename... Args>
   418→    T* allocate(const std::string& class_desc, uint32_t pc, Args&&... args) {
   419→        uint32_t id = id_manager_.allocate_id();
   420→        ++allocation_sequence_;
   421→        
   422→        T* obj = new T(id, class_desc, pc, allocation_sequence_, 
   423→                       std::forward<Args>(args)...);
   424→        
   425→        objects_[id] = std::unique_ptr<RuntimeObject>(obj);
   426→        
   427→        AllocationRecord record;
   428→        record.object_id = id;
   429→        record.class_descriptor = class_desc;
   430→        record.pc = pc;
   431→        record.sequence = allocation_sequence_;
   432→        record.timestamp_ms = static_cast<uint64_t>(
   433→            std::chrono::duration_cast<std::chrono::milliseconds>(
   434→                std::chrono::system_clock::now().time_since_epoch()
   435→            ).count()
   436→        );
   437→        record.final_state = ObjectLifetime::ALLOCATED;
   438→        
   439→        allocation_history_[id] = record;
   440→        
   441→        return obj;
   442→    }
   443→    
   444→    /**
   445→     * Register externally created object
   446→     */
   447→    void register_object(std::unique_ptr<RuntimeObject> obj, uint32_t pc) {
   448→        uint32_t id = obj->get_object_id();
   449→        ++allocation_sequence_;
   450→        
   451→        objects_[id] = std::move(obj);
   452→        
   453→        AllocationRecord record;
   454→        record.object_id = id;
   455→        record.class_descriptor = objects_[id]->get_class_descriptor();
   456→        record.pc = pc;
   457→        record.sequence = allocation_sequence_;
   458→        record.timestamp_ms = static_cast<uint64_t>(
   459→            std::chrono::duration_cast<std::chrono::milliseconds>(
   460→                std::chrono::system_clock::now().time_since_epoch()
   461→            ).count()
   462→        );
   463→        record.final_state = ObjectLifetime::ALLOCATED;
   464→        
   465→        allocation_history_[id] = record;
   466→    }
   467→    
   468→    /**
   469→     * Get object by ID with type casting
   470→     */
   471→    template<typename T>
   472→    T* get_as(uint32_t id) {
   473→        auto it = objects_.find(id);
   474→        if (it != objects_.end()) {
   475→            return dynamic_cast<T*>(it->second.get());
   476→        }
   477→        return nullptr;
   478→    }
   479→    
   480→    /**
   481→     * Get raw object pointer
   482→     */
   483→    RuntimeObject* get(uint32_t id) {
   484→        auto it = objects_.find(id);
   485→        if (it != objects_.end()) {
   486→            return it->second.get();
   487→        }
   488→        return nullptr;
   489→    }
   490→    
   491→    const RuntimeObject* get(uint32_t id) const {
   492→        auto it = objects_.find(id);
   493→        if (it != objects_.end()) {
   494→            return it->second.get();
   495→        }
   496→        return nullptr;
   497→    }
   498→    
   499→    /**
   500→     * Check if object exists and is of expected type
   501→     */
   502→    bool is_type(uint32_t id, const std::string& expected_class) const {
   503→        auto* obj = get(id);
   504→        if (!obj) return false;
   505→        return obj->is_instance_of(expected_class);
   506→    }
   507→    
   508→    /**
   509→     * Update object lifetime state
   510→     */
   511→    void set_lifetime(uint32_t id, ObjectLifetime state) {
   512→        auto* obj = get(id);
   513→        if (obj) {
   514→            obj->set_lifetime(state);
   515→        }
   516→        if (allocation_history_.count(id)) {
   517→            allocation_history_[id].final_state = state;
   518→        }
   519→    }
   520→    
   521→    /**
   522→     * Free an object
   523→     */
   524→    void free(uint32_t id) {
   525→        set_lifetime(id, ObjectLifetime::COLLECTED);
   526→        objects_.erase(id);
   527→        id_manager_.free_id(id);
   528→    }
   529→    
   530→    // Query methods
   531→    size_t size() const { return objects_.size(); }
   532→    bool contains(uint32_t id) const { return objects_.count(id) > 0; }
   533→    
   534→    std::vector<uint32_t> get_all_ids() const {
   535→        std::vector<uint32_t> ids;
   536→        for (const auto& pair : objects_) {
   537→            ids.push_back(pair.first);
   538→        }
   539→        return ids;
   540→    }
   541→    
   542→    std::vector<uint32_t> get_objects_by_class(const std::string& cls) const {
   543→        std::vector<uint32_t> result;
   544→        for (const auto& pair : objects_) {
   545→            if (pair.second->get_runtime_class() == cls) {
   546→                result.push_back(pair.first);
   547→            }
   548→        }
   549→        return result;
   550→    }
   551→    
   552→    // Statistics
   553→    const ObjectIdManager& get_id_manager() const { return id_manager_; }
   554→    uint64_t get_allocation_sequence() const { return allocation_sequence_; }
   555→    
   556→    // Serialization
   557→    json dump_objects() const {
   558→        json arr = json::array();
   559→        for (const auto& pair : objects_) {
   560→            arr.push_back(pair.second->to_json());
   561→        }
   562→        return arr;
   563→    }
   564→    
   565→    json dump_allocations() const {
   566→        json arr = json::array();
   567→        for (const auto& pair : allocation_history_) {
   568→            arr.push_back(pair.second.to_json());
   569→        }
   570→        return arr;
   571→    }
   572→    
   573→    json to_full_report() const {
   574→        json report;
   575→        report["heap_size"] = size();
   576→        report["total_allocated"] = id_manager_.get_total_allocated();
   577→        report["total_freed"] = id_manager_.get_total_freed();
   578→        report["active_objects"] = id_manager_.get_active_count();
   579→        report["allocation_sequence"] = allocation_sequence_;
   580→        report["objects"] = dump_objects();
   581→        report["allocation_history"] = dump_allocations();
   582→        report["id_manager"] = id_manager_.to_json();
   583→        return report;
   584→    }
   585→
   586→private:
   587→    std::map<uint32_t, std::unique_ptr<RuntimeObject>> objects_;
   588→    ObjectIdManager id_manager_;
   589→    uint64_t allocation_sequence_;
   590→    std::map<uint32_t, AllocationRecord> allocation_history_;
   591→};

   637→    void register_class(const std::string& class_name, 
   638→                        const std::string& parent_class,
   639→                        const std::string& descriptor = "") {
   640→        classes_[class_name] = {class_name, parent_class, descriptor, 0};
   641→        recalculate_depths();
   642→    }
   643→    
   644→    bool has_class(const std::string& class_name) const {
   645→        return classes_.count(class_name) > 0;
   646→    }
   647→    
   648→    std::optional<InheritanceNode> get_class(const std::string& class_name) const {
   649→        auto it = classes_.find(class_name);
   650→        if (it != classes_.end()) {
   651→            return it->second;
   652→        }
   653→        return std::nullopt;
   654→    }
   655→    
   656→    std::string get_parent(const std::string& class_name) const {
   657→        auto node = get_class(class_name);
   658→        if (node) {
   659→            return node->parent_class;
   660→        }
   661→        return "";
   662→    }
   663→    
   664→    /**
   665→     * Get full inheritance chain from this class to java.lang.Object
   666→     */
   667→    std::vector<std::string> get_chain(const std::string& class_name) const {
   668→        std::vector<std::string> chain;
   669→        std::string current = class_name;
   670→        
   671→        while (!current.empty() && classes_.count(current)) {
   672→            chain.push_back(current);
   673→            current = classes_.at(current).parent_class;
   674→            
   675→            // Prevent infinite loops
   676→            if (chain.size() > 20) break;
   677→        }
   678→        
   679→        return chain;
   680→    }
   681→    
   682→    /**
   683→     * Check if 'derived' is a subclass of 'base'
   684→     */
   685→    bool is_subclass(const std::string& derived, const std::string& base) const {
   686→        std::string current = derived;
   687→        int guard = 0;
   688→        
   689→        while (!current.empty() && guard < 30) {
   690→            if (current == base) return true;
   691→            
   692→            auto it = classes_.find(current);
   693→            if (it == classes_.end()) break;
   694→            
   695→            current = it->second.parent_class;
   696→            guard++;
   697→        }
   698→        
   699→        return false;
   700→    }
   701→    
   702→    /**
   703→     * Find path between two classes
   704→     */
   705→    InheritancePath find_path(const std::string& from, const std::string& to) const {
   706→        InheritancePath result;
   707→        result.from_class = from;
   708→        result.to_class = to;
   709→        
   710→        std::string current = from;
   711→        while (!current.empty()) {
   712→            result.path.push_back(current);
   713→            if (current == to) {
   714→                result.found = true;
   715→                break;
   716→            }
   717→            
   718→            auto it = classes_.find(current);
   719→            if (it == classes_.end()) break;
   720→            current = it->second.parent_class;
   721→            
   722→            if (result.path.size() > 20) break;
   723→        }
   724→        
   725→        return result;
   726→    }
   727→    
   728→    // Get all registered classes
   729→    std::vector<std::string> get_all_classes() const {
   730→        std::vector<std::string> result;
   731→        for (const auto& pair : classes_) {
   732→            result.push_back(pair.first);
   733→        }
   734→        return result;
   735→    }
   736→    
   737→    size_t get_class_count() const { return classes_.size(); }
   738→    
   739→    // Serialization
   740→    json to_json() const {
   741→        json j;
   742→        j["registry_name"] = "Android_Inheritance_Registry";
   743→        j["class_count"] = get_class_count();
   744→        
   745→        json classes_arr = json::array();
   746→        for (const auto& pair : classes_) {
   747→            classes_arr.push_back(pair.second.to_json());
   748→        }
   749→        j["classes"] = classes_arr;
   750→        
   751→        return j;
   752→    }
   753→    
   754→    json trace_inheritance(const std::string& class_name) const {
   755→        json trace;
   756→        trace["target_class"] = class_name;
   757→        trace["chain"] = get_chain(class_name);
   758→        
   759→        json details = json::array();
   760→        auto chain = get_chain(class_name);
   761→        for (size_t i = 0; i < chain.size(); i++) {
   762→            json node;
   763→            node["position"] = i;
   764→            node["class"] = chain[i];
   765→            if (i < chain.size() - 1) {
   766→                node["inherits_from"] = chain[i + 1];
   767→            } else {
   768→                node["inherits_from"] = "(root)";
   769→            }
   770→            details.push_back(node);
   771→        }
   772→        trace["details"] = details;
   773→        
   774→        return trace;
   775→    }
   776→
   777→private:
   778→    std::map<std::string, InheritanceNode> classes_;
   779→    
   780→    void initialize_core_hierarchy() {
   781→        // Java foundation
   782→        register_class("java.lang.Object", "", "Ljava/lang/Object;");
   783→        
   784→        // Android framework
   785→        register_class("android.content.Context", "java.lang.Object", "Landroid/content/Context;");
   786→        register_class("android.app.Activity", "android.content.Context", "Landroid/app/Activity;");
   787→        
   788→        // View hierarchy
   789→        register_class("android.view.View", "java.lang.Object", "Landroid/view/View;");
   790→        register_class("android.view.ViewGroup", "android.view.View", "Landroid/view/ViewGroup;");
   791→        register_class("android.widget.TextView", "android.view.View", "Landroid/widget/TextView;");
   792→        
   793→        recalculate_depths();
   794→    }
   795→    
   796→    void recalculate_depths() {
   797→        for (auto& pair : classes_) {
   798→            pair.second.depth = calculate_depth(pair.first);
   799→        }
   800→    }
   801→    
   802→    int calculate_depth(const std::string& class_name) const {
   803→        std::string current = class_name;
   804→        int depth = 0;
   805→        int guard = 0;
   806→        
   807→        while (!current.empty() && guard < 30) {
   808→            auto it = classes_.find(current);
   809→            if (it == classes_.end()) break;
   810→            current = it->second.parent_class;
   811→            depth++;
   812→            guard++;
   813→        }
   814→        
   815→        return depth;
   816→    }
   817→};
