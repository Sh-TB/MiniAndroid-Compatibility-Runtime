    68→void BitmapFont::initialize_ascii_glyphs() {
    69→    // EXP-092: Initialize glyphs_ from the auto-generated bitmap_font_table.
    70→    // Table index i corresponds to ASCII char (i + 32).
    71→    for (int i = 0; i < 95; i++) {
    72→        const auto& entry = miniandroid::renderer::bitmap_font_table[i];
    73→        glyphs_[i] = {entry.character, entry.width, entry.height, entry.advance, entry.bitmap};
    74→    }
    75→    fill_remaining_glyphs();
    76→}
    77→
    78→void BitmapFont::fill_remaining_glyphs() {
    79→    // EXP-092: With the auto-generated table, every slot is already populated.
    80→    // This function is kept for API compatibility with the previous design,
    81→    // but it is now effectively a no-op — every glyph already has its
    82→    // character/bitmap/advance set correctly.
    83→    //
    84→    // Safety check: if any glyph is still '\0' (shouldn't happen, but
    85→    // defends against future regressions), fill it with the space bitmap
    86→    // so we never dereference a null `bitmap` pointer.
    87→    static const uint8_t fallback_bitmap[16] = {
    88→        0x00, 0x00, 0x00, 0x00, 0x18, 0x18, 0x18, 0x18,
    89→        0x18, 0x18, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00
    90→    };
    91→    for (int i = 0; i < 95; i++) {
    92→        if (glyphs_[i].character == '\0') {
    93→            char c = static_cast<char>(i + 32);
    94→            glyphs_[i] = {c, 5, 16, 6, fallback_bitmap};
    95→        }
    96→    }
    97→}
    98→
    99→const BitmapFont::Glyph* BitmapFont::get_glyph(char c) const {
   100→    if (c >= 32 && c <= 126) {
   101→        return &glyphs_[c - 32];
   102→    }
   103→    return &glyphs_[0];  // Return space for unknown chars
   104→}

   106→BitmapFont::TextMetrics BitmapFont::measure_text(const std::string& text) const {
   107→    TextMetrics metrics{0, 16, 12, 4};
   108→    
   109→    for (char c : text) {
   110→        const Glyph* glyph = get_glyph(c);
   111→        metrics.width += glyph->advance;
   112→    }
   113→    
   114→    return metrics;
   115→}

   163→void SoftwareCanvas::draw_rect(float left, float top, float right, float bottom, RGBA color) {
   164→    int x1 = static_cast<int>(std::round(left));
   165→    int y1 = static_cast<int>(std::round(top));
   166→    int x2 = static_cast<int>(std::round(right));
   167→    int y2 = static_cast<int>(std::round(bottom));
   168→    
   169→    for (int y = y1; y < y2 && y < framebuffer_->get_height(); y++) {
   170→        for (int x = x1; x < x2 && x < framebuffer_->get_width(); x++) {
   171→            if (x >= 0 && y >= 0) {
   172→                framebuffer_->set_pixel(x, y, color);
   173→            }
   174→        }
   175→    }
   176→    
   177→    CanvasCommand cmd;
   178→    cmd.type = "drawRect";
   179→    cmd.sequence = ++command_sequence_;
   180→    cmd.params["rect"] = {{"left", x1}, {"top", y1}, {"right", x2}, {"bottom", y2}};
   181→    cmd.params["color"] = color.to_json();
   182→    commands_.push_back(cmd);
   183→}

   185→void SoftwareCanvas::draw_text(const std::string& text, float x, float y, RGBA color,
   186→                               const BitmapFont* font) {
   187→    const BitmapFont* use_font = font ? font : &default_font_;
   188→    
   189→    auto metrics = use_font->measure_text(text);
   190→    int start_x = static_cast<int>(std::round(x));
   191→    int start_y = static_cast<int>(std::round(y)) - use_font->get_baseline_offset();
   192→    
   193→    // Render each character
   194→    int current_x = start_x;
   195→    for (char c : text) {
   196→        const BitmapFont::Glyph* glyph = use_font->get_glyph(c);
   197→        
   198→        // Render glyph bitmap
   199→        for (int row = 0; row < glyph->height; row++) {
   200→            uint8_t bitmap_row = glyph->bitmap[row];
   201→            for (int col = 0; col < 8; col++) {
   202→                if (bitmap_row & (0x80 >> col)) {
   203→                    framebuffer_->set_pixel(current_x + col, start_y + row, color);
   204→                }
   205→            }
   206→        }
   207→        
   208→        current_x += glyph->advance;
   209→    }
   210→    
   211→    CanvasCommand cmd;
   212→    cmd.type = "drawText";
   213→    cmd.sequence = ++command_sequence_;
   214→    cmd.params["text"] = text;
   215→    cmd.params["position"] = {{"x", start_x}, {"y", start_y}};
   216→    cmd.params["color"] = color.to_json();
   217→    cmd.params["measured_width"] = metrics.width;
   218→    cmd.params["measured_height"] = metrics.height;
   219→    commands_.push_back(cmd);
   220→}

   232→// ============================================================================
   233→// EXP-088 Phase A4: SoftwareCanvas::draw_image
   234→//
   235→// Draws an RGBA pixel buffer (decoded from a PNG via PNGDecoder) into the
   236→// framebuffer at (dst_x, dst_y). Optional scaling to dst_w x dst_h.
   237→//
   238→// Uses nearest-neighbour sampling for scaling — no fancy interpolation,
   239→// because the source PNGs we care about (simplestopwatch icons) are tiny
   240→// and need to render at their natural size anyway.
   241→//
   242→// Pixels with alpha < 255 are alpha-blended onto the existing framebuffer
   243→// pixel using the standard "over" compositing operator. This is essential
   244→// because:
   245→//   - simplestopwatch's lock.png is grayscale+alpha with partial transparency
   246→//   - if we just stomped (R,G,B,0) over the framebuffer we'd lose the icon
   247→// ============================================================================
   248→
   249→void SoftwareCanvas::draw_image(const uint8_t* src_rgba, int src_w, int src_h,
   250→                                int dst_x, int dst_y,
   251→                                int dst_w, int dst_h) {
   252→    if (!src_rgba || src_w <= 0 || src_h <= 0) return;
   253→    if (dst_w <= 0)  dst_w = src_w;
   254→    if (dst_h <= 0)  dst_h = src_h;
   255→
   256→    int fb_w = framebuffer_->get_width();
   257→    int fb_h = framebuffer_->get_height();
   258→
   259→    for (int dy = 0; dy < dst_h; dy++) {
   260→        int sy = (dy * src_h) / dst_h;  // nearest-neighbour
   261→        if (sy < 0 || sy >= src_h) continue;
   262→        for (int dx = 0; dx < dst_w; dx++) {
   263→            int sx = (dx * src_w) / dst_w;
   264→            if (sx < 0 || sx >= src_w) continue;
   265→
   266→            const uint8_t* p = src_rgba + (sy * src_w + sx) * 4;
   267→            RGBA src(p[0], p[1], p[2], p[3]);
   268→
   269→            int x = dst_x + dx;
   270→            int y = dst_y + dy;
   271→            if (x < 0 || x >= fb_w || y < 0 || y >= fb_h) continue;
   272→
   273→            // Alpha-blend with existing pixel
   274→            RGBA dst = framebuffer_->get_pixel(x, y);
   275→            RGBA out = blend(src, dst);
   276→            framebuffer_->set_pixel(x, y, out);
   277→        }
   278→    }
   279→
   280→    CanvasCommand cmd;
   281→    cmd.type = "drawImage";
   282→    cmd.sequence = ++command_sequence_;
   283→    cmd.params["src_size"] = {{"w", src_w}, {"h", src_h}};
   284→    cmd.params["dst_pos"]   = {{"x", dst_x}, {"y", dst_y}};
   285→    cmd.params["dst_size"]  = {{"w", dst_w}, {"h", dst_h}};
   286→    commands_.push_back(cmd);
   287→}

   289→// ============================================================================
   290→// PNG Decoder — CAMPAIGN 010 R1: libpng-backed (replaces the EXP-088 A4
   291→// hand-written decoder after a differential benchmark over 7,036 real APK
   292→// PNGs showed the custom decoder failing 206 files (bit_depth 1/2/4) and
   293→// mis-decoding 3 tRNS files, while libpng decoded 100% byte-identically to
   294→// stb_image v2.30 and 1.65x faster than the custom code.
   295→//
   296→// libpng is the reference PNG implementation (libpng.org) and is the same
   297→// decoder Android's own stack uses (libjpeg-turbo + libpng inside Skia).
   298→// Coverage now includes everything the old code rejected:
   299→//   - bit depths 1/2/4/8/16 (16 stripped to 8 per Android Bitmap ARGB_8888)
   300→//   - color types 0/2/3/4/6 (palette via png_set_palette_to_rgb)
   301→//   - tRNS transparency (png_set_expand)  [the 3-file custom-decoder bug]
   302→//   - Adam7 interlacing (png_set_interlace_handling)
   303→// Output contract unchanged: RGBA8 buffer, .color_type_name mirrors the
   304→// source color type, .ok=false with descriptive error on failure.
   305→// ============================================================================
   306→
   307→namespace {
   308→
   309→// libpng memory reader (whole-file buffer context)
   310→struct PngMemCtx { const uint8_t* data; size_t size; size_t off; };
   311→
   312→void png_mem_read(png_structp png, png_bytep out, png_size_t n) {
   313→    PngMemCtx* c = static_cast<PngMemCtx*>(png_get_io_ptr(png));
   314→    if (c->off + n > c->size) {
   315→        png_error(png, "read past end of PNG buffer");
   316→        return;
   317→    }
   318→    std::memcpy(out, c->data + c->off, n);
   319→    c->off += n;
   320→}
   321→
   322→const char* png_color_type_name(png_byte ct) {
   323→    switch (ct) {
   324→        case 0: return "gray";
   325→        case 2: return "rgb";
   326→        case 3: return "palette→rgba";
   327→        case 4: return "ga";
   328→        case 6: return "rgba";
   329→        default: return "unknown";
   330→    }
   331→}

   335→DecodedImage PNGDecoder::decode(const std::vector<uint8_t>& png_bytes) {
   336→    DecodedImage result;
   337→    if (png_bytes.size() < 8) {
   338→        result.error = "PNG too small";
   339→        return result;
   340→    }
   341→    static const uint8_t SIG[8] = {137, 80, 78, 71, 13, 10, 26, 10};
   342→    if (std::memcmp(png_bytes.data(), SIG, 8) != 0) {
   343→        result.error = "bad PNG signature";
   344→        return result;
   345→    }
   346→
   347→    png_structp png = png_create_read_struct(PNG_LIBPNG_VER_STRING,
   348→                                             nullptr, nullptr, nullptr);
   349→    if (!png) {
   350→        result.error = "png_create_read_struct failed";
   351→        return result;
   352→    }
   353→    png_infop info = png_create_info_struct(png);
   354→    if (!info) {
   355→        png_destroy_read_struct(&png, nullptr, nullptr);
   356→        result.error = "png_create_info_struct failed";
   357→        return result;
   358→    }
   359→
   360→    PngMemCtx ctx{png_bytes.data(), png_bytes.size(), 0};
   361→    if (setjmp(png_jmpbuf(png))) {
   362→        png_destroy_read_struct(&png, &info, nullptr);
   363→        result.error = "libpng: invalid or corrupted PNG stream";
   364→        return result;
   365→    }
   366→
   367→    png_set_read_fn(png, &ctx, png_mem_read);
   368→    png_read_info(png, info);
   369→
   370→    png_byte orig_ct = png_get_color_type(png, info);
   371→    result.color_type_name = png_color_type_name(orig_ct);
   372→
   373→    // Android Bitmap contract: everything expands to ARGB_8888.
   374→    png_set_palette_to_rgb(png);
   375→    if (png_get_bit_depth(png, info) < 8) png_set_packing(png);
   376→    if (png_get_bit_depth(png, info) == 16) png_set_strip_16(png);
   377→    if (orig_ct == PNG_COLOR_TYPE_GRAY || orig_ct == PNG_COLOR_TYPE_GRAY_ALPHA)
   378→        png_set_gray_to_rgb(png);
   379→    // tRNS → full alpha channel. NOTE: png_set_expand alone does NOT expand
   380→    // tRNS for 8-bit RGB/gray (caught by the a4 fixture test) —
   381→    // png_set_tRNS_to_alpha covers palette/gray/RGB tRNS per PNG spec §11.3.2.
   382→    png_set_tRNS_to_alpha(png);
   383→    png_set_filler(png, 0xFF, PNG_FILLER_AFTER);
   384→    png_set_interlace_handling(png);
   385→    png_read_update_info(png, info);
   386→
   387→    result.width  = static_cast<int>(png_get_image_width(png, info));
   388→    result.height = static_cast<int>(png_get_image_height(png, info));
   389→    size_t rowbytes = png_get_rowbytes(png, info);
   390→    result.rgba.resize(rowbytes * result.height);
   391→    std::vector<png_bytep> rows(result.height);
   392→    for (int y = 0; y < result.height; y++)
   393→        rows[y] = result.rgba.data() + static_cast<size_t>(y) * rowbytes;
   394→
   395→    png_read_image(png, rows.data());
   396→    png_read_end(png, nullptr);
   397→    png_destroy_read_struct(&png, &info, nullptr);
   398→
   399→    result.ok = true;
   400→    return result;
   401→}

   403→DecodedImage PNGDecoder::decode_file(const std::string& path) {
   404→    std::ifstream f(path, std::ios::binary | std::ios::ate);
   405→    if (!f.is_open()) {
   406→        DecodedImage r;
   407→        r.error = "cannot open file: " + path;
   408→        return r;
   409→    }
   410→    auto sz = f.tellg();
   411→    f.seekg(0, std::ios::beg);
   412→    std::vector<uint8_t> buf(sz);
   413→    f.read(reinterpret_cast<char*>(buf.data()), sz);
   414→    return decode(buf);
   415→}

   460→bool RenderPipeline::render(runtime::EnhancedObjectHeap& heap, const std::string& activity_text) {
   461→    auto total_start = std::chrono::high_resolution_clock::now();
   462→    
   463→    stage_ = RenderStage::LAYOUT;
   464→    
   465→    // Stage 1: Layout
   466→    auto layout_start = std::chrono::high_resolution_clock::now();
   467→    bool layout_ok = perform_layout(heap);
   468→    auto layout_end = std::chrono::high_resolution_clock::now();
   469→    stats_.layout_time_ms = std::chrono::duration<double, std::milli>(layout_end - layout_start).count();
   470→    
   471→    if (!layout_ok) {
   472→        stage_ = RenderStage::ERROR;
   473→        return false;
   474→    }
   475→    
   476→    // Stage 2: Measure
   477→    stage_ = RenderStage::MEASURE;
   478→    auto measure_start = std::chrono::high_resolution_clock::now();
   479→    bool measure_ok = perform_measurement();
   480→    auto measure_end = std::chrono::high_resolution_clock::now();
   481→    stats_.measure_time_ms = std::chrono::duration<double, std::milli>(measure_end - measure_start).count();
   482→    
   483→    if (!measure_ok) {
   484→        stage_ = RenderStage::ERROR;
   485→        return false;
   486→    }
   487→    
   488→    // Stage 3: Draw
   489→    stage_ = RenderStage::DRAW;
   490→    auto draw_start = std::chrono::high_resolution_clock::now();
   491→    bool draw_ok = perform_draw(heap, activity_text);
   492→    auto draw_end = std::chrono::high_resolution_clock::now();
   493→    stats_.draw_time_ms = std::chrono::duration<double, std::milli>(draw_end - draw_start).count();
   494→    
   495→    if (!draw_ok) {
   496→        stage_ = RenderStage::ERROR;
   497→        return false;
   498→    }
   499→    
   500→    // Complete
   501→    auto total_end = std::chrono::high_resolution_clock::now();
   502→    stats_.total_time_ms = std::chrono::duration<double, std::milli>(total_end - total_start).count();
   503→    stats_.pixels_written = framebuffer_.get_draw_count();
   504→    stats_.total_commands_issued = static_cast<int>(canvas_.get_command_count());
   505→    
   506→    stage_ = RenderStage::COMPLETE;
   507→    frame_number_++;
   508→    
   509→    return true;
   510→}
   511→
   512→bool RenderPipeline::perform_layout(runtime::EnhancedObjectHeap& heap) {
   513→    // Find Activity and build layout tree from it
   514→    auto activity_ids = heap.get_objects_by_class("android.app.Activity");
   515→    
   516→    if (activity_ids.empty()) {
   517→        // No Activity found, try to build from all objects
   518→        auto all_ids = heap.get_all_ids();
   519→        if (!all_ids.empty()) {
   520→            layout_tree_ = build_layout_tree(all_ids[0], heap);
   521→            return true;
   522→        }
   523→        return false;
   524→    }
   525→    
   526→    layout_tree_ = build_layout_tree(activity_ids[0], heap);
   527→    
   528→    // Apply initial layout to root (full screen)
   529→    layout_view(layout_tree_, 0, 0, framebuffer_.get_width(), framebuffer_.get_height());
   530→    
   531→    stats_.total_views_processed++;
   532→    
   533→    return true;
   534→}

   579→LayoutNode RenderPipeline::build_layout_tree(uint32_t object_id, runtime::EnhancedObjectHeap& heap, int depth) {
   580→    LayoutNode node;
   581→    node.object_id = object_id;
   582→    
   583→    auto* obj = heap.get(object_id);
   584→    if (obj) {
   585→        node.view_class = obj->get_runtime_class();
   586→        
   587→        // Try to get geometry from View/TextView objects
   588→        auto* view_obj = dynamic_cast<const runtime::ViewRuntimeObject*>(obj);
   589→        if (view_obj) {
   590→            auto geom = view_obj->get_geometry();
   591→            node.bounds.set(geom.left, geom.top, geom.width, geom.height);
   592→        }
   593→        
   594→        // For Activity, find child TextViews and add them as children
   595→        if (node.view_class == "android.app.Activity") {
   596→            auto tv_ids = heap.get_objects_by_class("android.widget.TextView");
   597→            for (uint32_t tv_id : tv_ids) {
   598→                LayoutNode child_node = build_layout_tree(tv_id, heap, depth + 1);
   599→                node.children.push_back(child_node);
   600→            }
   601→        }
   602→    }
   603→    
   604→    // Limit depth to prevent infinite loops
   605→    if (depth > 10) {
   606→        return node;
   607→    }
   608→    
   609→    return node;
   610→}

   612→void RenderPipeline::layout_view(LayoutNode& node, int parent_left, int parent_top, 
   613→                                 int parent_width, int parent_height) {
   614→    if (node.bounds.is_empty()) {
   615→        // Default: use full parent size for Activity, reasonable size for others
   616→        if (node.view_class == "android.app.Activity") {
   617→            node.bounds.set(parent_left, parent_top, parent_width, parent_height);
   618→        } else if (node.view_class.find("TextView") != std::string::npos) {
   619→            // TextView gets centered area with padding
   620→            int padding = 40;
   621→            int text_height = font_.get_line_height() + 20;
   622→            node.bounds.set(parent_left + padding, parent_top + 100, 
   623→                          parent_width - 2 * padding, text_height);
   624→        } else {
   625→            node.bounds.set(parent_left + 10, parent_top + 10, 
   626→                          parent_width - 20, parent_height - 20);
   627→        }
   628→    }
   629→    
   630→    // Layout children
   631→    for (auto& child : node.children) {
   632→        layout_view(child, node.bounds.left, node.bounds.top, 
   633→                   node.bounds.width(), node.bounds.height());
   634→    }

   707→// PNG Writer — CAMPAIGN 010 R1: libpng-backed (replaces the hand-written
   708→// chunk/zlib encoder; same pixel content, standard libpng byte stream).
   709→// ============================================================================
   710→
   711→bool PNGWriter::write_png(const std::string& filename, const FrameBuffer& fb) {
   712→    std::ofstream file(filename, std::ios::binary);
   713→    if (!file.is_open()) {
   714→        return false;
   715→    }
   716→    file.close();
   717→
   718→    int width = fb.get_width();
   719→    int height = fb.get_height();
   720→
   721→    FILE* fp = std::fopen(filename.c_str(), "wb");
   722→    if (!fp) return false;
   723→
   724→    png_structp png = png_create_write_struct(PNG_LIBPNG_VER_STRING,
   725→                                              nullptr, nullptr, nullptr);
   726→    if (!png) { std::fclose(fp); return false; }
   727→    png_infop info = png_create_info_struct(png);
   728→    if (!info) {
   729→        png_destroy_write_struct(&png, nullptr);
   730→        std::fclose(fp);
   731→        return false;
   732→    }
   733→    if (setjmp(png_jmpbuf(png))) {
   734→        png_destroy_write_struct(&png, &info);
   735→        std::fclose(fp);
   736→        return false;
   737→    }
   738→    png_init_io(png, fp);
   739→    // Same output pixel contract as the custom encoder: 8-bit RGB.
   740→    png_set_IHDR(png, info, width, height, 8, PNG_COLOR_TYPE_RGB,
   741→                 PNG_INTERLACE_NONE, PNG_COMPRESSION_TYPE_DEFAULT,
   742→                 PNG_FILTER_TYPE_DEFAULT);
   743→    png_write_info(png, info);
   744→
   745→    std::vector<uint8_t> row(size_t(width) * 3);
   746→    for (int y = 0; y < height; y++) {
   747→        for (int x = 0; x < width; x++) {
   748→            RGBA pixel = fb.get_pixel(x, y);
   749→            row[size_t(x) * 3 + 0] = pixel.r;
   750→            row[size_t(x) * 3 + 1] = pixel.g;
   751→            row[size_t(x) * 3 + 2] = pixel.b;
   752→        }
   753→        png_write_row(png, row.data());
   754→    }
   755→    png_write_end(png, info);
   756→    png_destroy_write_struct(&png, &info);
   757→    std::fclose(fp);
   758→    return true;
   759→}

   787→// ============================================================================
   788→// EXP-097 §5: WebPDecoder — libwebp-backed WebP → RGBA decoder.
   789→//
   790→// Per AOSP `BitmapFactory.nativeDecodeByteArray`:
   791→//   1. Validate WebP RIFF header (libwebp's WebPGetInfo does this).
   792→//   2. Allocate output buffer of width * height * 4 bytes (RGBA).
   793→//   3. Call WebPDecodeRGBA which delegates to the right internal decoder
   794→//      (VP8 lossy, VP8L lossless, or VP8X extended). For animated WebPs
   795→//      the FIRST frame is returned (animation is a separate concern).
   796→//   4. Free libwebp's allocated buffer via WebPFree after copying.
   797→//
   798→// We mirror that flow exactly. The runtime's PNGDecoder returns RGBA in
   799→// the same DecodedImage shape — so the renderer's draw_image() works with
   800→// either decoder without specialization.
   801→// ============================================================================
   802→namespace miniandroid {
   803→namespace renderer {
   804→
   805→DecodedImage WebPDecoder::decode(const std::vector<uint8_t>& webp_bytes) {
   806→    DecodedImage result;
   807→    if (webp_bytes.size() < 12) {
   808→        result.error = "WebP too small (need at least 12 bytes for RIFF header)";
   809→        return result;
   810→    }
   811→    // Per WebP Container Spec §2: signature is 'RIFF' + 4-byte size + 'WEBP'
   812→    if (webp_bytes[0] != 'R' || webp_bytes[1] != 'I' ||
   813→        webp_bytes[2] != 'F' || webp_bytes[3] != 'F' ||
   814→        webp_bytes[8] != 'W' || webp_bytes[9] != 'E' ||
   815→        webp_bytes[10] != 'B' || webp_bytes[11] != 'P') {
   816→        result.error = "bad WebP signature (expected RIFF...WEBP)";
   817→        return result;
   818→    }
   819→
   820→    int w = 0, h = 0;
   821→    // WebPGetInfo validates the bitstream and returns width/height.
   822→    if (!WebPGetInfo(webp_bytes.data(), webp_bytes.size(), &w, &h)) {
   823→        result.error = "WebPGetInfo failed (corrupt or unsupported WebP)";
   824→        return result;
   825→    }
   826→    if (w <= 0 || h <= 0 || w > 65536 || h > 65536) {
   827→        result.error = "WebP invalid dimensions: " + std::to_string(w) + "x" + std::to_string(h);
   828→        return result;
   829→    }
   830→
   831→    // Decode to RGBA. The returned pointer is libwebp-owned; we copy.
   832→    uint8_t* rgba = WebPDecodeRGBA(webp_bytes.data(), webp_bytes.size(), &w, &h);
   833→    if (rgba == nullptr) {
   834→        result.error = "WebPDecodeRGBA failed (decoder rejected bitstream)";
   835→        return result;
   836→    }
   837→
   838→    // Verify width/height didn't change during decode (shouldn't happen).
   839→    size_t pixel_count = static_cast<size_t>(w) * h * 4;
   840→    result.rgba.assign(rgba, rgba + pixel_count);
   841→    WebPFree(rgba);
   842→
   843→    result.ok = true;
   844→    result.width = w;
   845→    result.height = h;
   846→    // Determine format from VP8 chunk type for diagnostics.
   847→    char fmt[5] = {0, 0, 0, 0, 0};
   848→    if (webp_bytes.size() >= 16) {
   849→        std::memcpy(fmt, webp_bytes.data() + 12, 4);
   850→    }
   851→    result.color_type_name = std::string("webp-") + fmt;
   852→    return result;
   853→}

   890→// ============================================================================
   891→// EXP-097 §6: JPEGDecoder — libjpeg-backed JPEG → RGBA decoder.
   892→//
   893→// Per AOSP `BitmapFactory.nativeDecodeByteArray`: libjpeg's standard
   894→// jpeg_decompress_struct flow:
   895→//   1. jpeg_create_decompress
   896→//   2. jpeg_mem_src (read from memory)
   897→//   3. jpeg_read_header
   898→//   4. Set output color space to JCS_EXT_RGBA (libjpeg-turbo) — but base
   899→//      libjpeg only supports JCS_RGB. We request RGB and expand to RGBA
   900→//      ourselves so both libjpeg and libjpeg-turbo work.
   901→//   5. jpeg_start_decompress
   902→//   6. Read scanlines into a row buffer
   903→//   7. jpeg_finish_decompress + jpeg_destroy_decompress
   904→//
   905→// libjpeg uses setjmp/longjmp for fatal error recovery; we install a
   906→// jmp_buf and a custom error_exit handler so a corrupt JPEG returns
   907→// ok=false instead of aborting the whole process.
   908→// ============================================================================
   909→namespace miniandroid {
   910→namespace renderer {
   911→
   912→namespace {
   913→struct JpegErrorCtx {
   914→    jpeg_error_mgr pub;  // libjpeg's standard error manager
   915→    jmp_buf setjmp_buf;  // fatal-error escape hatch
   916→    std::string error_msg;
   917→};
   918→
   919→void jpeg_error_exit(j_common_ptr cinfo) {
   920→    auto* ctx = reinterpret_cast<JpegErrorCtx*>(cinfo->err);
   921→    // Format the libjpeg error message for diagnostics.
   922→    char buf[JMSG_LENGTH_MAX];
   923→    (*ctx->pub.format_message)(cinfo, buf);
   924→    ctx->error_msg = std::string("libjpeg error: ") + buf;
   925→    longjmp(ctx->setjmp_buf, 1);
   926→}
   927→}  // namespace
   928→
   929→DecodedImage JPEGDecoder::decode(const std::vector<uint8_t>& jpeg_bytes) {
   930→    DecodedImage result;
   931→    if (jpeg_bytes.size() < 2 ||
   932→        jpeg_bytes[0] != 0xFF || jpeg_bytes[1] != 0xD8) {
   933→        result.error = "bad JPEG signature (expected FF D8)";
   934→        return result;
   935→    }
   936→
   937→    struct jpeg_decompress_struct cinfo;
   938→    JpegErrorCtx jerr;
   939→    cinfo.err = jpeg_std_error(&jerr.pub);
   940→    jerr.pub.error_exit = jpeg_error_exit;
   941→
   942→    if (setjmp(jerr.setjmp_buf)) {
   943→        // Fatal libjpeg error: cleanup and return.
   944→        jpeg_destroy_decompress(&cinfo);
   945→        result.error = jerr.error_msg;
   946→        return result;
   947→    }
   948→
   949→    jpeg_create_decompress(&cinfo);
   950→    jpeg_mem_src(&cinfo, jpeg_bytes.data(), jpeg_bytes.size());
   951→
   952→    if (jpeg_read_header(&cinfo, TRUE) != JPEG_HEADER_OK) {
   953→        jpeg_destroy_decompress(&cinfo);
   954→        result.error = "jpeg_read_header failed (not a valid JPEG)";
   955→        return result;
   956→    }
   957→
   958→    // Force RGB output (works with both libjpeg and libjpeg-turbo).
   959→    cinfo.out_color_space = JCS_RGB;
   960→
   961→    if (!jpeg_start_decompress(&cinfo)) {
   962→        jpeg_destroy_decompress(&cinfo);
   963→        result.error = "jpeg_start_decompress failed";
   964→        return result;
   965→    }
   966→
   967→    int w = static_cast<int>(cinfo.output_width);
   968→    int h = static_cast<int>(cinfo.output_height);
   969→    if (w <= 0 || h <= 0) {
   970→        jpeg_destroy_decompress(&cinfo);
   971→        result.error = "JPEG invalid dimensions";
   972→        return result;
   973→    }
   974→
   975→    // Read scanlines into a temporary RGB buffer, then expand to RGBA.
   976→    result.rgba.assign(static_cast<size_t>(w) * h * 4, 0xFF);  // default alpha=255
   977→    std::vector<uint8_t> row_buf(static_cast<size_t>(w) * 3);
   978→    while (cinfo.output_scanline < static_cast<JDIMENSION>(h)) {
   979→        JSAMPROW row = row_buf.data();
   980→        JSAMPARRAY rows = &row;
   981→        jpeg_read_scanlines(&cinfo, rows, 1);
   982→        size_t y = cinfo.output_scanline - 1;
   983→        for (int x = 0; x < w; x++) {
   984→            result.rgba[(y * w + x) * 4 + 0] = row_buf[x * 3 + 0];
   985→            result.rgba[(y * w + x) * 4 + 1] = row_buf[x * 3 + 1];
   986→            result.rgba[(y * w + x) * 4 + 2] = row_buf[x * 3 + 2];
   987→            // alpha left at 0xFF (set above)
   988→        }
   989→    }
   990→
   991→    jpeg_finish_decompress(&cinfo);
   992→    jpeg_destroy_decompress(&cinfo);
   993→
   994→    result.ok = true;
   995→    result.width = w;
   996→    result.height = h;
   997→    result.color_type_name = (cinfo.num_components == 1) ? "jpeg-gray→rgba" : "jpeg-rgb→rgba";
   998→    return result;
   999→}

  1014→// EXP-097 §7: RLottieDecoder — rlottie-backed Lottie JSON → RGBA renderer.

  1016→extern "C" {
  1017→#include <rlottie_capi.h>
  1018→}
  1019→
  1020→RLottieDecoder::DecodedAnim RLottieDecoder::decode(const std::string& json_str,
  1021→                                                    int target_w, int target_h,
  1022→                                                    int max_frames) {
  1023→    DecodedAnim result;
  1024→    if (json_str.empty()) {
  1025→        result.error = "empty Lottie JSON";
  1026→        return result;
  1027→    }
  1028→    if (target_w <= 0 || target_h <= 0) {
  1029→        result.error = "invalid render dimensions: " +
  1030→                       std::to_string(target_w) + "x" + std::to_string(target_h);
  1031→        return result;
  1032→    }
  1033→
  1034→    // rlottie requires the JSON be null-terminated. We append a NUL — the
  1035→    // input string_view's data() may already be NUL-terminated if it came
  1036→    // from a std::string with capacity, but we ensure it explicitly.
  1037→    std::string json_nul = json_str;
  1038→    if (json_nul.back() != '\0') json_nul.push_back('\0');
  1039→
  1040→    // key is an opaque cache identifier — rlottie uses it for caching
  1041→    // parsed animations. We pass a synthetic key derived from the JSON size.
  1042→    std::string key = "exp097_" + std::to_string(json_str.size());
  1043→
  1044→    Lottie_Animation* anim = lottie_animation_from_data(json_nul.data(),
  1045→                                                        key.c_str(), "");
  1046→    if (anim == nullptr) {
  1047→        result.error = "lottie_animation_from_data failed (invalid JSON or unsupported)";
  1048→        return result;
  1049→    }
  1050→
  1051→    // Query the animation's intrinsic size and frame count.
  1052→    size_t anim_w = 0, anim_h = 0;
  1053→    lottie_animation_get_size(anim, &anim_w, &anim_h);
  1054→    size_t total = lottie_animation_get_totalframe(anim);
  1055→    double fps = lottie_animation_get_framerate(anim);
  1056→    result.width = target_w;
  1057→    result.height = target_h;
  1058→    result.total_frames = static_cast<int>(total);
  1059→    result.frame_rate = fps;
  1060→
  1061→    int frames_to_render = (max_frames < 0)
  1062→                              ? static_cast<int>(total)
  1063→                              : std::min(static_cast<int>(total), max_frames);
  1064→    if (frames_to_render <= 0) frames_to_render = 1;
  1065→
  1066→    size_t pixels_per_frame = static_cast<size_t>(target_w) * target_h;
  1067→    size_t bytes_per_frame = pixels_per_frame * 4;
  1068→    result.frames_rgba.resize(pixels_per_frame * frames_to_render);
  1069→
  1070→    bool any_nonzero = false;
  1071→    for (int i = 0; i < frames_to_render; i++) {
  1072→        uint32_t* buf = reinterpret_cast<uint32_t*>(
  1073→            result.frames_rgba.data() + i * pixels_per_frame);
  1074→        // Zero the buffer first (rlottie writes pixels with alpha — transparent
  1075→        // areas should be 0 not garbage).
  1076→        std::memset(buf, 0, bytes_per_frame);
  1077→        lottie_animation_render(anim, static_cast<size_t>(i), buf,
  1078→                                static_cast<size_t>(target_w),
  1079→                                static_cast<size_t>(target_h),
  1080→                                static_cast<size_t>(target_w) * 4);
  1081→        // Sample a few pixels to detect non-empty render
  1082→        for (size_t p = 0; p < std::min<size_t>(pixels_per_frame, 1024); p++) {
  1083→            if (buf[p] != 0) { any_nonzero = true; break; }
  1084→        }
  1085→    }
  1086→
  1087→    lottie_animation_destroy(anim);
  1088→
  1089→    if (!any_nonzero) {
  1090→        result.error = "rlottie rendered 0 non-zero pixels across " +
  1091→                       std::to_string(frames_to_render) + " frames";
  1092→        // Still return ok=true — the animation may legitimately render to
  1093→        // transparent pixels in its first frames (e.g. fade-in animations).
  1094→        // Callers can inspect frames_rgba for verification.
  1095→    }
  1096→    result.ok = true;
  1097→    return result;
  1098→}
