    35→std::string get_timestamp() {
    36→    auto now = std::time(nullptr);
    37→    char buf[64];
    38→    std::strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%SZ", std::gmtime(&now));
    39→    return std::string(buf);
    40→}

    46→bool write_json_file(const std::string& path, const json& data) {
    47→    try {
    48→        std::ofstream file(path);
    49→        if (!file.is_open()) {
    50→            std::cerr << "[ERROR] Cannot write to: " << path << std::endl;
    51→            return false;
    52→        }
    53→        file << std::setw(4) << data << "\n";
    54→        file.close();
    55→        return true;
    56→    } catch (const std::exception& e) {
    57→        std::cerr << "[ERROR] JSON write failed: " << e.what() << std::endl;
    58→        return false;
    59→    }

    66→struct GoldenComparisonResult {
    67→    bool instruction_trace_match = false;
    68→    bool api_trace_match = false;
    69→    bool object_creation_match = false;
    70→    bool overall_pass = false;

    75→GoldenComparisonResult compare_with_golden(const miniandroid::dex::BatchExecutionTrace& trace,
    76→                                            const std::string& golden_dir) {
    77→    GoldenComparisonResult result;
    78→    result.differences = json::array();
    79→    
    80→    // Load expected execution trace
    81→    std::ifstream expected_exec(golden_dir + "/expected_execution.json");
    82→    if (expected_exec.is_open()) {
    83→        json expected;
    84→        expected_exec >> expected;
    85→        
    86→        // Compare instruction count
    87→        size_t expected_count = expected.value("executed_instructions", 0);
    88→        if (trace.executed_instructions == expected_count) {
    89→            result.instruction_trace_match = true;
    90→        } else {
    91→            result.differences.push_back({
    92→                {"field", "instruction_count"},
    93→                {"expected", expected_count},
    94→                {"actual", trace.executed_instructions}
    95→            });
    96→        }
    97→        
    98→        // Compare final status
    99→        json status_obj = expected.value("status", json::object());
   100→        std::string expected_status = status_obj.value("result", std::string(""));
   101→        std::string actual_status = [trace]() -> std::string {
   102→            switch (trace.status) {
   103→                case miniandroid::dex::BatchExecutionTrace::BatchStatus::PASS: return "PASS";
   104→                case miniandroid::dex::BatchExecutionTrace::BatchStatus::PARTIAL: return "PARTIAL";
   105→                case miniandroid::dex::BatchExecutionTrace::BatchStatus::FAIL: return "FAIL";
   106→            }
   107→            return "UNKNOWN";
   108→        }();
   109→        
   110→        if (expected_status.empty() || actual_status == expected_status) {
   111→            // OK or no comparison available
   112→        } else {
   113→            result.differences.push_back({
   114→                {"field", "status"},
   115→                {"expected", expected_status},
   116→                {"actual", actual_status}
   117→            });
   118→        }
   119→    } else {
   120→        // No golden file - can't compare
   121→        result.instruction_trace_match = true;  // Assume pass if no golden to compare
   122→    }
   123→    
   124→    // Load expected API trace
   125→    std::ifstream expected_api(golden_dir + "/expected_api_trace.json");
   126→    if (expected_api.is_open()) {
   127→        json expected;
   128→        expected_api >> expected;
   129→        
   130→        size_t expected_api_count = expected.size();
   131→        if (trace.api_calls.size() == expected_api_count) {
   132→            result.api_trace_match = true;
   133→            
   134→            // Compare API call details
   135→            for (size_t i = 0; i < std::min(trace.api_calls.size(), expected.size()); ++i) {
   136→                std::string expected_api_str = expected[i].value("api", "");
   137→                std::string actual_api_str = trace.api_calls[i].api_class + "." + 
   138→                                                trace.api_calls[i].method;
   139→                
   140→                if (expected_api_str != actual_api_str && !expected_api_str.empty()) {
   141→                    result.differences.push_back({
   142→                        {"field", "api_call[" + std::to_string(i) + "]"},
   143→                        {"expected", expected_api_str},
   144→                        {"actual", actual_api_str}
   145→                    });
   146→                    result.api_trace_match = false;
   147→                }
   148→            }
   149→        } else {
   150→            result.differences.push_back({
   151→                {"field", "api_call_count"},
   152→                {"expected", expected_api_count},
   153→                {"actual", trace.api_calls.size()}
   154→            });
   155→        }
   156→    } else {
   157→        result.api_trace_match = true;
   158→    }
   159→    
   160→    // Object creation comparison
   161→    std::ifstream expected_obj(golden_dir + "/expected_objects.json");
   162→    if (expected_obj.is_open()) {
   163→        json expected;
   164→        expected_obj >> expected;
   165→        
   166→        size_t expected_obj_count = expected.size();
   167→        if (trace.object_creations.size() == expected_obj_count) {
   168→            result.object_creation_match = true;
   169→        } else {
   170→            result.differences.push_back({
   171→                {"field", "object_creation_count"},
   172→                {"expected", expected_obj_count},
   173→                {"actual", trace.object_creations.size()}
   174→            });
   175→        }
   176→    } else {
   177→        result.object_creation_match = true;
   178→    }
   179→    
   180→    // Overall pass if all comparisons passed or were inconclusive
   181→    result.overall_pass = result.instruction_trace_match && 
   182→                         result.api_trace_match && 
   183→                         result.object_creation_match;
   184→    
   185→    return result;
   186→}

   191→
   192→void generate_report(const miniandroid::dex::BatchExecutionTrace& trace,
   193→                     const GoldenComparisonResult& comparison,
   194→                     const std::string& output_path) {
   195→    std::ofstream report(output_path);
   196→    
   197→    report << "# MiniAndroid Execution Report - EXP-003-BATCH\n\n";
   198→    report << "**Experiment:** Minimal DEX Execution Engine\n\n";
   199→    report << "---\n\n";
   200→    
   201→    // Status section
   202→    std::string status_str;
   203→    switch (trace.status) {
   204→        case miniandroid::dex::BatchExecutionTrace::BatchStatus::PASS: 
   205→            status_str = "**✅ PASS**"; break;
   206→        case miniandroid::dex::BatchExecutionTrace::BatchStatus::PARTIAL: 
   207→            status_str = "**⚠️ PARTIAL**"; break;
   208→        case miniandroid::dex::BatchExecutionTrace::BatchStatus::FAIL: 
   209→            status_str = "**❌ FAIL**"; break;
   210→    }
   211→    
   212→    report << "## Execution Status: " << status_str << "\n\n";
   213→    
   214→    // Method context
   215→    report << "## Method Context\n\n";
   216→    report << "| Property | Value |\n";
   217→    report << "|----------|-------|\n";
   218→    report << "| Class | `" << trace.class_name << "` |\n";
   219→    report << "| Method | `" << trace.method_name << "` |\n";
   220→    report << "| Descriptor | `" << trace.method_descriptor << "` |\n";
   221→    report << "| Completed | " << (trace.completed_successfully ? "Yes ✅" : "No ❌") << " |\n\n";
   222→    
   223→    // Execution statistics
   224→    report << "## Execution Statistics\n\n";
   225→    report << "| Metric | Value |\n";
   226→    report << "|--------|-------|\n";
   227→    report << "| Total Instructions in Method | " << trace.total_instructions_in_method << " |\n";
   228→    report << "| Instructions Executed | " << trace.executed_instructions << " |\n";
   229→    report << "| Initial PC | `0x" << std::hex << trace.initial_pc << "` |\n";
   230→    report << "| Final PC | `0x" << trace.final_pc << "` |\n";
   231→    report << std::dec;  // Reset to decimal
   232→    report << "| Objects Created | " << trace.object_heap.size() << " |\n";
   233→    report << "| API Calls Made | " << trace.api_calls.size() << " |\n\n";
   234→    
   235→    // Instruction trace summary
   236→    report << "## Executed Instructions\n\n";
   237→    report << "| Seq | Opcode | PC | Status |\n";
   238→    report << "|-----|--------|-----|--------|\n";
   239→    for (const auto& instr : trace.instructions) {
   240→        std::string status_icon;
   241→        switch (instr.status) {
   242→            case miniandroid::dex::InstructionTraceEntry::Status::SUCCESS: 
   243→                status_icon = "✅"; break;
   244→            case miniandroid::dex::InstructionTraceEntry::Status::HALT_RETURN: 
   245→                status_icon = "↩️"; break;
   246→            case miniandroid::dex::InstructionTraceEntry::Status::HALT_UNIMPLEMENTED: 
   247→                status_icon = "⛔"; break;
   248→            default: status_icon = "❓"; break;
   249→        }
   250→        report << "| " << instr.sequence << " | `" << instr.opcode_name 
   251→               << "` | `0x" << std::hex << instr.pc_before << "` | " 
   252→               << status_icon << " |\n";
   253→    }
   254→    report << std::dec;
   255→    report << "\n";
   256→    
   257→    // Object heap
   258→    if (!trace.object_heap.get_all_ids().empty()) {
   259→        report << "## Object Heap\n\n";
   260→        report << "| ID | Class | Initialized |\n";
   261→        report << "|----|-------|-------------|\n";
   262→        for (uint32_t obj_id : trace.object_heap.get_all_ids()) {
   263→            const auto* obj = trace.object_heap.get(obj_id);
   264→            if (obj) {
   265→                report << "| " << obj->object_id << " | `" << obj->class_descriptor 
   266→                       << "` | " << (obj->is_initialized ? "Yes" : "No") << " |\n";
   267→            }
   268→        }
   269→        report << "\n";
   270→    }
   271→    
   272→    // API calls
   273→    if (!trace.api_calls.empty()) {
   274→        report << "## API Calls\n\n";
   275→        report << "| # | API | Arguments | Status |\n";
   276→        report << "|---|-----|-----------|--------|\n";
   277→        for (const auto& call : trace.api_calls) {
   278→            std::string args_str;
   279→            for (const auto& arg : call.arguments) {
   280→                args_str += arg + " ";
   281→            }
   282→            std::string icon = (call.status == "IMPLEMENTED") ? "✅" : "📋";
   283→            report << "| " << call.sequence << " | `" << call.api_class << "." << call.method 
   284→                   << "` | `" << args_str << "` | " << icon << " |\n";
   285→        }
   286→        report << "\n";
   287→    }
   288→    
   289→    // Failures
   290→    if (!trace.failures.empty()) {
   291→        report << "## Issues / Failures\n\n";
   292→        report << "| Type | Details | Severity |\n";
   293→        report << "|------|---------|----------|\n";
   294→        for (const auto& fail : trace.failures) {
   295→            report << "| " << fail.type << " | " << fail.details << " | " << fail.severity << " |\n";
   296→        }
   297→        report << "\n";
   298→    }
   299→    
   300→    // Golden comparison
   301→    report << "## Golden Test Comparison\n\n";
   302→    report << "| Check | Result |\n";
   303→    report << "|-------|--------|\n";
   304→    report << "| Instruction Trace | " << (comparison.instruction_trace_match ? "[MATCH]" : "[MISMATCH]") << " |\n";
   305→    report << "| API Trace | " << (comparison.api_trace_match ? "[MATCH]" : "[MISMATCH]") << " |\n";
   306→    report << "| Object Creation | " << (comparison.object_creation_match ? "[MATCH]" : "[MISMATCH]") << " |\n";
   307→    report << "| **Overall** | **" << (comparison.overall_pass ? "[PASS]" : "[FAIL]") << "**" << " |\n";
   308→    report << "\n";
   309→    
   310→    if (!comparison.differences.empty() && !comparison.overall_pass) {
   311→        report << "### Differences\n\n";
   312→        for (const auto& diff : comparison.differences) {
   313→            std::string field = diff.value("field", std::string(""));
   314→            std::string expected_val = diff.value("expected", std::string(""));
   315→            std::string actual_val = diff.value("actual", std::string(""));
   316→            report << "- **" << field << "**: "
   317→                   << "expected `" << expected_val 
   318→                   << "`, got `" << actual_val << "`\n";
   319→        }
   320→        report << "\n";
   321→    }
   322→    
   323→    // Execution chain verification
   324→    report << "## Execution Chain Verification\n\n";
   325→    report << "```\n";
   326→    report << "MainActivity.onCreate()\n";
   327→    report << "    ↓\n";
   328→    report << "const-string → new-instance TextView → invoke-direct <init>\n";
   329→    report << "    ↓\n";
   330→    report << "invoke-virtual setText(\"Hello MiniAndroid\")\n";
   331→    report << "    ↓\n";
   332→    report << "return-void\n";
   333→    report << "```\n\n";
   334→    
   335→    bool chain_complete = trace.completed_successfully && 
   336→                        (trace.executed_instructions >= 5);  // Minimum for full chain
   337→    
   338→    report << "**Chain Status:** " << (chain_complete ? "✅ **COMPLETE**" : "⚠️ **INCOMPLETE**") << "\n\n";
   339→    
   340→    // Footer
   341→    report << "---\n";
   342→    report << "*Generated by MiniAndroid v0.1 - EXP-003-BATCH*\n";
   343→    report << "*Timestamp: " << get_timestamp() << "*\n";
   344→    
   345→    report.close();
   346→}
