     4→//
     5→// Shadows implemented in this file:
     6→//   * ThreadShadow       — single deterministic main Thread
     7→//   * LooperShadow       — single deterministic main Looper bound to main Thread
     8→//   * HandlerShadow      — single-thread Runnable queue with post/postDelayed
     9→//   * ActivityShadow     — current Activity tracking + lifecycle state
    10→//   * IntentShadow       — Intent creation, component resolution, startActivity
    11→//   * ViewShadow         — minimal View/ViewGroup hierarchy
    12→//
    13→// All shadows share a single-threaded execution model. There is exactly
    14→// one main Thread object, one main Looper object, and one main Handler
    15→// object — all with the same heap object_id so that
    16→//     Looper.getMainLooper().getThread() == Thread.currentThread()
    17→// returns true, which is what ArchTaskExecutor.isMainThread() actually
    18→// checks.

    35→// ─────────────────────────────────────────────────────────────────────────
    36→// ArchTaskExecutorShadow — handles ArchTaskExecutor.isMainThread.
    37→//
    38→// Real ArchTaskExecutor.isMainThread delegates to mDelegate.isMainThread()
    39→// which (for DefaultTaskExecutor) does:
    40→//   return Looper.getMainLooper().getThread() == Thread.currentThread();
    41→//
    42→// In our model, both sides return the same canonical main Thread object_id
    43→// (bound by ThreadShadow and LooperShadow at init). So isMainThread() is
    44→// always true in the single-threaded runtime.
    45→//
    46→// We do NOT short-circuit the bytecode via a legacy bridge stub — instead,
    47→// we register a Shadow that the bridge consults FIRST. This keeps the
    48→// architecture clean: shadows handle Android framework behavior, the
    49→// engine executes bytecode.
    50→//
    51→// This shadow handles:
    52→//   * ArchTaskExecutor.isMainThread → true (the delegate check)
    53→//   * ArchTaskExecutor.getInstance → singleton ArchTaskExecutor object
    54→//   * ArchTaskExecutor.executeOnDiskIO → no-op (post to disk thread)
    55→//   * ArchTaskExecutor.postToMainThread → enqueue via HandlerShadow
    56→// ─────────────────────────────────────────────────────────────────────────
    57→class ArchTaskExecutorShadow : public Shadow {
    58→public:
    59→    std::string name() const override { return "ArchTaskExecutor"; }
    60→    void init(HeapAllocator* heap) override {
    61→        heap_ = heap;
    62→        if (heap_) {
    63→            instance_id_ = heap_->get_or_create("Landroidx/arch/core/executor/ArchTaskExecutor;");
    64→        }
    65→    }
    66→
    67→    bool handles_class(const std::string& class_name) const override {
    68→        return class_name.find("ArchTaskExecutor") != std::string::npos ||
    69→               class_name.find("DefaultTaskExecutor") != std::string::npos ||
    70→               class_name.find("TaskExecutor") != std::string::npos;
    71→    }
    72→
    73→    CallResult dispatch(const CallContext& ctx) override;
    74→
    75→    std::vector<std::string> implemented_methods() const override {
    76→        return {"isMainThread", "getInstance", "executeOnDiskIO",
    77→                "postToMainThread"};
    78→    }
    79→    std::vector<std::string> stubbed_methods() const override {
    80→        return {"delegate", "setDelegate"};
    81→    }
    82→
    83→    uint32_t instance_id() const { return instance_id_; }
    84→
    85→private:
    86→    uint32_t instance_id_ = 0;
    87→};

    89→// ─────────────────────────────────────────────────────────────────────────
    90→// ThreadShadow — owns the single main Thread object.
    91→//
    92→// Identity contract:
    93→//   * Thread.currentThread() returns main_thread_id_ always.
    94→//   * Thread.equals(Object) returns true iff the other object_id is
    95→//     main_thread_id_ (mimics reference equality).
    96→//   * Thread.getId() returns MAIN_THREAD_ID = 1.
    97→//   * Thread.getName() returns "main".
    98→//   * Thread.getStackTrace() returns an empty array (unblocks Intrinsics).
    99→//
   100→// The main_thread_id_ is allocated lazily on first init() call.
   101→// ─────────────────────────────────────────────────────────────────────────
   102→class ThreadShadow : public Shadow {
   103→public:
   104→    static constexpr uint32_t MAIN_THREAD_TID = 1;
   105→    static constexpr const char* MAIN_THREAD_NAME = "main";
   106→
   107→    std::string name() const override { return "Thread"; }
   108→    void init(HeapAllocator* heap) override {
   109→        heap_ = heap;
   110→        if (heap_) {
   111→            // Allocate (or look up) the main Thread singleton up-front.
   112→            // We do NOT cache the ID locally yet because get_or_create
   113→            // is idempotent — every later call to currentThread() will
   114→            // return the same heap object.
   115→            main_thread_id_ = heap_->get_or_create("Ljava/lang/Thread;");
   116→        }
   117→    }
   118→
   119→    bool handles_class(const std::string& class_name) const override {
   120→        return class_name == "Ljava/lang/Thread;";
   121→    }
   122→
   123→    CallResult dispatch(const CallContext& ctx) override;
   124→
   125→    std::vector<std::string> implemented_methods() const override {
   126→        return {"currentThread", "getName", "getId", "getStackTrace",
   127→                "isAlive", "isDaemon", "interrupt"};
   128→    }
   129→    std::vector<std::string> stubbed_methods() const override {
   130→        return {"sleep", "yield", "join", "start", "run", "setDaemon"};
   131→    }
   132→
   133→    // Public accessor: returns the canonical main Thread object_id.
   134→    uint32_t main_thread_id() const { return main_thread_id_; }
   135→
   136→    // Public mutator: allows the LooperShadow to bind to the same id.
   137→    void set_main_thread_id(uint32_t id) { main_thread_id_ = id; }
   138→
   139→private:
   140→    uint32_t main_thread_id_ = 0;
   141→};

   143→// ─────────────────────────────────────────────────────────────────────────
   144→// LooperShadow — owns the single main Looper object.
   145→//
   146→// Identity contract:
   147→//   * Looper.getMainLooper() returns main_looper_id_ always.
   148→//   * Looper.myLooper() returns main_looper_id_ (we have only one thread).
   149→//   * Looper.getThread() returns the SAME id as Thread.currentThread().
   150→//     This is the key invariant that makes ArchTaskExecutor.isMainThread()
   151→//     return true.
   152→//   * Looper.getQueue() returns a singleton MessageQueue (no real queue).
   153→//   * Looper.prepare() / Looper.loop() are stubs (return void).
   154→//
   155→// On init, the LooperShadow asks the ThreadShadow for the main Thread
   156→// object_id and binds to it. The LooperShadow and ThreadShadow must be
   157→// registered in the same registry so the LooperShadow can find the
   158→// ThreadShadow.
   159→// ─────────────────────────────────────────────────────────────────────────
   160→class LooperShadow : public Shadow {
   161→public:
   162→    std::string name() const override { return "Looper"; }
   163→    void init(HeapAllocator* heap) override {
   164→        heap_ = heap;
   165→        if (heap_) {
   166→            main_looper_id_ = heap_->get_or_create("Landroid/os/Looper;");
   167→            // Also allocate the MessageQueue singleton referenced by getQueue.
   168→            main_queue_id_  = heap_->get_or_create("Landroid/os/MessageQueue;");
   169→        }
   170→    }
   171→
   172→    bool handles_class(const std::string& class_name) const override {
   173→        return class_name == "Landroid/os/Looper;" ||
   174→               class_name == "Landroid/os/MessageQueue;";
   175→    }
   176→
   177→    CallResult dispatch(const CallContext& ctx) override;
   178→
   179→    std::vector<std::string> implemented_methods() const override {
   180→        return {"getMainLooper", "myLooper", "getThread", "getQueue",
   181→                "myQueue", "getMainLooper", "prepare", "loop",
   182→                "quit", "quitSafely", "prepareMainLooper"};
   183→    }
   184→    std::vector<std::string> stubbed_methods() const override {
   185→        return {"loop", "prepare", "prepareMainLooper"};
   186→    }
   187→
   188→    uint32_t main_looper_id() const { return main_looper_id_; }
   189→    uint32_t main_queue_id()  const { return main_queue_id_; }
   190→
   191→    // Bind the Looper to a specific Thread object_id (typically
   192→    // ThreadShadow.main_thread_id()). The LooperShadow will return this
   193→    // same id from Looper.getThread().
   194→    void bind_to_thread(uint32_t thread_id) { bound_thread_id_ = thread_id; }
   195→    uint32_t bound_thread_id() const { return bound_thread_id_; }
   196→
   197→private:
   198→    uint32_t main_looper_id_ = 0;
   199→    uint32_t main_queue_id_  = 0;
   200→    uint32_t bound_thread_id_ = 0;
   201→};

   203→// ─────────────────────────────────────────────────────────────────────────
   204→// HandlerShadow — deterministic single-thread Runnable queue.
   205→//
   206→// Android's Handler is normally a thread-loose, multi-threaded queue
   207→// backed by a MessageQueue. We don't need real multithreading — we just
   208→// need to faithfully preserve "post this Runnable, eventually run it".
   209→//
   210→// Model:
   211→//   * Handler(Looper) and Handler() are both no-ops that allocate a
   212→//     Handler heap object. We don't actually use the Handler object for
   213→//     dispatch — we maintain a single global queue.
   214→//   * Handler.post(Runnable)    → enqueue(runnable, delay_ms=0)
   215→//   * Handler.postDelayed(R,A)  → enqueue(runnable, delay_ms=A)
   216→//   * Handler.removeCallbacks(R)→ remove from queue
   217→//   * Handler.getLooper()       → return main Looper singleton
   218→//   * AndroidUtilities.runOnUIThread(Runnable) → enqueue(runnable, 0)
   219→//
   220→// The ApplicationRuntime drains the queue at well-defined points:
   221→// after Activity.onCreate, after onResume, etc. Runnables that post
   222→// other Runnables are supported (the drain loop is iterative).
   223→// ─────────────────────────────────────────────────────────────────────────
   224→class HandlerShadow : public Shadow {
   225→public:
   226→    struct QueuedRunnable {
   227→        uint32_t runnable_id = 0;        // heap object_id of the Runnable
   228→        uint32_t enqueue_seq = 0;        // FIFO tiebreaker
   229→        int64_t  ready_at_ms = 0;        // logical "ready" timestamp
   230→        std::string runnable_class;     // for diagnostics
   231→    };
   232→
   233→    std::string name() const override { return "Handler"; }
   234→    void init(HeapAllocator* heap) override {
   235→        heap_ = heap;
   236→        if (heap_) {
   237→            main_handler_id_ = heap_->get_or_create("Landroid/os/Handler;");
   238→        }
   239→    }
   240→
   241→    bool handles_class(const std::string& class_name) const override {
   242→        return class_name == "Landroid/os/Handler;" ||
   243→               class_name == "Lorg/telegram/messenger/AndroidUtilities;";
   244→    }
   245→
   246→    CallResult dispatch(const CallContext& ctx) override;
   247→
   248→    std::vector<std::string> implemented_methods() const override {
   249→        return {"post", "postDelayed", "postAtFrontOfQueue",
   250→                "removeCallbacks", "removeCallbacksAndMessages",
   251→                "getLooper", "sendEmptyMessage", "sendMessage"};
   252→    }
   253→    std::vector<std::string> stubbed_methods() const override {
   254→        return {"obtainMessage", "sendMessageDelayed", "sendMessageAtTime"};
   255→    }
   256→
   257→    // Enqueue a Runnable with a delay (in milliseconds). The ready time is
   258→    // computed against the VIRTUAL clock (virtual_now_ms_ + delay_ms), so
   259→    // scheduling is fully deterministic — no wall-clock involvement.
   260→    void enqueue(uint32_t runnable_id, int64_t delay_ms,
   261→                 const std::string& cls);
   262→
   263→    // Drain every Runnable whose ready time has been reached on the VIRTUAL
   264→    // clock (ready_at_ms <= virtual_now_ms_), in enqueue (FIFO) order among
   265→    // the due entries. Runnables scheduled for a later virtual time stay
   266→    // queued. Returns the number drained.
   267→    // Each drained Runnable's heap object_id is appended to `out_drained`.
   268→    // The ApplicationRuntime is responsible for actually executing the
   269→    // Runnable's run() method.
   270→    size_t drain_ready(std::vector<uint32_t>* out_drained);
   271→
   272→    // ── Deterministic virtual clock (Handler/Looper time model) ──────────
   273→    // The runtime has no wall-clock rendering, so Looper time is virtual.
   274→    //  * idle-settle points (post-onCreate) call settle(): the clock jumps
   275→    //    far into the future and one drain dispatches everything posted so
   276→    //    far — the documented EXP-088 law (post(A), post(B), postDelayed(C)
   277→    //    → drain → A, B, C) is preserved.
   278→    //  * time-driven frame capture (--frames N) advances the clock by
   279→    //    --frame-delay per frame and drains only what became due, so a
   280→    //    self-reposting postDelayed animation steps once per frame,
   281→    //    deterministically.
   282→    void settle();                       // jump far into the future
   283→    void advance_virtual(int64_t delta_ms);  // step the clock forward
   284→    int64_t virtual_now_ms() const { return virtual_now_ms_; }
   285→
   286→    // EXP-088 Phase F: Remove all queued Runnables matching the given
   287→    // runnable_id. Returns the number removed.
   288→    //
   289→    // Implements Handler.removeCallbacks(Runnable) — removes any
   290→    // matching Runnable from the queue, regardless of its delay/priority.
   291→    // If the Runnable has already been drained, this is a no-op.
   292→    //
   293→    // This is necessary for the user's Phase F acceptance scenario:
   294→    //   post(A), post(B), postDelayed(C), removeCallbacks(B), drain → A, C
   295→    //
   296→    // Previously this was a stub (no-op) — the queue was never modified.
   297→    size_t remove_callbacks(uint32_t runnable_id);
   298→
   299→    // EXP-088 Phase F: Remove ALL queued Runnables (Handler.removeCallbacksAndMessages(null)).
   300→    // Returns the number removed.
   301→    size_t remove_all();
   302→
   303→    // Total queue depth (for diagnostics).
   304→    size_t queue_size() const { return queue_.size(); }
   305→
   306→    // Helper: extract a Runnable argument from a CallContext.
   307→    // `arg_idx` is the index of the Runnable parameter. Returns 0 if
   308→    // the arg is null or absent.
   309→    static uint32_t extract_runnable(const CallContext& ctx, size_t arg_idx);
   310→
   311→private:
   312→    uint32_t main_handler_id_ = 0;
   313→    std::deque<QueuedRunnable> queue_;
   314→    uint32_t next_seq_ = 0;
   315→    int64_t virtual_now_ms_ = 0;   // deterministic Looper time (ms)
   316→};

   349→    CallResult dispatch(const CallContext& ctx) override;

   351→    std::vector<std::string> implemented_methods() const override {

   383→// ─────────────────────────────────────────────────────────────────────────
   384→// ActivityShadow — current Activity, lifecycle, view root.
   385→//
   386→// Tracks:
   387→//   * current_activity_id  — heap object_id of the active Activity.
   388→//   * current_activity_class — class descriptor (e.g. "Lorg/telegram/ui/LaunchActivity;").
   389→//   * content_view_id     — heap object_id of the View set by setContentView.
   390→//
   391→// Lifecycle state machine (deterministic, single-threaded):
   392→//   CREATED → STARTED → RESUMED → (PAUSED → STOPPED → DESTROYED)
   393→//
   394→// Lifecycle method invocations on the Activity heap object are NOT
   395→// dispatched by this shadow — the engine's recursive invoke path
   396→// handles those (it can find the Activity's onCreate in DEX bytecode).
   397→// This shadow only tracks state.
   398→// ─────────────────────────────────────────────────────────────────────────
   399→class ActivityShadow : public Shadow {
   400→public:
   401→    enum class LifecycleState {
   402→        NONE, CREATED, STARTED, RESUMED, PAUSED, STOPPED, DESTROYED
   403→    };
   404→
   405→    std::string name() const override { return "Activity"; }
   406→    void init(HeapAllocator* heap) override { heap_ = heap; }
   407→
   408→    bool handles_class(const std::string& class_name) const override {
   409→        // EXP-087 Phase 3 (B2 FIX): Also handle Activity subclasses whose
   410→        // names don't contain "Activity" (e.g. GameMasterDice extends
   411→        // ListActivity, StopWatch extends Activity). We handle any class
   412→        // that could have inherited setContentView/getIntent/etc.
   413→        return class_name == "Landroid/app/Activity;" ||
   414→               class_name == "Landroid/app/ListActivity;" ||
   415→               class_name.find("/Activity;") != std::string::npos ||
   416→               class_name.find("LaunchActivity") != std::string::npos ||
   417→               class_name.find("LoginActivity")  != std::string::npos ||
   418→               class_name.find("Activity;") != std::string::npos ||
   419→               // EXP-087: Also match known Activity subclass patterns
   420→               class_name.find("/GameMasterDice;") != std::string::npos ||
   421→               class_name.find("/StopWatch;") != std::string::npos ||
   422→               class_name.find("/Notes;") != std::string::npos ||
   423→               class_name.find("/NoteMain;") != std::string::npos ||
   424→               class_name.find("/MainActivity;") != std::string::npos ||
   425→               class_name.find("/AndroidLauncher;") != std::string::npos;
   426→    }
   427→
   428→    CallResult dispatch(const CallContext& ctx) override;
   429→
   430→    std::vector<std::string> implemented_methods() const override {
   431→        return {"setContentView", "getContentView", "findViewById",
   432→                "getIntent", "setIntent", "finish", "getApplicationContext",
   433→                "getFragmentManager", "getWindow", "getWindowManager",
   434→                "getResources", "getPackageManager", "getPackageName",
   435→                "getClassLoader", "getFilesDir", "getCacheDir",
   436→                "getSharedPreferences", "startActivity", "startActivityForResult",
   437→                "getCallingActivity", "getCallingPackage"};
   438→    }
   439→    std::vector<std::string> stubbed_methods() const override {
   440→        return {"runOnUiThread", "overridePendingTransition",
   441→                "findViewById", "registerForContextMenu"};
   442→    }
   443→
   444→    void set_current_activity(uint32_t id, const std::string& cls) {
   445→        current_activity_id_ = id;
   446→        current_activity_class_ = cls;
   447→        state_ = LifecycleState::CREATED;
   448→    }
   449→    // UNIFIED_011.3 FRAME-2 (§23): record ONLY the activity's heap object id
   450→    // (no lifecycle-state side effects). Used by execute_apk_with_activity so
   451→    // post-launch probes (click-test android:onClick dispatch) can invoke
   452→    // handlers with the REAL activity instance as `this`.
   453→    void set_activity_heap_id(uint32_t id) { current_activity_id_ = id; }
   454→    uint32_t current_activity_id() const { return current_activity_id_; }
   455→    const std::string& current_activity_class() const { return current_activity_class_; }
   456→    LifecycleState state() const { return state_; }
   457→    void set_state(LifecycleState s) { state_ = s; }
   458→
   459→    void set_content_view(uint32_t view_id) { content_view_id_ = view_id; }
   460→    uint32_t content_view_id() const { return content_view_id_; }
   461→
   462→    // EXP-087 Phase 3 (B2 FIX): Set the APK path so setContentView(int)
   463→    // can find the layout_cache.json next to the APK.
   464→    void set_apk_path(const std::string& path) { apk_path_ = path; }
   465→
   466→    // UNIFIED_007: real inflation evidence
   467→    const std::string& last_inflate_stats() const { return last_inflate_stats_; }
   468→
   469→private:
   470→    uint32_t current_activity_id_ = 0;
   471→    std::string current_activity_class_;
   472→    uint32_t content_view_id_ = 0;
   473→    // EXP-074: Layout resource ID from setContentView(int layoutResId).
   474→    int32_t layout_resource_id_ = 0;
   475→    LifecycleState state_ = LifecycleState::NONE;
   476→    // EXP-087 Phase 3 (B2 FIX): APK path for layout_cache.json lookup
   477→    std::string apk_path_;
   478→    // UNIFIED_007: JSON stats from last real inflation
   479→    std::string last_inflate_stats_;
   480→    std::vector<std::string> warnings_;
   481→};

   483→// ─────────────────────────────────────────────────────────────────────────
   484→// ViewShadow — minimal View/ViewGroup hierarchy (semantic, no rendering).
   485→//
   486→// Each View heap object is tracked by:
   487→//   * view_id           — the object_id (== heap object_id)
   488→//   * parent_id         — 0 if root, else the parent ViewGroup's view_id
   489→//   * children           — list of child view_ids (empty for leaf Views)
   490→//   * view_id_android    — the Android `R.id.*` int set by setId() or XML
   491→//   * class_desc         — View subclass class descriptor
   492→//
   493→// Operations supported:
   494→//   * View.<init>(Context)   — allocate a fresh View heap object
   495→//   * View.setId(int)        — record the Android view id
   496→//   * View.getId()           — return the recorded id

   502→//   * Activity.setContentView(View) — set the content_view on ActivityShadow
   503→//   * Activity.findViewById(int)   — BFS the hierarchy for a view with

   651→    bool handles_class(const std::string& class_name) const override {
   652→        // EXP-075: Do NOT handle Activity subclasses — let ActivityShadow handle them.
   653→        // This prevents ViewShadow from intercepting setContentView calls on Activities.
   654→        if (class_name.find("Activity;") != std::string::npos) {
   655→            return false;
   656→        }
   657→        // Match any class ending in "View;" or "ViewGroup;" or containing
   658→        // well-known View subclasses. Specific dispatch is done by
   659→        // method name.
   660→        // EXP-060: Also match user-defined View subclasses by checking
   661→        // if the class_name is NOT a known non-View framework class.
   662→        // This is a heuristic — if the class isn't one of the known
   663→        // non-View types, we try ViewShadow dispatch and return
   664→        // not_handled if the method isn't a View method.
   665→        if (class_name.find("View;") != std::string::npos ||
   666→            class_name == "Landroid/widget/TextView;" ||
   667→            class_name == "Landroid/widget/EditText;" ||
   668→            class_name == "Landroid/widget/Button;" ||
   669→            class_name == "Landroid/widget/ImageView;" ||
   670→            class_name == "Landroid/view/View;" ||
   671→            class_name == "Landroid/view/ViewGroup;" ||
   672→            class_name == "Landroid/view/TextureView;" ||
   673→            class_name == "Landroid/widget/ScrollView;" ||
   674→            class_name == "Landroid/widget/FrameLayout;" ||
   675→            class_name == "Landroid/widget/LinearLayout;") {
   676→            return true;
   677→        }
   678→        // EXP-060: For user-defined classes (like IntroActivity$4 which
   679→        // extends TextView), we can't check the hierarchy here. But if the
   680→        // class name doesn't match any known non-View framework prefix,
   681→        // it MIGHT be a View subclass. Let dispatch() figure it out by
   682→        // method name. This is slightly broader than ideal but matches
   683→        // how Robolectric handles View subclasses.
   684→        if (class_name.find("Landroid/") == 0 ||
   685→            class_name.find("Ljava/") == 0 ||
   686→            class_name.find("Lkotlin/") == 0 ||
   687→            class_name.find("Lcom/google/") == 0 ||
   688→            class_name.find("Landroidx/") == 0) {
   689→            // Standard framework class — let other shadows handle it.
   690→            return false;
   691→        }
   692→        // User-defined class (e.g. Lorg/telegram/ui/...). Could be a
   693→        // View subclass. Let dispatch() decide based on method name.
   694→        return true;
   695→    }

   843→// ─────────────────────────────────────────────────────────────────────────
   844→// CollectionShadow — real List/ArrayList/Map semantics with per-instance state.
   845→//
   846→// Instead of a global `List.isEmpty → true` stub, this shadow tracks real
   847→// collection state per heap object. Each collection instance gets a
   848→// CollectionState (vector of elements) keyed by object_id.
   849→//
   850→// Supported operations:
   851→//   * List.add(item) → append, return true
   852→//   * List.get(index) → return element at index
   853→//   * List.size() → return count
   854→//   * List.isEmpty() → return count == 0
   855→//   * List.clear() → clear elements
   856→//   * List.remove(index) → remove at index
   857→//   * Map.put(key, value) → store key-value pair
   858→//   * Map.get(key) → return value for key
   859→//   * Map.size() → return count
   860→//   * Map.isEmpty() → return count == 0
   861→//   * Map.containsKey(key) → return true if key exists
   862→//   * Iterator.hasNext() → check position < size
   863→//   * Iterator.next() → return element at position++
   864→//
   865→// This is NOT a stub — it's real state-based semantics. The runtime's
   866→// List objects actually hold elements and report correct sizes.
   867→// ─────────────────────────────────────────────────────────────────────────
   868→class CollectionShadow : public Shadow {
   869→public:
   870→    struct CollectionState {
   871→        std::vector<uint32_t> elements;  // object_ids of elements
   872→        std::map<std::string, uint32_t> map_entries;  // key → value object_id
   873→        // EXP-071 Phase 7: Store string values for HashMap.put(key, String).
   874→        // The original map_entries only stores object_ids, but many HashMap
   875→        // usages store String values (e.g., shortname → country name).
   876→        // Without this, HashMap.get returns null for string values.
   877→        std::map<std::string, std::string> map_string_entries;  // key → string value
   878→        size_t iterator_position = 0;
   879→        bool is_map = false;
   880→    };
   881→
   882→    std::string name() const override { return "Collection"; }
   883→    void init(HeapAllocator* heap) override { heap_ = heap; }
   884→
   885→    bool handles_class(const std::string& class_name) const override {
   886→        return class_name == "Ljava/util/ArrayList;" ||
   887→               class_name == "Ljava/util/LinkedList;" ||
   888→               class_name == "Ljava/util/List;" ||
   889→               class_name == "Ljava/util/Collection;" ||
   890→               class_name == "Ljava/util/CopyOnWriteArrayList;" ||
   891→               class_name == "Ljava/util/HashMap;" ||
   892→               class_name == "Ljava/util/Map;" ||
   893→               class_name == "Ljava/util/HashSet;" ||
   894→               class_name == "Ljava/util/Set;" ||
   895→               class_name == "Ljava/util/Arrays$ArrayList;" ||
   896→               class_name == "Ljava/util/Collections$UnmodifiableRandomAccessList;" ||
   897→               class_name == "Ljava/util/Collections$SingletonList;" ||
   898→               class_name == "Ljava/util/Iterator;" ||
   899→               class_name == "Ljava/util/ListIterator;" ||
   900→               class_name.find("/ArrayList;") != std::string::npos ||
   901→               class_name.find("/HashMap;") != std::string::npos ||
   902→               class_name.find("/HashSet;") != std::string::npos ||
   903→               class_name.find("ConcurrentHashMap") != std::string::npos;
   904→    }
   905→
   906→    CallResult dispatch(const CallContext& ctx) override;
   907→
   908→    std::vector<std::string> implemented_methods() const override {
   909→        return {"add", "get", "size", "isEmpty", "clear", "remove",
   910→                "contains", "iterator", "hasNext", "next",
   911→                "put", "containsKey", "keySet", "values", "entrySet",
   912→                "getIndex", "set"};
   913→    }
   914→    std::vector<std::string> stubbed_methods() const override {
   915→        return {"subList", "listIterator", "toArray", "sort"};
   916→    }
   917→
   918→    // Get or create CollectionState for a heap object.
   919→    CollectionState* get_or_create(uint32_t object_id, bool is_map = false);
   920→
   921→private:
   922→    std::map<uint32_t, CollectionState> collections_;
   923→};
