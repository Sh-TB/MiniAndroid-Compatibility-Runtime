    29→std::string value_type_to_string(ValueType type) {
    30→    switch (type) {
    31→        case ValueType::UNINITIALIZED: return "UNINITIALIZED";
    32→        case ValueType::INT32: return "INT32";
    33→        case ValueType::FLOAT: return "FLOAT";
    34→        case ValueType::LONG: return "LONG";
    35→        case ValueType::DOUBLE: return "DOUBLE";
    36→        case ValueType::STRING_REF: return "STRING_REF";
    37→        case ValueType::OBJECT_REF: return "OBJECT_REF";
    38→        case ValueType::NULL_REF: return "NULL_REF";
    39→        case ValueType::REGISTER_UNSET: return "REGISTER_UNSET";
    40→        case ValueType::BOOLEAN: return "BOOLEAN";
    41→        case ValueType::BYTE: return "BYTE";
    42→        case ValueType::SHORT: return "SHORT";
    43→        case ValueType::CHAR: return "CHAR";
    44→        default: return "UNKNOWN";
    45→    }
    46→}

   121→std::string Value::to_string() const {
   122→    switch (type) {
   123→        case ValueType::UNINITIALIZED:
   124→            return "<uninitialized>";
   125→        case ValueType::REGISTER_UNSET:
   126→            return "<unset>";
   127→        case ValueType::INT32:
   128→            return std::to_string(int_val);
   129→        case ValueType::FLOAT:
   130→            return std::to_string(float_val);
   131→        case ValueType::LONG:
   132→            return std::to_string(long_val) + "L";
   133→        case ValueType::DOUBLE:
   134→            return std::to_string(double_val);
   135→        case ValueType::STRING_REF:
   136→            return "\"" + string_val + "\" (ref:" + std::to_string(reference_id) + ")";
   137→        case ValueType::OBJECT_REF:
   138→            return object_class + " (obj:" + std::to_string(reference_id) + ")";
   139→        case ValueType::NULL_REF:
   140→            return "null";
   141→        case ValueType::BOOLEAN:
   142→            return bool_val ? "true" : "false";
   143→        case ValueType::BYTE:
   144→            return std::to_string(byte_val);
   145→        case ValueType::SHORT:
   146→            return std::to_string(short_val);
   147→        case ValueType::CHAR:
   148→            return std::string("'") + char_val + "'";
   149→        default:
   150→            return "<unknown>";
   151→    }
   152→}

   166→        case ValueType::OBJECT_REF:
   167→            j["class_name"] = object_class;
   168→            break;
   169→        case ValueType::BOOLEAN:

   171→            break;
   172→        case ValueType::NULL_REF:
   173→            j["is_null"] = true;
   174→            break;
   175→        default:
   176→            j["raw_value"] = to_string();
   177→    }
   178→    
   179→    // Lifetime tracking
   180→    j["defined_at_pc"] = defined_at_pc;
   181→    j["is_live"] = is_live;

   278→json InstructionTraceEntry::to_json() const {
   279→    json j;
   280→    
   281→    j["sequence"] = sequence;
   282→    j["pc_before"] = pc_before;
   283→    j["pc_after"] = pc_after;
   284→    
   285→    j["opcode"] = opcode_name;
   286→    j["opcode_hex"] = "0x" + ([&]() { 
   287→        std::stringstream ss; 
   288→        ss << std::hex << std::setw(4) << std::setfill('0') << opcode_hex; 
   289→        return ss.str(); 
   290→    })();
   291→    
   292→    // Operands
   293→    json ops = json::array();
   294→    for (const auto& op : operands) {
   295→        ops.push_back({
   296→            {"name", op.name},
   297→            {"value", op.value},
   298→            {"numeric_value", op.numeric_value}
   299→        });
   300→    }
   301→    j["operands"] = ops;
   302→    
   303→    // Resolved values
   304→    if (resolved_string.has_value()) {
   305→        j["resolved_string"] = resolved_string.value();
   306→    }
   307→    if (resolved_type.has_value()) {
   308→        j["resolved_type"] = resolved_type.value();
   309→    }
   310→    
   311→    // Execution status
   312→    std::string status_str;
   313→    switch (status) {
   314→        case Status::SUCCESS: status_str = "SUCCESS"; break;
   315→        case Status::HALT_UNIMPLEMENTED: status_str = "HALT_UNIMPLEMENTED"; break;
   316→        case Status::CRASH_INVALID_STATE: status_str = "CRASH_INVALID_STATE"; break;
   317→        case Status::HALT_EXPERIMENT_BOUNDARY: status_str = "HALT_EXPERIMENT_BOUNDARY"; break;
   318→        case Status::METHOD_RETURNED: status_str = "METHOD_RETURNED"; break;
   319→        case Status::API_CALL_DISPATCHED: status_str = "API_CALL_DISPATCHED"; break;
   320→    }
   321→    j["execution"] = {
   322→        {"status", status_str},
   323→        {"cycles", cycles}
   324→    };
   325→    
   326→    // State changes
   327→    if (!registers_written.empty()) {
   328→        j["state_change"] = {
   329→            {"registers_written", registers_written}
   330→        };
   331→        
   332→        json reg_snapshots = json::array();
   333→        for (const auto& reg : registers_written) {
   334→            auto it = register_snapshots.find(reg);
   335→            if (it != register_snapshots.end()) {
   336→                reg_snapshots.push_back({
   337→                    {"register", reg},
   338→                    {"value": it->second.to_json()}
   339→                });
   340→            }
   341→        }
   342→        j["state_change"]["register_values"] = reg_snapshots;
   343→    }
   344→    
   345→    // EXP-014 new: Object allocation
   346→    if (object_allocated.has_value()) {
   347→        j["object_allocation"] = object_allocated.value().to_json();
   348→    }
   349→    
   350→    // EXP-014 new: API call
   351→    if (api_call_made.has_value()) {
   352→        j["api_call"] = api_call_made.value().to_json();
   353→    }
   354→    
   355→    // Halt reason
   356→    if (halt_reason.has_value()) {
   357→        j["halt_reason"] = halt_reason.value();
   358→    }
   359→    
   360→    return j;
   361→}

   367→json InstructionTrace::to_json() const {
   368→    json j;
   369→    
   370→    j["experiment_id"] = experiment_id;
   371→    j["timestamp"] = timestamp;
   372→    
   373→    j["method_context"] = {
   374→        {"class", class_name},
   375→        {"method", method_name},
   376→        {"descriptor", method_descriptor}
   377→    };
   378→    
   379→    j["initial_state"] = {
   380→        {"program_counter", initial_pc}
   381→    };
   382→    
   383→    // Instructions
   384→    json instrs = json::array();
   385→    for (const auto& instr : instructions) {
   386→        instrs.push_back(instr.to_json());
   387→    }
   388→    j["instructions"] = instrs;
   389→    
   390→    // Final state
   391→    j["final_state"] = {
   392→        {"program_counter", final_pc},
   393→        {"halted", halted},
   394→        {"returned", returned},  // EXP-014 new
   395→        {"halt_reason", halt_reason}
   396→    };
   397→    
   398→    if (next_opcode_hex.has_value()) {
   399→        j["final_state"]["next_opcode_if_known"] = "0x" + ([&]() {
   400→            std::stringstream ss;
   401→            ss << std::hex << std::setw(4) << std::setfill('0') << next_opcode_hex.value();
   402→            return ss.str();
   403→        })();
   404→        j["final_state"]["next_opcode_name"] = next_opcode_name;
   405→    }
   406→    
   407→    // Statistics
   408→    j["statistics"] = {
   409→        {"total_instructions", total_instructions_in_method},
   410→        {"executed_instructions", executed_instructions},
   411→        {"success_rate", success_rate},
   412→        {"unimplemented_opcodes_encountered", unimplemented_opcodes_encountered},
   413→        // EXP-014 new statistics
   414→        {"objects_created", objects_created},
   415→        {"api_calls_dispatched", api_calls_dispatched},
   416→        {"real_execution_percentage", real_execution_percentage}
   417→    };
   418→    
   419→    // EXP-014 new: Register trace
   420→    j["register_trace"] = register_trace.to_json();
   421→    
   422→    // EXP-014 new: Allocations
   423→    json allocs = json::array();
   424→    for (const auto& alloc : allocations) {
   425→        allocs.push_back(alloc.to_json());
   426→    }
   427→    j["allocations"] = allocs;
   428→    
   429→    // EXP-014 new: API calls
   430→    json calls = json::array();
   431→    for (const auto& call : api_calls) {
   432→        calls.push_back(call.to_json());
   433→    }
   434→    j["api_calls"] = calls;
   435→    
   436→    return j;
   437→}

   504→
   505→void DexInterpreterV2::ensure_register_capacity(uint8_t reg) {
   506→    while (reg >= registers_.size()) {
   507→        registers_.push_back(Value::make_uninitialized());
   508→    }
   509→}
   510→
   511→void DexInterpreterV2::set_register(uint8_t reg, const Value& value, uint32_t pc) {
   512→    ensure_register_capacity(reg);
   513→    
   514→    // Mark previous value as no longer live at this PC
   515→    registers_[reg].last_used_at_pc = pc;
   516→    
   517→    // Set new value
   518→    registers_[reg] = value;
   519→    registers_[reg].defined_at_pc = pc;
   520→    registers_[reg].is_live = true;
   521→    
   522→    log("    [REG] v" + std::to_string(reg) + " ← " + value.to_string());
   523→}
   524→
   525→Value DexInterpreterV2::get_register(uint8_t reg) const {
   526→    if (reg < registers_.size()) {
   527→        return registers_[reg];
   528→    }
   529→    return Value::make_uninitialized();
   530→}

   549→// ============================================================================
   550→// Method Resolution and Dispatch (EXP-014)
   551→// ============================================================================
   552→
   553→ApiCallRecord* DexInterpreterV2::resolve_and_dispatch(
   554→    const std::string& class_name,
   555→    const std::string& method_name,
   556→    const std::string& descriptor,
   557→    DispatchType dispatch_type,
   558→    const std::vector<uint8_t>& arg_registers,
   559→    uint32_t pc,
   560→    const InterpreterConfigV2& config
   561→) {
   562→    auto* record = new ApiCallRecord();
   563→    record->sequence = api_call_sequence_++;
   564→    record->timestamp = get_timestamp();
   565→    record->class_name = class_name;
   566→    record->method_name = method_name;
   567→    record->descriptor = descriptor;
   568→    record->dispatch_type = dispatch_type;
   569→    record->caller_pc = pc;
   570→    record->was_dispatched_from_dex = true;
   571→    record->dispatch_path = "DEX_INTERPRETER_" + dispatch_type_to_string(dispatch_type);
   572→    
   573→    // Collect arguments from registers
   574→    for (uint8_t reg : arg_registers) {
   575→        record->arguments.push_back(get_register(reg));
   576→    }
   577→    
   578→    // Look up in API registry
   579→    if (config.api_registry) {
   580→        for (const auto& entry : *config.api_registry) {
   581→            if (entry.class_name == class_name && 
   582→                entry.method_name == method_name &&
   583→                entry.descriptor == descriptor) {
   584→                
   585→                // Found implementation!
   586→                log("    [API] Dispatching to " + class_name + "." + method_name + descriptor);
   587→                
   588→                if (entry.implementation) {
   589→                    try {
   590→                        Value result = entry.implementation(record->arguments, config.object_heap);
   591→                        record->result_value = result;
   592→                        record->success = true;
   593→                        record->result_summary = "Returned: " + result.to_string();
   594→                        
   595→                        // Store return value in result register if applicable
   596→                        // (For now, we'll handle this in the invoke-* implementations)
   597→                    } catch (const std::exception& e) {
   598→                        record->success = false;
   599→                        record->error_message = e.what();
   600→                        record->result_summary = "Exception: " + std::string(e.what());
   601→                    }
   602→                } else if (entry.status == ApiImplementationStatus::STUBBED ||
   603→                           entry.status == ApiImplementationStatus::SIMULATED) {
   604→                    record->success = true;
   605→                    record->result_summary = "STUBBED/SIMULATED - " + api_status_to_string(entry.status);
   606→                    log("    [API] STUBBED: " + class_name + "." + method_name);
   607→                } else {
   608→                    record->success = false;
   609→                    record->error_message = "API not implemented: " + class_name + "." + method_name;
   610→                    record->result_summary = "UNIMPLEMENTED";
   611→                    log("    [API] UNIMPLEMENTED: " + class_name + "." + method_name);
   612→                }
   613→                
   614→                api_calls_.push_back(*record);
   615→                return record;
   616→            }
   617→        }
   618→    }
   619→    
   620→    // Not found in registry
   621→    record->success = false;
   622→    record->error_message = "API not in registry: " + class_name + "." + method_name + descriptor;
   623→    record->result_summary = "MISSING_API";
   624→    log("    [API] MISSING: " + class_name + "." + method_name + descriptor);
   625→    
   626→    api_calls_.push_back(*record);
   627→    return record;
   628→}

   631→// Main Entry Points
   632→// ============================================================================
   633→
   634→InstructionTrace DexInterpreterV2::execute(
   635→    const MethodInfo& method,
   636→    const std::vector<std::string>& strings,
   637→    const InterpreterConfigV2& config
   638→) {
   639→    InstructionTrace trace;
   640→    trace.experiment_id = config.experiment_scope;
   641→    trace.timestamp = get_timestamp();
   642→    
   643→    // Store state
   644→    strings_ = strings;
   645→    bytecode_ = method.bytecode;
   646→    verbose_ = config.verbose;
   647→    code_offset_ = method.code_offset;
   648→    
   649→    // Initialize trace metadata
   650→    trace.class_name = method.defining_class;
   651→    trace.method_name = method.name;
   652→    trace.method_descriptor = method.descriptor;
   653→    trace.total_instructions_in_method = static_cast<uint32_t>(bytecode_.size());
   654→    
   655→    // Initialize VM state
   656→    pc_ = 0;
   657→    halted_ = false;
   658→    returned_ = false;
   659→    halt_reason_.clear();
   660→    instruction_sequence_ = 0;
   661→    next_ref_id_ = 1;
   662→    next_object_id_ = 100;
   663→    allocations_.clear();
   664→    api_calls_.clear();
   665→    
   666→    // Initialize register trace
   667→    register_trace_.experiment_id = config.experiment_scope;
   668→    register_trace_.timestamp = get_timestamp();
   669→    register_trace_.method_context = method.defining_class + "->" + method.name + method.descriptor;
   670→    
   671→    // Initialize registers based on method signature
   672→    // For onCreate(Bundle): v0=this, v1=savedInstanceState
   673→    uint16_t num_registers = 5;  // Allocate enough registers for our test case
   674→    if (!method.bytecode.empty()) {
   675→        // Ensure we have enough for the bytecode we're executing
   676→        num_registers = std::max(num_registers, static_cast<uint16_t>(10));
   677→    }
   678→    
   679→    registers_.resize(num_registers, Value::make_uninitialized());
   680→    
   681→    // Set up 'this' reference (v0) as an Activity object
   682→    if (config.object_heap) {
   683→        // In real execution, 'this' would be passed as a parameter
   684→        // For now, create a reference to represent the Activity
   685→        registers_[0] = Value::make_object("Landroid/app/Activity;", 1, 0);
   686→        registers_[0].is_live = true;
   687→    }
   688→    
   689→    // Record initial state
   690→    trace.initial_pc = pc_;
   691→    for (uint8_t i = 0; i < registers_.size(); i++) {
   692→        trace.initial_registers[register_name(i)] = registers_[i];
   693→    }
   694→    register_trace_.initial_state.pc = pc_;
   695→    register_trace_.initial_state.instruction_sequence = 0;
   696→    for (uint8_t i = 0; i < registers_.size(); i++) {
   697→        register_trace_.initial_state.register_values[i] = registers_[i];
   698→    }
   699→    
   700→    log("Starting execution of " + method.name + method.descriptor);
   701→    log("Bytecode size: " + std::to_string(bytecode_.size()) + " bytes");
   702→    log("Registers allocated: " + std::to_string(num_registers));
   703→    
   704→    // Execute
   705→    bool success = fetch_decode_execute(trace, config);
   706→    
   707→    // Record final state
   708→    trace.final_pc = pc_;
   709→    trace.halted = halted_;
   710→    trace.returned = returned_;
   711→    trace.halt_reason = halt_reason_;
   712→    trace.executed_instructions = instruction_sequence_;
   713→    trace.objects_created = static_cast<uint32_t>(allocations_.size());
   714→    trace.api_calls_dispatched = static_cast<uint32_t>(api_calls_.size());
   715→    trace.allocations = allocations_;
   716→    trace.api_calls = api_calls_;
   717→    
   718→    // Calculate success rate
   719→    if (returned_) {
   720→        trace.success_rate = std::to_string(trace.executed_instructions) + "/" + 
   721→                             std::to_string(trace.executed_instructions) + " (100% - RETURNED)";
   722→        trace.real_execution_percentage = 100;
   723→    } else if (trace.executed_instructions > 0 && !halted_) {
   724→        trace.success_rate = std::to_string(trace.executed_instructions) + "/" + 
   725→                             std::to_string(trace.executed_instructions) + " (100%)";
   726→        trace.real_execution_percentage = 100;
   727→    } else if (trace.executed_instructions > 0) {
   728→        trace.real_execution_percentage = (trace.executed_instructions * 100) / trace.total_instructions_in_method;
   729→        trace.success_rate = std::to_string(trace.executed_instructions) + "/" + 
   730→                             std::to_string(trace.total_instructions_in_method) + " (" +
   731→                             std::to_string(trace.real_execution_percentage) + "%)";
   732→    } else {
   733→        trace.success_rate = "0/" + std::to_string(trace.total_instructions_in_method) + " (0%)";
   734→        trace.real_execution_percentage = 0;
   735→    }
   736→    
   737→    // Final register snapshot
   738→    register_trace_.final_state.pc = pc_;
   739→    register_trace_.final_state.instruction_sequence = instruction_sequence_;
   740→    for (uint8_t i = 0; i < registers_.size(); i++) {
   741→        register_trace_.final_state.register_values[i] = registers_[i];
   742→    }
   743→    trace.register_trace = register_trace_;
   744→    
   745→    log("Execution complete. Instructions executed: " + std::to_string(trace.executed_instructions));
   746→    log("Objects created: " + std::to_string(trace.objects_created));
   747→    log("API calls dispatched: " + std::to_string(trace.api_calls_dispatched));
   748→    if (halted_) {
   749→        log("Halted reason: " + halt_reason_);
   750→    } else if (returned_) {
   751→        log("Method returned normally");
   752→    }
   753→    
   754→    return trace;
   755→}
   756→
   757→InstructionTrace DexInterpreterV2::execute_entry_point(
   758→    const EntryPoint& entry,
   759→    const DexReport& dex_report,
   760→    const InterpreterConfigV2& config
   761→) {
   762→    if (!entry.resolved || !entry.has_bytecode) {
   763→        InstructionTrace empty_trace;
   764→        empty_trace.experiment_id = config.experiment_scope;
   765→        empty_trace.halted = true;
   766→        empty_trace.halt_reason = "Entry point not resolved or no bytecode available";
   767→        return empty_trace;
   768→    }
   769→    
   770→    return execute(entry.method_info, dex_report.strings, config);
   771→}

   773→// ============================================================================
   774→// Core Execution Loop
   775→// ============================================================================
   776→
   777→bool DexInterpreterV2::fetch_decode_execute(InstructionTrace& trace, const InterpreterConfigV2& config) {
   778→    while (!halted_ && !returned_ && pc_ < bytecode_.size()) {
   779→        // Infinite loop protection
   780→        if (instruction_sequence_ >= config.max_instructions) {
   781→            halted_ = true;
   782→            halt_reason_ = "INFINITE_LOOP_DETECTED - exceeded max instructions (" + 
   783→                          std::to_string(config.max_instructions) + ")";
   784→            
   785→            InstructionTraceEntry entry;
   786→            entry.sequence = instruction_sequence_;
   787→            entry.pc_before = pc_;
   788→            entry.status = InstructionTraceEntry::Status::HALT_EXPERIMENT_BOUNDARY;
   789→            entry.opcode_name = "(limit reached)";
   790→            entry.halt_reason = halt_reason_;
   791→            add_instruction_to_trace(trace, entry);
   792→            
   793→            return false;
   794→        }
   795→        
   796→        // Fetch current opcode
   797→        uint16_t opcode = fetch_opcode(pc_);
   798→        
   799→        InstructionTraceEntry entry;
   800→        entry.sequence = instruction_sequence_;
   801→        entry.pc_before = pc_;
   802→        entry.opcode_hex = opcode;
   803→        
   804→        log("PC=" + std::to_string(pc_) + ": Opcode 0x" + 
   805→            ([&]() { std::stringstream ss; ss << std::hex << std::setw(4) << std::setfill('0') << opcode; return ss.str(); })());
   806→        
   807→        // Execute based on opcode
   808→        bool executed = false;
   809→        
   810→        switch (opcode) {
   811→            // Already implemented
   812→            case Opcodes::CONST_STRING:
   813→                executed = execute_const_string(pc_, entry);
   814→                break;
   815→                
   816→            // NEW: Return opcodes
   817→            case Opcodes::RETURN_VOID:
   818→                executed = execute_return_void(pc_, entry);
   819→                break;
   820→            case Opcodes::RETURN_OBJECT:
   821→                executed = execute_return_object(pc_, entry);
   822→                break;
   823→                
   824→            // NEW: Object creation
   825→            case Opcodes::NEW_INSTANCE:
   826→                executed = execute_new_instance(pc_, entry, config);
   827→                break;
   828→                
   829→            // NEW: Method invocation
   830→            case Opcodes::INVOKE_DIRECT:
   831→                executed = execute_invoke_direct(pc_, entry, config);
   832→                break;
   833→            case Opcodes::INVOKE_VIRTUAL:
   834→                executed = execute_invoke_virtual(pc_, entry, config);
   835→                break;
   836→                
   837→            // NEW: Move operations
   838→            case Opcodes::MOVE_OBJECT:
   839→                executed = execute_move_object(pc_, entry);
   840→                break;
   841→                
   842→            // NEW: Constant loading
   843→            case Opcodes::CONST_4:
   844→                executed = execute_const_4(pc_, entry);
   845→                break;
   846→                
   847→            // NEW: Field operations
   848→            case Opcodes::IGET_OBJECT:
   849→                executed = execute_iget_object(pc_, entry);
   850→                break;
   851→            case Opcodes::IPUT_OBJECT:
   852→                executed = execute_iput_object(pc_, entry);
   853→                break;
   854→                
   855→            default:
   856→                // Unimplemented opcode
   857→                handle_unimplemented(opcode, pc_, entry);
   858→                break;
   859→        }
   860→        
   861→        // Update PC from execution result
   862→        if (!halted_ && !returned_) {
   863→            pc_ = entry.pc_after;
   864→        }
   865→        
   866→        // Add to trace
   867→        add_instruction_to_trace(trace, entry);
   868→        
   869→        // Capture register snapshot
   870→        capture_register_snapshot(register_trace_, pc_);
   871→        
   872→        // Increment sequence counter
   873→        instruction_sequence_++;
   874→        
   875→        // Check if we should stop
   876→        if (halted_ && config.stop_on_unimplemented) {
   877→            log("Halting due to: " + halt_reason_);
   878→            break;
   879→        }
   880→    }
   881→    
   882→    // If we exited loop without halting or returning
   883→    if (!halted_ && !returned_) {
   884→        // Check if we reached end of bytecode
   885→        if (pc_ >= bytecode_.size()) {
   886→            returned_ = true;
   887→            halt_reason_ = "END_OF_BYTECODE_REACHED";
   888→            trace.halt_reason = halt_reason_;
   889→        }
   890→    }
   891→    
   892→    return true;
   893→}
   894→
   895→uint16_t DexInterpreterV2::fetch_opcode(uint32_t pc) {
   896→    if (pc >= bytecode_.size()) {
   897→        return 0xFFFF;  // Invalid opcode marker
   898→    }
   899→    return bytecode_[pc];
   900→}

   902→// ============================================================================
   903→// Opcode Implementation: const-string (0x1A) - From EXP-003-A
   904→// ============================================================================
   905→
   906→bool DexInterpreterV2::execute_const_string(uint32_t pc, InstructionTraceEntry& entry) {
   907→    if (pc + 2 >= bytecode_.size()) {
   908→        entry.status = InstructionTraceEntry::Status::CRASH_INVALID_STATE;
   909→        entry.halt_reason = "Insufficient bytes for const-string instruction";
   910→        halted_ = true;
   911→        halt_reason_ = entry.halt_reason.value();
   912→        last_error_ = "const-string: insufficient bytecode at PC=" + std::to_string(pc);
   913→        return false;
   914→    }
   915→    
   916→    uint16_t first_word = bytecode_[pc];
   917→    uint8_t vAA = static_cast<uint8_t>((first_word >> 8) & 0xFF);
   918→    uint16_t BBBB = bytecode_[pc + 1];
   919→    
   920→    entry.opcode_name = "const-string";
   921→    entry.operands.push_back({"destination_register", register_name(vAA)});
   922→    entry.operands.push_back({"string_index", std::to_string(BBBB)});
   923→    
   924→    if (vAA >= registers_.size()) {
   925→        entry.status = InstructionTraceEntry::Status::CRASH_INVALID_STATE;
   926→        entry.halt_reason = "Register v" + std::to_string(vAA) + " out of bounds";
   927→        halted_ = true;
   928→        halt_reason_ = entry.halt_reason.value();
   929→        return false;
   930→    }
   931→    
   932→    if (BBBB >= strings_.size()) {
   933→        entry.status = InstructionTraceEntry::Status::CRASH_INVALID_STATE;
   934→        entry.halt_reason = "String index out of bounds";
   935→        halted_ = true;
   936→        halt_reason_ = entry.halt_reason.value();
   937→        return false;
   938→    }
   939→    
   940→    std::string resolved_string = strings_[BBBB];
   941→    entry.resolved_string = resolved_string;
   942→    
   943→    log("  const-string v" + std::to_string(vAA) + ", \"" + resolved_string + "\"");
   944→    
   945→    Value value = Value::make_string(resolved_string, next_ref_id_++, pc);
   946→    set_register(vAA, value, pc);
   947→    
   948→    entry.registers_written.push_back(register_name(vAA));
   949→    entry.register_snapshots[register_name(vAA)] = value;
   950→    
   951→    entry.status = InstructionTraceEntry::Status::SUCCESS;
   952→    entry.pc_after = pc + 3;
   953→    
   954→    return true;
   955→}

   978→// Opcode Implementation: return-object (0x11) - NEW in EXP-014
   979→// ============================================================================
   980→
   981→bool DexInterpreterV2::execute_return_object(uint32_t pc, InstructionTraceEntry& entry) {
   982→    if (pc + 1 >= bytecode_.size()) {

   991→

  1006→    
  1007→    return true;

  1010→// ============================================================================
  1011→// Opcode Implementation: new-instance (0x22) - NEW in EXP-014 P0 CRITICAL
  1012→// ============================================================================
  1013→
  1014→bool DexInterpreterV2::execute_new_instance(uint32_t pc, InstructionTraceEntry& entry, 
  1015→                                          const InterpreterConfigV2& config) {
  1016→    if (pc + 2 >= bytecode_.size()) {
  1017→        entry.status = InstructionTraceEntry::Status::CRASH_INVALID_STATE;
  1018→        entry.halt_reason = "Insufficient bytes for new-instance";
  1019→        halted_ = true;
  1020→        halt_reason_ = entry.halt_reason.value();
  1021→        return false;
  1022→    }
  1023→    
  1024→    // Decode: new-instance vAA, type@BBBB
  1025→    uint16_t first_word = bytecode_[pc];
  1026→    uint8_t vAA = static_cast<uint8_t>((first_word >> 8) & 0xFF);  // Destination register
  1027→    uint16_t type_idx = bytecode_[pc + 1];  // Type index
  1028→    
  1029→    entry.opcode_name = "new-instance";
  1030→    entry.operands.push_back({"destination_register", register_name(vAA)});
  1031→    entry.operands.push_back({"type_index", std::to_string(type_idx)});
  1032→    
  1033→    // Validate register
  1034→    if (vAA >= registers_.size()) {
  1035→        ensure_register_capacity(vAA);
  1036→    }
  1037→    
  1038→    // Resolve type from DEX report
  1039→    std::string type_descriptor = "?";
  1040→    std::string resolved_class = "?";
  1041→    
  1042→    if (config.dex_report && type_idx < config.dex_report->types.size()) {
  1043→        type_descriptor = config.dex_report->types[type_idx];
  1044→        resolved_class = type_descriptor;
  1045→        // Convert descriptor to readable name
  1046→        if (resolved_class.size() > 2 && resolved_class[0] == 'L' && resolved_class.back() == ';') {
  1047→            resolved_class = resolved_class.substr(1, resolved_class.size() - 2);
  1048→            std::replace(resolved_class.begin(), resolved_class.end(), '/', '.');
  1049→        }
  1050→        entry.resolved_type = resolved_class;
  1051→    }
  1052→    
  1053→    log("  new-instance v" + std::to_string(vAA) + ", " + type_descriptor);
  1054→    
  1055→    // ALLOCATE OBJECT via object heap
  1056→    uint32_t obj_id = allocate_object(type_descriptor, pc);
  1057→    
  1058→    // Create allocation record
  1059→    ObjectAllocationRecord alloc_record;
  1060→    alloc_record.sequence = allocation_sequence_++;
  1061→    alloc_record.pc = pc;
  1062→    alloc_record.opcode = "new-instance";
  1063→    alloc_record.type_descriptor = type_descriptor;
  1064→    alloc_record.resolved_class_name = resolved_class;
  1065→    alloc_record.object_id = obj_id;
  1066→    alloc_record.destination_register = vAA;
  1067→    alloc_record.allocation_success = true;
  1068→    alloc_record.allocated_via_interpreter = true;  // KEY: Real DEX allocation!
  1069→    alloc_record.allocation_source = "DEX_NEW_INSTANCE";
  1070→    
  1071→    // Set register to object reference
  1072→    Value obj_value = Value::make_object(resolved_class, obj_id, pc);
  1073→    set_register(vAA, obj_value, pc);
  1074→    
  1075→    entry.registers_written.push_back(register_name(vAA));
  1076→    entry.register_snapshots[register_name(vAA)] = obj_value;
  1077→    entry.object_allocated = alloc_record;
  1078→    
  1079→    allocations_.push_back(alloc_record);
  1080→    
  1081→    entry.status = InstructionTraceEntry::Status::SUCCESS;
  1082→    entry.pc_after = pc + 3;  // new-instance is 3 code units wide
  1083→    
  1084→    log("  → v" + std::to_string(vAA) + " = " + resolved_class + " (obj:" + std::to_string(obj_id) + ")");
  1085→    
  1086→    return true;
  1087→}

  1089→// ============================================================================
  1090→// Opcode Implementation: invoke-direct (0x70) - NEW in EXP-014 P0 CRITICAL
  1091→// ============================================================================
  1092→
  1093→bool DexInterpreterV2::execute_invoke_direct(uint32_t pc, InstructionTraceEntry& entry,
  1094→                                           const InterpreterConfigV2& config) {
  1095→    if (pc + 2 >= bytecode_.size()) {
  1096→        entry.status = InstructionTraceEntry::Status::CRASH_INVALID_STATE;
  1097→        entry.halt_reason = "Insufficient bytes for invoke-direct";
  1098→        halted_ = true;
  1099→        halt_reason_ = entry.halt_reason.value();
  1100→        return false;
  1101→    }
  1102→    
  1103→    // Decode: invoke-direct {vC, vD, vE, vF, vG}, method@BBBB
  1104→    // Format: 35c
  1105→    uint16_t first_word = bytecode_[pc];
  1106→    uint8_t arg_count = (first_word & 0xF);       // Number of arguments
  1107→    uint8_t arg_reg_start = ((first_word >> 4) & 0xF);  // First argument register
  1108→    uint16_t method_idx = bytecode_[pc + 1];      // Method index
  1109→    
  1110→    entry.opcode_name = "invoke-direct";
  1111→    entry.operands.push_back({"method_index", std::to_string(method_idx)});
  1112→    entry.operands.push_back({"arg_count", std::to_string(arg_count)});
  1113→    entry.operands.push_back({"args_start_register", "v" + std::to_string(arg_reg_start)});
  1114→    
  1115→    // Collect argument registers
  1116→    std::vector<uint8_t> arg_regs;
  1117→    for (uint8_t i = 0; i <= arg_count; i++) {  // +1 for 'this'
  1118→        arg_regs.push_back(static_cast<uint8_t>(arg_reg_start + i));
  1119→        entry.operands.push_back({
  1120→            "arg_v" + std::to_string(i), 
  1121→            register_name(static_cast<uint8_t>(arg_reg_start + i))
  1122→        });
  1123→    }
  1124→    
  1125→    // Resolve method from DEX report
  1126→    std::string class_name = "?";
  1127→    std::string method_name = "?";
  1128→    std::string descriptor = "?";
  1129→    
  1130→    if (config.dex_report && method_idx < config.dex_report->methods.size()) {
  1131→        const auto& method = config.dex_report->methods[method_idx];
  1132→        class_name = method.defining_class;
  1133→        method_name = method.name;
  1134→        descriptor = method.descriptor;
  1135→

  1137→        if (class_name.size() > 2 && class_name[0] == 'L' && class_name.back() == ';') {
  1138→            std::string temp = class_name.substr(1, class_name.size() - 2);

  1141→        }

  1147→    auto* api_record = resolve_and_dispatch(
  1148→        class_name, method_name, descriptor,

  1155→            InstructionTraceEntry::Status::API_CALL_DISPATCHED :

  1157→

  1167→// Opcode Implementation: invoke-virtual (0x6E) - NEW in EXP-014 P0 CRITICAL
  1168→// ============================================================================
  1169→
  1170→bool DexInterpreterV2::execute_invoke_virtual(uint32_t pc, InstructionTraceEntry& entry,
  1171→                                            const InterpreterConfigV2& config) {
  1172→    if (pc + 2 >= bytecode_.size()) {
  1173→        entry.status = InstructionTraceEntry::Status::CRASH_INVALID_STATE;
  1174→        entry.halt_reason = "Insufficient bytes for invoke-virtual";
  1175→        halted_ = true;
  1176→        halt_reason_ = entry.halt_reason.value();
  1177→        return false;
  1178→    }
  1179→    
  1180→    // Same format as invoke-direct: 35c
  1181→    uint16_t first_word = bytecode_[pc];
  1182→    uint8_t arg_count = (first_word & 0xF);
  1183→    uint8_t arg_reg_start = ((first_word >> 4) & 0xF);
  1184→    uint16_t method_idx = bytecode_[pc + 1];
  1185→    
  1186→    entry.opcode_name = "invoke-virtual";
  1187→    entry.operands.push_back({"method_index", std::to_string(method_idx)});
  1188→    entry.operands.push_back({"arg_count", std::to_string(arg_count)});
  1189→    entry.operands.push_back({"args_start_register", "v" + std::to_string(arg_reg_start)});
  1190→    
  1191→    // Collect argument registers
  1192→    std::vector<uint8_t> arg_regs;
  1193→    for (uint8_t i = 0; i <= arg_count; i++) {
  1194→        arg_regs.push_back(static_cast<uint8_t>(arg_reg_start + i));
  1195→        entry.operands.push_back({
  1196→            "arg_v" + std::to_string(i),
  1197→            register_name(static_cast<uint8_t>(arg_reg_start + i))
  1198→        });
  1199→    }
  1200→    
  1201→    // Resolve method
  1202→    std::string class_name = "?";
  1203→    std::string method_name = "?";
  1204→    std::string descriptor = "?";
  1205→    
  1206→    if (config.dex_report && method_idx < config.dex_report->methods.size()) {
  1207→        const auto& method = config.dex_report->methods[method_idx];
  1208→        class_name = method.defining_class;
  1209→        method_name = method.name;
  1210→        descriptor = method.descriptor;
  1211→        
  1212→        if (class_name.size() > 2 && class_name[0] == 'L' && class_name.back() == ';') {
  1213→            std::string temp = class_name.substr(1, class_name.size() - 2);
  1214→            std::replace(temp.begin(), temp.end(), '/', '.');
  1215→            entry.resolved_string = temp + "." + method_name + descriptor;
  1216→        }
  1217→    }
  1218→    
  1219→    // For virtual calls, get actual class from 'this' object
  1220→    Value this_obj = get_register(arg_regs[0]);
  1221→    if (this_obj.type == ValueType::OBJECT_REF) {
  1222→        class_name = this_obj.object_class;
  1223→        log("  [virtual] Resolved this to: " + class_name);
  1224→    }
  1225→    
  1226→    log("  invoke-virtual " + class_name + "." + method_name + descriptor);
  1227→    
  1228→    // DISPATCH through API registry
  1229→    auto* api_record = resolve_and_dispatch(
  1230→        class_name, method_name, descriptor,
  1231→        DispatchType::VIRTUAL, arg_regs, pc, config
  1232→    );
  1233→    
  1234→    if (api_record) {
  1235→        entry.api_call_made = *api_record;
  1236→        entry.status = api_record->success ?
  1237→            InstructionTraceEntry::Status::API_CALL_DISPATCHED :
  1238→            InstructionTraceEntry::Status::SUCCESS;
  1239→        
  1240→        delete api_record;
  1241→    }
  1242→    
  1243→    entry.pc_after = pc + 3;
  1244→    
  1245→    return true;
  1246→}

  1262→    uint8_t vA = bytecode_[pc + 1] & 0xF;   // Destination (low nibble)

  1275→

  1277→    entry.register_snapshots[register_name(vA)] = source_val;

  1281→

  1400→// Unimplemented Opcode Handler
  1401→// ============================================================================
  1402→
  1403→void DexInterpreterV2::handle_unimplemented(uint16_t opcode, uint32_t pc, InstructionTraceEntry& entry) {
  1404→    std::string opcode_name;
  1405→    switch (opcode) {
  1406→        case Opcodes::NEW_ARRAY: opcode_name = "new-array"; break;
  1407→        case Opcodes::INVOKE_SUPER: opcode_name = "invoke-super"; break;
  1408→        case Opcodes::INVOKE_STATIC: opcode_name = "invoke-static"; break;
  1409→        case Opcodes::INVOKE_INTERFACE: opcode_name = "invoke-interface"; break;
  1410→        case Opcodes::RETURN: opcode_name = "return"; break;
  1411→        case Opcodes::MOVE: opcode_name = "move"; break;
  1412→        case Opcodes::MOVE_FROM16: opcode_name = "move/from16"; break;
  1413→        case Opcodes::INSTANCE_OF: opcode_name = "instance-of"; break;
  1414→        case Opcodes::CHECK_CAST: opcode_name = "check-cast"; break;
  1415→        default: 
  1416→            opcode_name = "unknown(0x" + ([&]() { 
  1417→                std::stringstream ss; ss << std::hex << std::setw(4) << std::setfill('0') << opcode; return ss.str(); 
  1418→            })() + ")";
  1419→            break;
  1420→    }
  1421→    
  1422→    entry.opcode_name = opcode_name;
  1423→    entry.status = InstructionTraceEntry::Status::HALT_UNIMPLEMENTED;
  1424→    entry.pc_after = pc;
  1425→    entry.halt_reason = "UNIMPLEMENTED_OPCODE: " + opcode_name + " (0x" + 
  1426→                        ([&]() { std::stringstream ss; ss << std::hex << std::setw(4) << std::setfill('0') << opcode; return ss.str(); })() + ")";
  1427→    
  1428→    entry.operands.push_back({"status", "NOT_IMPLEMENTED"});
  1429→    
  1430→    // Only halt if configured to stop on unimplemented
  1431→    // In EXP-014, we may want to continue to see what else would execute
  1432→    halted_ = true;
  1433→    halt_reason_ = entry.halt_reason.value();
  1434→    
  1435→    log("  ⛔ UNIMPLEMENTED: " + opcode_name + " at PC=" + std::to_string(pc));
  1436→}
