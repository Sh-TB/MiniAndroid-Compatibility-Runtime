   843→   843→// ─────────────────────────────────────────────────────────────────────────
   844→   844→// CollectionShadow — real List/ArrayList/Map semantics with per-instance state.
   845→   845→//
   846→   846→// Instead of a global `List.isEmpty → true` stub, this shadow tracks real
   847→   847→// collection state per heap object. Each collection instance gets a
   848→   848→// CollectionState (vector of elements) keyed by object_id.
   849→   849→//
   850→   850→// Supported operations:
   851→   851→//   * List.add(item) → append, return true
   852→   852→//   * List.get(index) → return element at index
   853→   853→//   * List.size() → return count
   854→   854→//   * List.isEmpty() → return count == 0
   855→   855→//   * List.clear() → clear elements
   856→   856→//   * List.remove(index) → remove at index
   857→   857→//   * Map.put(key, value) → store key-value pair
   858→   858→//   * Map.get(key) → return value for key
   859→   859→//   * Map.size() → return count
   860→   860→//   * Map.isEmpty() → return count == 0
   861→   861→//   * Map.containsKey(key) → return true if key exists
   862→   862→//   * Iterator.hasNext() → check position < size
   863→   863→//   * Iterator.next() → return element at position++
   864→   864→//
   865→   865→// This is NOT a stub — it's real state-based semantics. The runtime's
   866→   866→// List objects actually hold elements and report correct sizes.
   867→   867→// ─────────────────────────────────────────────────────────────────────────
   868→   868→class CollectionShadow : public Shadow {
   869→   869→public:
   870→   870→    struct CollectionState {
   871→   871→        std::vector<uint32_t> elements;  // object_ids of elements
   872→   872→        std::map<std::string, uint32_t> map_entries;  // key → value object_id
   873→   873→        // EXP-071 Phase 7: Store string values for HashMap.put(key, String).
   874→   874→        // The original map_entries only stores object_ids, but many HashMap
   875→   875→        // usages store String values (e.g., shortname → country name).
   876→   876→        // Without this, HashMap.get returns null for string values.
   877→   877→        std::map<std::string, std::string> map_string_entries;  // key → string value
   878→   878→        size_t iterator_position = 0;
   879→   879→        bool is_map = false;
   880→   880→    };
   881→   881→
   882→   882→    std::string name() const override { return "Collection"; }
   883→   883→    void init(HeapAllocator* heap) override { heap_ = heap; }
   884→   884→
   885→   885→    bool handles_class(const std::string& class_name) const override {
   886→   886→        return class_name == "Ljava/util/ArrayList;" ||
   887→   887→               class_name == "Ljava/util/LinkedList;" ||
   888→   888→               class_name == "Ljava/util/List;" ||
   889→   889→               class_name == "Ljava/util/Collection;" ||
   890→   890→               class_name == "Ljava/util/CopyOnWriteArrayList;" ||
   891→   891→               class_name == "Ljava/util/HashMap;" ||
   892→   892→               class_name == "Ljava/util/Map;" ||
   893→   893→               class_name == "Ljava/util/HashSet;" ||
   894→   894→               class_name == "Ljava/util/Set;" ||
   895→   895→               class_name == "Ljava/util/Arrays$ArrayList;" ||
   896→   896→               class_name == "Ljava/util/Collections$UnmodifiableRandomAccessList;" ||
   897→   897→               class_name == "Ljava/util/Collections$SingletonList;" ||
   898→   898→               class_name == "Ljava/util/Iterator;" ||
   899→   899→               class_name == "Ljava/util/ListIterator;" ||
   900→   900→               class_name.find("/ArrayList;") != std::string::npos ||
   901→   901→               class_name.find("/HashMap;") != std::string::npos ||
   902→   902→               class_name.find("/HashSet;") != std::string::npos ||
   903→   903→               class_name.find("ConcurrentHashMap") != std::string::npos;
   904→   904→    }
   905→   905→
   906→   906→    CallResult dispatch(const CallContext& ctx) override;
   907→   907→
   908→   908→    std::vector<std::string> implemented_methods() const override {
   909→   909→        return {"add", "get", "size", "isEmpty", "clear", "remove",
   910→   910→                "contains", "iterator", "hasNext", "next",

   916→   916→    }
   917→   917→
   918→   918→    // Get or create CollectionState for a heap object.
   919→   919→    CollectionState* get_or_create(uint32_t object_id, bool is_map = false);
   920→   920→
   921→   921→private:
   922→   922→    std::map<uint32_t, CollectionState> collections_;
   923→   923→};
